  1600→      * URL (with ``${var}`` normalised to ``{var}`` and ``/api`` prepended)
  1601→      * Response type (raw text)
  1602→    """
  1603→    endpoints: List[TSEndpoint] = []
  1604→    # We need to also capture the surrounding API group and method name
  1605→    # for context. Walk the file looking for `export const xxxApi = {`
  1606→    # blocks and attribute each api.* call to the most recently seen
  1607→    # group.
  1608→    current_group = ''
  1609→    current_method = ''
  1610→    lines = content.splitlines()
  1611→    # Build a map of line -> (group, method) for context attribution.
  1612→    context_by_line: Dict[int, Tuple[str, str]] = {}
  1613→    line_no = 0
  1614→    in_group = False
  1615→    for line in lines:
  1616→        line_no += 1
  1617→        # `export const authApi = {`
  1618→        gm = re.match(r'^\s*export\s+const\s+(\w+Api)\s*=\s*\{', line)
  1619→        if gm:
  1620→            current_group = gm.group(1)
  1621→            current_method = ''
  1622→            in_group = True
  1623→            continue
  1624→        if in_group:
  1625→            # `methodName: (args) =>` — capture method name.
  1626→            mm = re.match(r'^\s*(\w+)\s*:\s*(?:async\s*)?\(', line)
  1627→            if mm:
  1628→                current_method = mm.group(1)
  1629→            # Detect end of group: a line with just `};` at column 0.
  1630→            if re.match(r'^\};?\s*$', line):
  1631→                in_group = False
  1632→                current_group = ''
  1633→                current_method = ''
  1634→            else:
  1635→                context_by_line[line_no] = (current_group, current_method)
  1636→
  1637→    for m in RE_API_CALL.finditer(content):
  1638→        method = m.group('method').upper()
  1639→        type_str = (m.group('type') or '').strip()
  1640→        line = line_of(content, m.start())
  1641→        # The URL follows the open paren.
  1642→        rest = content[m.end():]
  1643→        url = _extract_url(rest)
  1644→        if url is None:
  1645→            continue
  1646→        normalised = _normalise_ts_url(url)
  1647→        # Classify the response type.
  1648→        response_kind, response_fields = _classify_ts_response(type_str)
  1649→        group, method_name = context_by_line.get(line, ('', ''))
  1650→        endpoints.append(TSEndpoint(
  1651→            method=method,
  1652→            path=normalised,
  1653→            response_type=type_str or '(unknown)',
  1654→            response_kind=response_kind,
  1655→            response_fields=response_fields,
  1656→            file=file,
  1657→            line=line,
  1658→            api_group=group,
  1659→            method_name=method_name,
  1660→        ))
  1661→    return endpoints
  1662→

  1664→def _extract_url(rest: str) -> Optional[str]:
  1665→    """Extract the URL string (or template literal) from the start of
  1666→    ``rest`` (the text immediately after ``api.method<TYPE>(``).
  1667→    """
  1668→    # Try a simple quoted string first.
  1669→    m = RE_URL_STRING.match(rest)
  1670→    if m:
  1671→        return m.group('url')
  1672→    # Try a template literal.
  1673→    m = RE_URL_TEMPLATE.match(rest)
  1674→    if m:
  1675→        return m.group('url')
  1676→    # Backtick template that may span multiple lines — find the next
  1677→    # backtick.
  1678→    if rest.lstrip().startswith('`'):
  1679→        idx = rest.find('`')
  1680→        end = rest.find('`', idx + 1)
  1681→        if end != -1:
  1682→            return rest[idx + 1:end]
  1683→    return None

  1686→def _normalise_ts_url(url: str) -> str:
  1687→    """Normalise a TS URL to match the Go route form.
  1688→
  1689→    * Convert ``${var}`` to ``{var}`` (gorilla/mux syntax).
  1690→    * Prepend ``/api`` if not already present.
  1691→    * Collapse duplicate slashes.
  1692→    """
  1693→    # Replace ${var} with {var}.
  1694→    url = re.sub(r'\$\{(\w+)\}', r'{\1}', url)
  1695→    # Prepend /api if missing.
  1696→    if not url.startswith('/api'):
  1697→        if url.startswith('/'):
  1698→            url = '/api' + url
  1699→        else:
  1700→            url = '/api/' + url
  1701→    # Collapse duplicate slashes (but keep the leading double-slash if any).
  1702→    url = re.sub(r'/+', '/', url)
  1703→    return url

  1706→def _classify_ts_response(type_str: str) -> Tuple[str, List[TSField]]:
  1707→    """Classify a TS response type string and parse inline fields.
  1708→
  1709→    Returns ``(kind, fields)`` where kind is one of
  1710→    ``"named"``, ``"inline"``, ``"unknown"``.
  1711→    """
  1712→    if not type_str:
  1713→        return ('unknown', [])
  1714→    t = type_str.strip()
  1715→    # Strip a union like `AuthResponse | MFARequiredResponse` — take
  1716→    # the first named branch.
  1717→    if '|' in t and not t.startswith('{'):
  1718→        bits = [b.strip() for b in t.split('|')]
  1719→        for b in bits:
  1720→            if re.match(r'^[A-Z]\w*$', b):
  1721→                return ('named', [])
  1722→        return ('unknown', [])
  1723→    # Inline object literal: `{ user: User; memberships: MembershipInfo[] }`.
  1724→    if t.startswith('{'):
  1725→        # Parse the fields.
  1726→        body = t[1:]
  1727→        # Find the matching closing brace (the type string may have
  1728→        # been truncated by the `<...>` regex if it contained `>` —
  1729→        # best-effort: strip a trailing `}` if present).
  1730→        if body.endswith('}'):
  1731→            body = body[:-1]
  1732→        fields = _parse_ts_fields(body)
  1733→        return ('inline', fields)
  1734→    # Named type.
  1735→    if re.match(r'^[A-Z]\w*$', t):
  1736→        return ('named', [])
  1737→    return ('unknown', [])
  1738→

  1744→# Go primitive types that map to TS `number`.
  1745→GO_NUMERIC_TYPES = {
  1746→    'int', 'int8', 'int16', 'int32', 'int64',
  1747→    'uint', 'uint8', 'uint16', 'uint32', 'uint64',
  1748→    'float32', 'float64',
  1749→    'byte', 'rune',
  1750→}
  1751→
  1752→# Go types that JSON-marshal as a string.
  1753→GO_STRING_TYPES = {
  1754→    'string', 'time.Time', 'primitive.ObjectID', 'primitive.DateTime',
  1755→    'json.RawMessage',
  1756→}
  1757→
  1758→# Set of Go named types that are aliases for a primitive (e.g.
  1759→# ``type MemberRole string``). Populated at parse time by scanning
  1760→# every Go file for ``type X <primitive>`` declarations. Used by
  1761→# ``normalise_go_type`` to treat these aliases as their underlying
  1762→# primitive (so ``MemberRole`` matches TS ``string``).
  1763→GO_PRIMITIVE_ALIASES: Dict[str, str] = {}
  1764→
  1765→# Map of TS type-alias names to their normalised underlying type
  1766→# (e.g. ``ConfigVarType`` -> ``string`` when the alias is
  1767→# ``'string' | 'numeric' | 'enum' | 'template'``). Populated at parse
  1768→# time from ``type X = ...`` declarations whose RHS is a primitive
  1769→# literal union. Used by ``normalise_ts_type`` to collapse named
  1770→# aliases to their underlying primitive (mirroring Go's primitive
  1771→# alias handling).
  1772→TS_PRIMITIVE_ALIASES: Dict[str, str] = {}
  1773→

  1775→def _scan_primitive_aliases(repo: Path) -> None:
  1776→    """Populate ``GO_PRIMITIVE_ALIASES`` by scanning every Go file for
  1777→    ``type X string`` / ``type X int`` / etc. declarations.
  1778→    """
  1779→    GO_PRIMITIVE_ALIASES.clear()
  1780→    alias_re = re.compile(
  1781→        r'\btype\s+(?P<name>[A-Z]\w*)\s+(?P<underlying>string|int|int8|int16|int32|int64|uint|uint8|uint16|uint32|uint64|float32|float64|byte|rune|bool)\b'
  1782→    )
  1783→    for f in find_go_files(repo):
  1784→        try:
  1785→            text = f.read_text(encoding='utf-8', errors='replace')
  1786→        except Exception:
  1787→            continue
  1788→        for m in alias_re.finditer(text):
  1789→            GO_PRIMITIVE_ALIASES.setdefault(m.group('name'), m.group('underlying'))
  1790→
  1791→
  1792→def _scan_ts_primitive_aliases(ts_types: List['TSType']) -> None:
  1793→    """Populate ``TS_PRIMITIVE_ALIASES`` from parsed TS type aliases.
  1794→
  1795→    A TS alias is considered a "primitive alias" if its raw value is a
  1796→    union of string/number/boolean literals (e.g. ``'a' | 'b'``) or a
  1797→    single primitive. The alias name maps to its normalised primitive.
  1798→    """
  1799→    TS_PRIMITIVE_ALIASES.clear()
  1800→    for t in ts_types:
  1801→        if t.kind != 'type_alias' or not t.raw:
  1802→            continue
  1803→        # Try to normalise the raw value. If it resolves to a primitive
  1804→        # (string/number/boolean), record the alias.
  1805→        normalised = normalise_ts_type(t.raw)
  1806→        base = normalised.rstrip('?')
  1807→        if base in {'string', 'number', 'boolean'}:
  1808→            TS_PRIMITIVE_ALIASES.setdefault(t.name, base)
  1809→

  1811→def normalise_go_type(go_type: str) -> str:
  1812→    """Normalise a Go type to a canonical form for comparison with TS.
  1813→
  1814→    Examples:
  1815→      ``string``           -> ``string``
  1816→      ``*string``          -> ``string?``
  1817→      ``int64``            -> ``number``
  1818→      ``*time.Time``       -> ``string?``
  1819→      ``primitive.ObjectID`` -> ``string``
  1820→      ``models.User``      -> ``User``
  1821→      ``[]string``         -> ``string[]``
  1822→      ``[]models.User``    -> ``User[]``
  1823→      ``map[string]X``     -> ``Record<string, X>``
  1824→      ``MemberRole``       -> ``string`` (if ``type MemberRole string``
  1825→      was found in the codebase)
  1826→    """
  1827→    t = go_type.strip()
  1828→    # Strip a leading `*` (pointer).
  1829→    optional = False
  1830→    if t.startswith('*'):
  1831→        optional = True
  1832→        t = t[1:].strip()
  1833→    # Slice.
  1834→    if t.startswith('[]'):
  1835→        inner = t[2:].strip()
  1836→        return normalise_go_type(inner) + '[]'
  1837→    # Map (best-effort — we don't recurse into the value type).
  1838→    if t.startswith('map['):
  1839→        # Extract the value type.
  1840→        m = re.match(r'^map\[[^\]]+\](.+)$', t)
  1841→        if m:
  1842→            val = normalise_go_type(m.group(1).strip())
  1843→            return f'Record<string, {val}>'
  1844→    # String-serialising types (check BEFORE stripping the package
  1845→    # prefix — the set contains fully-qualified names like
  1846→    # `time.Time` and `primitive.ObjectID`).
  1847→    if t in GO_STRING_TYPES:
  1848→        return 'string' + ('?' if optional else '')
  1849→    # Package-qualified type.
  1850→    if '.' in t:
  1851→        base = t.rsplit('.', 1)[-1]
  1852→        # Check if the base is a known primitive alias.
  1853→        if base in GO_PRIMITIVE_ALIASES:
  1854→            underlying = GO_PRIMITIVE_ALIASES[base]
  1855→            return normalise_go_type(underlying) + ('?' if optional else '')
  1856→        return base + ('?' if optional else '')
  1857→    # Primitive alias (e.g. ``MemberRole`` -> ``string``).
  1858→    if t in GO_PRIMITIVE_ALIASES:
  1859→        underlying = GO_PRIMITIVE_ALIASES[t]
  1860→        return normalise_go_type(underlying) + ('?' if optional else '')
  1861→    # Primitive mappings.
  1862→    if t in GO_NUMERIC_TYPES:
  1863→        return 'number' + ('?' if optional else '')
  1864→    if t == 'bool':
  1865→        return 'boolean' + ('?' if optional else '')
  1866→    if t == 'interface{}' or t == 'any':
  1867→        return 'unknown'
  1868→    return t + ('?' if optional else '')
  1869→

  1871→def normalise_ts_type(ts_type: str) -> str:
  1872→    """Normalise a TS type to a canonical form for comparison with Go.
  1873→
  1874→    Examples:
  1875→      ``string``           -> ``string``
  1876→      ``string | undefined`` -> ``string?``
  1877→      ``number``           -> ``number``
  1878→      ``User``             -> ``User``
  1879→      ``User[]``           -> ``User[]``
  1880→      ``'a' | 'b'``        -> ``string``  (literal union collapses to string)
  1881→      ``Record<string, X>`` -> ``Record<string, X>``
  1882→      ``ConfigVarType``    -> ``string``  (if the alias resolves to a
  1883→      string literal union)
  1884→    """
  1885→    t = ts_type.strip()
  1886→    # Strip parens.
  1887→    while t.startswith('(') and t.endswith(')'):
  1888→        t = t[1:-1].strip()
  1889→    # Named alias that resolves to a primitive (e.g. ``ConfigVarType``
  1890→    # whose definition is ``'string' | 'numeric' | 'enum' | 'template'``).
  1891→    if re.match(r'^[A-Z]\w*$', t) and t in TS_PRIMITIVE_ALIASES:
  1892→        return TS_PRIMITIVE_ALIASES[t]
  1893→    # Union with undefined/null -> optional.
  1894→    optional = False
  1895→    if '|' in t:
  1896→        bits = [b.strip() for b in t.split('|')]
  1897→        if 'undefined' in bits:
  1898→            optional = True
  1899→            bits = [b for b in bits if b != 'undefined']
  1900→        # If all remaining bits are string literals, collapse to string.
  1901→        if bits and all(re.match(r"^['\"]", b) for b in bits):
  1902→            return 'string' + ('?' if optional else '')
  1903→        # If all remaining bits are number literals, collapse to number.
  1904→        if bits and all(re.match(r'^-?\d+(\.\d+)?$', b) for b in bits):
  1905→            return 'number' + ('?' if optional else '')
  1906→        # If exactly one non-null branch remains, recurse.
  1907→        if len(bits) == 1:
  1908→            t = bits[0]
  1909→        else:
  1910→            # Mixed union — keep as a sorted union string.
  1911→            return ' | '.join(sorted(bits)) + ('?' if optional else '')
  1912→    # Array.
  1913→    if t.endswith('[]'):
  1914→        inner = t[:-2].strip()
  1915→        return normalise_ts_type(inner) + '[]'
  1916→    m = re.match(r'^Array<(.+)>$', t)
  1917→    if m:
  1918→        return normalise_ts_type(m.group(1).strip()) + '[]'
  1919→    # String literal.
  1920→    if t.startswith("'") or t.startswith('"'):
  1921→        return 'string' + ('?' if optional else '')
  1922→    # Boolean literal.
  1923→    if t in {'true', 'false'}:
  1924→        return 'boolean' + ('?' if optional else '')
  1925→    # Number literal.
  1926→    if re.match(r'^-?\d+(\.\d+)?$', t):
  1927→        return 'number' + ('?' if optional else '')
  1928→    return t + ('?' if optional else '')
  1929→

  1931→def types_compatible(go_type: str, ts_type: str) -> Tuple[bool, str]:
  1932→    """Return ``(is_compatible, note)`` for a Go↔TS type pair.
  1933→
  1934→    The note explains the mismatch when not compatible.
  1935→
  1936→    A Go type of ``unknown`` (used when the response payload contains a
  1937→    value the auditor can't statically resolve — e.g. a package-qualified
  1938→    constant like ``version.Current``) is treated as compatible with any
  1939→    TS type, since we can't know the actual type without type info.
  1940→    """
  1941→    g = normalise_go_type(go_type)
  1942→    t = normalise_ts_type(ts_type)
  1943→    # `unknown` Go types are compatible with anything — we can't know
  1944→    # the actual type without full Go type inference.
  1945→    if g == 'unknown' or go_type.strip() == 'unknown':
  1946→        return (True, 'Go type unknown — not statically resolvable')
  1947→    if g == t:
  1948→        return (True, '')
  1949→    # Strip optional markers for the comparison — optionality is
  1950→    # handled separately (a Go field is optional if it's a pointer or
  1951→    # omitempty; a TS field is optional if marked with `?`).
  1952→    g_base = g.rstrip('?')
  1953→    t_base = t.rstrip('?')
  1954→    if g_base == t_base:
  1955→        return (True, 'optional differs')
  1956→    # Record<string, X> on both sides — recurse on X (best-effort: skip).
  1957→    if g_base.startswith('Record<string,') and t_base.startswith('Record<string,'):
  1958→        return (True, 'record values not deeply compared')
  1959→    # Known mismatch patterns:
  1960→    # Go int64 / number vs TS string — classic JS precision bug for IDs.
  1961→    if g_base == 'number' and t_base == 'string':
  1962→        return (False, f'Go {go_type} (number) vs TS {ts_type} (string) — possible ID precision bug')
  1963→    if g_base == 'string' and t_base == 'number':
  1964→        return (False, f'Go {go_type} (string) vs TS {ts_type} (number)')
  1965→    if g_base == 'boolean' and t_base != 'boolean':
  1966→        return (False, f'Go {go_type} (boolean) vs TS {ts_type}')
  1967→    # Named struct references that differ in name.
  1968→    if g_base and t_base and g_base[0].isupper() and t_base[0].isupper():
  1969→        if g_base != t_base:
  1970→            return (False, f'struct name mismatch: Go {g_base} vs TS {t_base}')
  1971→    return (False, f'Go {g} vs TS {t}')

  1978→def normalise_path_for_matching(path: str) -> str:
  1979→    """Normalise a URL path for endpoint matching.
  1980→
  1981→    Strips trailing slashes (except root) and lowercases path parameter
  1982→    names so ``/users/{userId}`` and ``/users/${userId}`` both become
  1983→    ``/users/{userid}``.
  1984→    """
  1985→    p = path
  1986→    # Strip query string.
  1987→    if '?' in p:
  1988→        p = p.split('?', 1)[0]
  1989→    # Strip trailing slash (except root).
  1990→    if len(p) > 1 and p.endswith('/'):
  1991→        p = p[:-1]
  1992→    # Lowercase path parameter names so {userId} and {userid} match.
  1993→    p = re.sub(r'\{(\w+)\}', lambda m: '{' + m.group(1).lower() + '}', p)
  1994→    return p

  1997→def build_go_endpoint_shapes(
  1998→    routes: List[GoRoute],
  1999→    handlers_by_name: Dict[str, GoHandler],
  2000→    structs_by_name: Dict[str, GoStruct],
  2001→) -> Dict[str, List[GoResponseShape]]:
  2002→    """Build a ``{(method, normalised_path): [GoResponseShape, ...]}`` map.
  2003→
  2004→    For each route, look up the handler and collect ALL its non-error
  2005→    response shapes (handlers may have multiple success-path responses,
  2006→    e.g. an MFA branch and a normal branch). Error responses (maps
  2007→    with only ``message``/``error`` keys) are skipped.
  2008→    """
  2009→    out: Dict[str, List[GoResponseShape]] = {}
  2010→    for r in routes:
  2011→        key = f'{r.method} {normalise_path_for_matching(r.path)}'
  2012→        handler = handlers_by_name.get(r.handler_func)
  2013→        if not handler:
  2014→            out[key] = [GoResponseShape(kind='unknown',
  2015→                                        raw=f'(no handler found: {r.handler_func})')]
  2016→            continue
  2017→        if not handler.responses:
  2018→            out[key] = [GoResponseShape(kind='unknown',
  2019→                                        raw='(no respondWithJSON call found)')]
  2020→            continue
  2021→        # Collect all non-error response shapes.
  2022→        shapes: List[GoResponseShape] = []
  2023→        for shape in handler.responses:
  2024→            if shape.kind == 'map_literal':
  2025→                names = [f.name for f in shape.fields]
  2026→                if names and all(n in {'message', 'error'} for n in names):
  2027→                    continue
  2028→            shapes.append(shape)
  2029→        if not shapes:
  2030→            shapes = [handler.responses[0]]
  2031→        out[key] = shapes
  2032→    return out

  2035→def resolve_go_shape_fields(
  2036→    shape: GoResponseShape,
  2037→    structs_by_name: Dict[str, GoStruct],
  2038→) -> List[GoResponseField]:
  2039→    """Resolve a GoResponseShape into a flat list of top-level fields.
  2040→
  2041→    For a ``struct_literal`` shape, look up the struct's fields and
  2042→    return them as GoResponseField entries (with ``struct_ref`` set to
  2043→    the struct name for nested comparison).
  2044→
  2045→    For a ``map_literal`` shape, return the parsed fields directly.
  2046→
  2047→    Inline-embedded struct fields (``models.EventDefinition
  2048→    `json:",inline"` ``) are recursively flattened: the embedded
  2049→    struct's fields are spliced into the output at the position of the
  2050→    embedding. This matches Go's standard JSON marshalling behaviour
  2051→    for embedded structs.
  2052→    """
  2053→    if shape.kind == 'struct_literal' and shape.struct_name:
  2054→        s = structs_by_name.get(shape.struct_name)
  2055→        if not s:
  2056→            return []
  2057→        return _flatten_struct_fields(s, structs_by_name, set())
  2058→    if shape.kind == 'map_literal':
  2059→        return shape.fields
  2060→    return []
  2061→
  2062→
  2063→def _flatten_struct_fields(
  2064→    struct: GoStruct,
  2065→    structs_by_name: Dict[str, GoStruct],
  2066→    visited: Set[str],
  2067→) -> List[GoResponseField]:
  2068→    """Return the struct's fields as GoResponseField entries, recursing
  2069→    into inline-embedded struct fields.
  2070→
  2071→    ``visited`` is the set of struct names already being expanded —
  2072→    prevents infinite recursion on cyclic embeddings (which shouldn't
  2073→    exist but are guarded against defensively).
  2074→    """
  2075→    if struct.name in visited:
  2076→        return []
  2077→    visited = visited | {struct.name}
  2078→    out: List[GoResponseField] = []
  2079→    for f in struct.fields:
  2080→        if f.is_skipped:
  2081→            continue
  2082→        if f.is_inline_embed:
  2083→            # Recurse into the embedded struct.
  2084→            embedded_name = _strip_package(f.go_type)
