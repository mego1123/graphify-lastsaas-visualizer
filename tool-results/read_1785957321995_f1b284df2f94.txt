     1→#!/usr/bin/env python3
     2→"""graphify api shapes — backend ↔ frontend response-shape auditor.
     3→
     4→Compares Go backend API response shapes against the TypeScript frontend's
     5→expected response types and reports mismatches.
     6→
     7→How it works
     8→------------
     9→
    10→1. **Go side** — walks every `.go` file under the target path (excluding
    11→   ``*_test.go`` by default) and:
    12→
    13→   - Parses every ``type X struct { ... }`` declaration, extracting the
    14→     field name, Go type, and JSON tag (so we know the wire name and
    15→     whether the field is omitempty / skipped via ``json:"-"``).
    16→   - Parses every HTTP handler function (``func (h *XHandler) Method(
    17→     w http.ResponseWriter, r *http.Request)``) and, inside its body,
    18→     looks for the response-write call. Two patterns are recognised:
    19→
    20→       * ``respondWithJSON(w, http.StatusOK, <payload>)`` — the project's
    21→         own helper, defined in ``internal/api/handlers/helpers.go``.
    22→       * ``json.NewEncoder(w).Encode(<payload>)`` — stdlib pattern used
    23→         directly in a few handlers (e.g. ``usage.go``).
    24→
    25→     The ``<payload>`` is then classified as one of:
    26→
    27→       * ``struct_literal`` — a typed composite literal such as
    28→         ``AuthResponse{...}`` or ``bootstrapStatusResponse{...}``.
    29→       * ``map_literal`` — an anonymous ``map[string]interface{}{...}``
    30→         (or ``map[string]string{...}``) literal. The keys become the
    31→         top-level response fields and the values are resolved to a
    32→         type by inspecting the enclosing handler's local variable
    33→         declarations (``var x models.User`` / ``x := Y{...}``).
    34→       * ``bare_identifier`` — a single variable name such as
    35→         ``user`` (rare; resolved via the same variable lookup).
    36→       * ``unknown`` — anything else (function calls, complex
    37→         expressions). The endpoint is still listed but the shape is
    38→         marked ``unknown`` so the report can flag it for manual review.
    39→
    40→   - Parses the production router setup in ``cmd/server/main.go`` (and
    41→     falls back to ``internal/api/handlers/testhelpers_test.go`` when
    42→     ``main.go`` is absent) to map ``(HTTP method, full URL path) ->
    43→     handler function``. Sub-routers built via ``parent.PathPrefix(
    44→     "/x").Subrouter()`` are flattened so a route registered as
    45→     ``protectedAuth.HandleFunc("/me", authHandler.GetMe)`` resolves
    46→     to ``/api/auth/me``.
    47→
    48→2. **TypeScript side** — parses ``frontend/src/api/client.ts`` and
    49→   ``frontend/src/types/index.ts``:
    50→
    51→   - For every ``api.<method><<ResponseType>>(url, ...)`` call (where
    52→     ``method`` ∈ {get, post, put, patch, delete}) it captures the
    53→     HTTP method, the URL (with template literals like ``${id}``
    54→     normalised to ``{id}`` and ``/api`` prepended), and the response
    55→     type — either a named type (``AuthResponse``) or an inline object
    56→     literal (``{ user: User; memberships: MembershipInfo[] }``).
    57→   - For every ``interface X { ... }`` and ``type X = ...`` declaration
    58→     it captures the field name, TS type, and optionality.
    59→
    60→3. **Comparison** — for each TS endpoint, finds the matching Go endpoint
    61→   by ``(method, normalised_path)``. For each matched pair, resolves
    62→   both response shapes to a flat ``{field_name: type}`` map and
    63→   reports:
    64→
    65→   - **Missing in Go** — TS expects a field the backend doesn't send
    66→     (frontend bug waiting to happen).
    67→   - **Extra in Go** — Go sends a field the frontend doesn't read
    68→     (wasted bandwidth / accidental info leak).
    69→   - **Type mismatch** — both sides have the field but the types
    70→     disagree (e.g. Go ``int64`` vs TS ``string`` for an ID — the
    71→     classic JS precision bug).
    72→
    73→4. **Output** — emits a comparison table and per-endpoint detail in
    74→   both Markdown (``API_SHAPES.md``) and JSON (``api-shapes.json``).
    75→
    76→CLI
    77→---
    78→
    79→    python graphify_api_shapes.py [path] [--out report.md] [--json]
    80→                                   [--include-tests]
    81→
    82→``path`` defaults to the current working directory. When ``--out`` is
    83→omitted and the script is run from the graphify workspace root, the
    84→reports are written to ``public/api-shapes.json`` and
    85→``public/API_SHAPES.md`` automatically.
    86→"""

   100→# Source masking helpers (so regex scans aren't fooled by strings/comments)

   103→def mask_go_source(text: str) -> str:
   104→    """Replace Go string/rune literals and comments with same-length
   105→    whitespace so brace/paren matching is robust. Newlines are preserved
   106→    so line numbers stay accurate.
   107→
   108→    Length is preserved exactly — every input character maps to exactly
   109→    one output character — so byte offsets in the masked source are
   110→    identical to offsets in the raw source. This lets us locate a
   111→    construct in the masked source (where brace matching is safe) and
   112→    then slice the same byte range out of the raw source (where string
   113→    contents are preserved).
   114→    """
   115→    out: List[str] = []
   116→    i, n = 0, len(text)
   117→    in_str = in_rune = in_line = in_block = False
   118→    while i < n:
   119→        c = text[i]
   120→        nxt = text[i + 1] if i + 1 < n else ''
   121→        if in_line:
   122→            if c == '\n':
   123→                in_line = False
   124→                out.append(c)
   125→            else:
   126→                out.append(' ')
   127→            i += 1
   128→            continue
   129→        if in_block:
   130→            if c == '*' and nxt == '/':
   131→                in_block = False
   132→                out.append('  ')
   133→                i += 2
   134→                continue
   135→            out.append('\n' if c == '\n' else ' ')
   136→            i += 1
   137→            continue
   138→        if in_str:
   139→            if c == '\\' and nxt:
   140→                out.append('  ')
   141→                i += 2
   142→                continue
   143→            if c == '"':
   144→                in_str = False
   145→                out.append('"')
   146→                i += 1
   147→                continue
   148→            out.append('\n' if c == '\n' else ' ')
   149→            i += 1
   150→            continue
   151→        if in_rune:
   152→            if c == '\\' and nxt:
   153→                out.append('  ')
   154→                i += 2
   155→                continue
   156→            if c == "'":
   157→                in_rune = False
   158→                out.append("'")
   159→                i += 1
   160→                continue
   161→            out.append(' ')
   162→            i += 1
   163→            continue
   164→        if c == '/' and nxt == '/':
   165→            in_line = True
   166→            out.append('  ')
   167→            i += 2
   168→            continue
   169→        if c == '/' and nxt == '*':
   170→            in_block = True
   171→            out.append('  ')
   172→            i += 2
   173→            continue
   174→        if c == '"':
   175→            in_str = True
   176→            out.append('"')
   177→            i += 1
   178→            continue
   179→        if c == "'":
   180→            in_rune = True
   181→            out.append("'")
   182→            i += 1
   183→            continue
   184→        out.append(c)
   185→        i += 1
   186→    return ''.join(out)

   189→def mask_ts_source(text: str) -> str:
   190→    """Mask TS string literals, template literals, line and block
   191→    comments. Template literals are tricky because they may contain
   192→    nested ``${...}`` expressions — for our purposes (URL extraction)
   193→    we keep the literal text of template strings intact so we can
   194→    convert ``${id}`` to ``{id}`` later. We only strip comments and
   195→    ordinary string literals.
   196→    """
   197→    out: List[str] = []
   198→    i, n = 0, len(text)
   199→    in_line = in_block = in_str = in_template = False
   200→    template_depth = 0  # depth of ${ ... } inside template
   201→    while i < n:
   202→        c = text[i]
   203→        nxt = text[i + 1] if i + 1 < n else ''
   204→        if in_line:
   205→            if c == '\n':
   206→                in_line = False
   207→                out.append(c)
   208→            else:
   209→                out.append(' ')
   210→            i += 1
   211→            continue
   212→        if in_block:
   213→            if c == '*' and nxt == '/':
   214→                in_block = False
   215→                out.append('  ')
   216→                i += 2
   217→                continue
   218→            out.append('\n' if c == '\n' else ' ')
   219→            i += 1
   220→            continue
   221→        if in_str:
   222→            if c == '\\' and nxt:
   223→                out.append(c)
   224→                out.append(nxt)
   225→                i += 2
   226→                continue
   227→            if c == '"':
   228→                in_str = False
   229→                out.append('"')
   230→                i += 1
   231→                continue
   232→            out.append(c)
   233→            i += 1
   234→            continue
   235→        if in_template:
   236→            # Keep template content verbatim (we want ${id} preserved).
   237→            if c == '\\' and nxt:
   238→                out.append(c)
   239→                out.append(nxt)
   240→                i += 2
   241→                continue
   242→            if c == '`':
   243→                in_template = False
   244→                out.append('`')
   245→                i += 1
   246→                continue
   247→            out.append(c)
   248→            i += 1
   249→            continue
   250→        if c == '/' and nxt == '/':
   251→            in_line = True
   252→            i += 2
   253→            continue
   254→        if c == '/' and nxt == '*':
   255→            in_block = True
   256→            i += 2
   257→            continue
   258→        if c == '"':
   259→            in_str = True
   260→            out.append('"')
   261→            i += 1
   262→            continue
   263→        if c == '`':
   264→            in_template = True
   265→            out.append('`')
   266→            i += 1
   267→            continue
   268→        out.append(c)
   269→        i += 1
   270→    return ''.join(out)

   272→
   273→def line_of(text: str, idx: int) -> int:
   274→    return text.count('\n', 0, idx) + 1
   275→
   276→
   277→def find_matching_brace(text: str, open_idx: int, open_ch: str = '{',
   278→                        close_ch: str = '}') -> Tuple[Optional[str], int]:
   279→    """Given ``text[open_idx] == open_ch``, return ``(body, end_idx)``
   280→    where body is the text between the matching braces and ``end_idx``
   281→    is one past the closing brace. Returns ``(None, -1)`` if unbalanced.
   282→    """
   283→    assert text[open_idx] == open_ch
   284→    depth = 0
   285→    i = open_idx
   286→    n = len(text)
   287→    while i < n:
   288→        c = text[i]
   289→        if c == open_ch:
   290→            depth += 1
   291→        elif c == close_ch:
   292→            depth -= 1
   293→            if depth == 0:
   294→                return text[open_idx + 1:i], i + 1
   295→        i += 1
   296→    return None, -1
   297→
   298→
   299→def find_matching_paren(text: str, open_idx: int) -> Tuple[Optional[str], int]:
   300→    return find_matching_brace(text, open_idx, '(', ')')

   413→# ---------------------------------------------------------------------------
   414→# Go struct parser
   415→# ---------------------------------------------------------------------------
   416→
   417→# `type Name struct {` — open brace at end of line. Name may be exported
   418→# (uppercase initial) or unexported (lowercase initial) — both can be
   419→# used as response shapes since JSON serialisation doesn't care about
   420→# Go's export visibility.
   421→RE_STRUCT_OPEN = re.compile(
   422→    r'\btype\s+(?P<name>[A-Za-z]\w*)\s+struct\s*\{'
   423→)
   424→
   425→# A field declaration inside a struct body. Field name is capitalised
   426→# identifier; type is everything up to the backtick (or end of line if
   427→# no tag); the JSON tag (if present) is captured.
   428→RE_STRUCT_FIELD = re.compile(
   429→    r'^\s*(?P<name>[A-Z]\w*)\s+(?P<type>[^\s`].*?)'
   430→    r'(?:\s+`(?P<tag>[^`]*)`)?\s*$'
   431→)
   432→
   433→# JSON tag inside a struct tag string.
   434→RE_JSON_TAG = re.compile(r'json:"(?P<val>[^"]*)"')
   435→
   436→RE_PACKAGE = re.compile(r'^\s*package\s+(\w+)', re.MULTILINE)
   437→
   438→
   439→def parse_go_structs(content: str, file: str, package: str,
   440→                     raw_content: Optional[str] = None) -> List[GoStruct]:
   441→    """Parse every ``type X struct { ... }`` declaration in ``content``.
   442→
   443→    ``content`` should be the masked source (for safe brace matching).
   444→    ``raw_content`` is the original source — when provided, the struct
   445→    body is re-sliced from the raw source so JSON tags are preserved.
   446→    Offsets are identical because masking preserves length.
   447→    """
   448→    if raw_content is None:
   449→        raw_content = content
   450→    out: List[GoStruct] = []
   451→    for m in RE_STRUCT_OPEN.finditer(content):
   452→        name = m.group('name')
   453→        line = line_of(content, m.start())
   454→        brace_idx = content.find('{', m.start())
   455→        if brace_idx == -1:
   456→            continue
   457→        body, _ = find_matching_brace(content, brace_idx, '{', '}')
   458→        if body is None:
   459→            continue
   460→        # Re-slice the body from the raw source so JSON tags survive.
   461→        raw_body = raw_content[brace_idx + 1: brace_idx + 1 + len(body)]
   462→        fields = _parse_struct_fields(raw_body)
   463→        out.append(GoStruct(
   464→            name=name, file=file, line=line,
   465→            package=package, fields=fields,
   466→        ))
   467→    return out

   469→
   470→def _parse_struct_fields(body: str) -> List[GoField]:
   471→    """Parse the body of a struct (text between the braces)."""
   472→    fields: List[GoField] = []
   473→    # We split on newlines but a field declaration may span multiple
   474→    # lines if the type contains a newline (rare for our codebase).
   475→    # Simple heuristic: process line by line, but if a line doesn't
   476→    # have a backtick AND the next line starts with whitespace + backtick,
   477→    # join them.
   478→    lines = body.splitlines()
   479→    i = 0
   480→    while i < len(lines):
   481→        line = lines[i]
   482→        # Join continuation lines: if this line has a field name + type
   483→        # but no backtick, look ahead for the tag.
   484→        while i + 1 < len(lines) and '`' not in line and line.strip() and \
   485→                not _is_field_start(lines[i + 1]):
   486→            # Heuristic: continuation lines are usually indented more
   487→            # than the field name line and contain the tag.
   488→            line = line.rstrip() + ' ' + lines[i + 1].strip()
   489→            i += 1
   490→            if '`' in line:
   491→                break
   492→        f = _parse_struct_field_line(line)
   493→        if f:
   494→            fields.append(f)
   495→        i += 1
   496→    return fields
   497→
   498→
   499→def _is_field_start(line: str) -> bool:
   500→    """True if ``line`` looks like the start of a new struct field."""
   501→    s = line.strip()
   502→    if not s:
   503→        return False
   504→    # Field names are capitalised identifiers.
   505→    return bool(re.match(r'^[A-Z]\w*\s+', s))
   506→
   507→
   508→def _parse_struct_field_line(line: str) -> Optional[GoField]:
   509→    s = line.strip()
   510→    if not s or s.startswith('//'):
   511→        return None
   512→    # Inline-embedded struct: `models.EventDefinition `json:",inline"``
   513→    # or `EventDefinition `json:",inline"``. The embedded struct's
   514→    # fields are serialized as if they were declared on the outer
   515→    # struct (Go's standard JSON marshalling behaviour for embedded
   516→    # structs without an explicit json tag — and the explicit
   517→    # `,inline` form is a project convention that signals the same
   518→    # intent).
   519→    inline_m = re.match(
   520→        r'^(?P<type>(?:[a-z]\w*\.)?[A-Z]\w*)\s+`(?P<tag>[^`]*)`\s*$',
   521→        s,
   522→    )
   523→    if inline_m:
   524→        tag_str = inline_m.group('tag')
   525→        jm = RE_JSON_TAG.search(tag_str)
   526→        if jm and (jm.group('val') == ',inline' or
   527→                   jm.group('val').startswith(',inline')):
   528→            return GoField(
   529→                name=inline_m.group('type'),
   530→                json_name='',     # inline — no wire name of its own
   531→                go_type=inline_m.group('type'),
   532→                is_inline_embed=True,
   533→            )
   534→    # Embedded field (just a type name, no field name, no tag) — Go's
   535→    # default JSON marshalling flattens these, but we conservatively
   536→    # skip them since they may be interface embeddings or non-data
   537→    # behaviour mixins. Tagged inline-embedded fields above ARE
   538→    # flattened.
   539→    if '`' not in s and re.match(r'^[A-Z]\w*\.?[A-Z]\w*$', s):
   540→        return None
   541→    # Comment-only or brace-only line.
   542→    if s in {'{', '}', '};'}:
   543→        return None
   544→    m = RE_STRUCT_FIELD.match(s)
   545→    if not m:
   546→        # Fall back to a more permissive split: name + type + tag.
   547→        # This catches unusual types like `map[string]EntitlementValue`.
   548→        parts = s.split('`', 1)
   549→        if len(parts) != 2:
   550→            return None
   551→        head, tag = parts
   552→        head = head.strip()
   553→        bits = head.split(None, 1)
   554→        if len(bits) != 2:
   555→            return None
   556→        name, go_type = bits
   557→        tag_str = tag.strip()
   558→    else:
   559→        name = m.group('name')
   560→        go_type = m.group('type').strip()
   561→        tag_str = m.group('tag') or ''
   562→    if not name or not go_type:
   563→        return None
   564→    json_name = name  # default to Go field name
   565→    is_omitempty = False
   566→    is_skipped = False
   567→    jm = RE_JSON_TAG.search(tag_str)
   568→    if jm:
   569→        val = jm.group('val')
   570→        if val == '-':
   571→            is_skipped = True
   572→            json_name = ''
   573→        else:
   574→            # Handle "name,omitempty" and "name"
   575→            bits = val.split(',', 1)
   576→            json_name = bits[0]
   577→            if len(bits) > 1 and 'omitempty' in bits[1]:
   578→                is_omitempty = True
   579→    is_optional = is_omitempty or go_type.startswith('*')
   580→    return GoField(
   581→        name=name, json_name=json_name, go_type=go_type,
   582→        is_omitempty=is_omitempty, is_skipped=is_skipped,
   583→        is_optional=is_optional,
   584→    )
   585→
   586→
   587→# ---------------------------------------------------------------------------
   588→# Go handler / response parser
   589→# ---------------------------------------------------------------------------
   590→
   591→# `func (h *XHandler) Method(w http.ResponseWriter, r *http.Request) {`
   592→# Also matches `func (h XHandler) Method(...)` (value receiver) and
   593→# unexported handler types like `func (h *bootstrapHandler) Status(...)`.
   594→RE_HANDLER_FUNC = re.compile(
   595→    r'\bfunc\s*\(\s*(?P<recv>\w+)\s+\*?(?P<recvtype>[A-Z]\w*)\s*\)\s+'
   596→    r'(?P<method>[A-Z]\w*)\s*\(\s*w\s+http\.ResponseWriter,\s*r\s+\*?http\.Request\s*\)\s*\{'
   597→)
   598→
   599→# A `respondWithJSON(w, http.StatusXxx, <payload>)` call. The payload
   600→# begins after the second comma.
   601→RE_RESPOND_WITH_JSON = re.compile(
   602→    r'\brespondWithJSON\s*\(\s*w\s*,\s*http\.Status\w+\s*,'
   603→)
   604→
   605→# A `json.NewEncoder(w).Encode(<payload>)` call.
   606→RE_JSON_ENCODER = re.compile(
   607→    r'\bjson\.NewEncoder\s*\(\s*w\s*\)\.Encode\s*\('
   608→)
   609→
   610→# A struct-literal payload: `Name{` or `pkg.Name{` or `&Name{` (Name may
   611→# be exported or unexported). Captures the struct name without the
   612→# package prefix.
   613→RE_STRUCT_LITERAL = re.compile(
   614→    r'^\s*&?\s*(?:(?P<pkg>[a-z]\w*)\.)?(?P<name>[A-Za-z]\w*)\s*\{'
   615→)
   616→
   617→# A map literal payload: `map[keyType]valueType{`.
   618→RE_MAP_LITERAL = re.compile(r'^\s*map\s*\[')
   619→
   620→# Variable declaration: `var name Type` (single var).
   621→RE_VAR_DECL = re.compile(
   622→    r'\bvar\s+(?P<name>\w+)\s+(?P<type>[^\s=;]+(?:\.[A-Z]\w*)?(?:\[\w*\])?)'
   623→)
   624→
   625→# Short declaration: `name := value` — we only care when value is a
   626→# composite literal `Type{...}` or a slice literal `[]Type{...}`.
   627→# Allows an optional lowercase package prefix (e.g. `models.Announcement`).
   628→RE_SHORT_DECL_LITERAL = re.compile(
   629→    r'\b(?P<name>\w+)\s*:?=\s*(?:'
   630→    r'make\(\[\]?(?P<maketype>(?:[a-z]\w*\.)?[A-Z]\w*)'
   631→    r'|\[\]?(?P<type>(?:[a-z]\w*\.)?[A-Z]\w*)\s*\{'
   632→    r'|(?P<vtype>(?:[a-z]\w*\.)?[A-Z]\w*)\s*\{'
   633→    r')'
   634→)
   635→
   636→# Slice variable: `var x []Type` or `var x = []Type{...}`.
   637→RE_VAR_SLICE = re.compile(
   638→    r'\bvar\s+(?P<name>\w+)\s+\[\]?(?P<type>(?:[a-z]\w*\.)?[A-Z]\w*)'
   639→)
   640→
   641→# Make for a slice: `x := make([]Type, ...)`.
   642→RE_MAKE_SLICE = re.compile(
   643→    r'\b(?P<name>\w+)\s*:?=\s*make\(\s*\[\]?\s*(?P<type>(?:[a-z]\w*\.)?[A-Z]\w*)'
   644→)
   645→
   646→# A `var name Type` declaration (single var, no initializer). Captures
   647→# the variable name and its type. Allows optional package prefix.
   648→RE_VAR_DECL_NAMED = re.compile(
   649→    r'\bvar\s+(?P<name>\w+)\s+(?P<type>(?:[a-z]\w*\.)?[A-Z]\w*(?:\[\])?)'
   650→)
   651→
   652→
   653→def parse_go_handlers(content: str, file: str,
   654→                      raw_content: Optional[str] = None) -> List[GoHandler]:
   655→    """Parse every HTTP handler function in ``content`` and extract its
   656→    response shape(s).
   657→
   658→    ``content`` should be the masked source (comments and string
   659→    contents blanked out) so brace matching is robust. ``raw_content``
   660→    is the original source — when provided, the response *payload* is
   661→    extracted from the raw source so string keys inside
   662→    ``map[string]interface{}{...}`` literals are preserved. Offsets in
   663→    the masked and raw sources are identical because masking preserves
   664→    length and newlines.
   665→    """
   666→    out: List[GoHandler] = []
   667→    if raw_content is None:
   668→        raw_content = content
   669→    for m in RE_HANDLER_FUNC.finditer(content):
   670→        recvtype = m.group('recvtype')
   671→        method = m.group('method')
   672→        brace_idx = content.find('{', m.start())
   673→        if brace_idx == -1:
   674→            continue
   675→        body, _ = find_matching_brace(content, brace_idx, '{', '}')
   676→        if body is None:
   677→            continue
   678→        # Extract the corresponding raw body (same offsets).
   679→        raw_body = raw_content[brace_idx + 1: brace_idx + 1 + len(body)]
   680→        h = GoHandler(
   681→            name=f'{recvtype}.{method}',
   682→            file=file,
   683→            line=line_of(content, m.start()),
   684→            body=body,
   685→        )
   686→        h.responses = _parse_handler_responses(body, raw_body)
   687→        out.append(h)
   688→    return out
   689→
   690→
   691→def _parse_handler_responses(body: str,
   692→                             raw_body: Optional[str] = None) -> List[GoResponseShape]:
   693→    """Find every response-write call in the handler body and parse
   694→    its payload.
   695→
   696→    ``body`` is the masked body (used to locate response calls safely).
   697→    ``raw_body`` is the original body text — when provided, payloads
   698→    are extracted from it so string keys are preserved.
   699→    """
   700→    if raw_body is None:
   701→        raw_body = body
   702→    shapes: List[GoResponseShape] = []
   703→    # Build a variable -> type map by scanning the body for declarations.
   704→    var_types = _collect_var_types(body)
   705→
   706→    # Find every respondWithJSON call.
   707→    for m in RE_RESPOND_WITH_JSON.finditer(body):
   708→        # The payload starts after the second comma. Find the matching
   709→        # close paren of the outer respondWithJSON call.
   710→        open_paren_idx = body.find('(', m.start())
   711→        if open_paren_idx == -1:
   712→            continue
   713→        args, _ = find_matching_paren(body, open_paren_idx)
   714→        if args is None:
   715→            continue
   716→        # The payload is everything after the second comma.
   717→        payload = _extract_third_arg(args)
   718→        if payload is None:
   719→            continue
   720→        # Re-extract the payload from the raw body so string keys
   721→        # are preserved. The offsets in `body` and `raw_body` are
   722→        # identical because masking preserves length.
   723→        raw_payload = _extract_payload_from_raw(raw_body, open_paren_idx)
   724→        if raw_payload is not None:
   725→            payload = raw_payload
   726→        shape = _classify_payload(payload, var_types)
   727→        shapes.append(shape)
   728→
   729→    # Find every json.NewEncoder(w).Encode call.
   730→    for m in RE_JSON_ENCODER.finditer(body):
   731→        open_paren_idx = body.find('(', m.end() - 1)
   732→        if open_paren_idx == -1:
   733→            continue
   734→        args, _ = find_matching_paren(body, open_paren_idx)
   735→        if args is None:
   736→            continue
   737→        # Re-extract from raw.
   738→        raw_payload = _extract_encode_payload_from_raw(raw_body, m.start())
   739→        payload = raw_payload if raw_payload is not None else args.strip()
   740→        shape = _classify_payload(payload, var_types)
   741→        shapes.append(shape)
   742→
   743→    return shapes
   744→
   745→
   746→def _extract_payload_from_raw(raw_body: str,
   747→                              masked_open_paren_idx: int) -> Optional[str]:
   748→    """Given the open-paren index of a ``respondWithJSON(...)`` call
   749→    in the masked body, extract the third argument (the payload) from
   750→    the raw body using the same offset.
   751→    """
   752→    # Find the matching close paren in the raw body (string-aware).
   753→    raw_args, _ = _find_matching_paren_raw(raw_body, masked_open_paren_idx)
   754→    if raw_args is None:
   755→        return None
   756→    return _extract_third_arg(raw_args)
   757→
   758→
   759→def _extract_encode_payload_from_raw(raw_body: str,
   760→                                     masked_call_start: int) -> Optional[str]:
   761→    """Given the start index of a ``json.NewEncoder(w).Encode(...)`` call
   762→    in the masked body, extract the payload from the raw body.
   763→    """
   764→    # Find the open paren of the Encode call. In the raw body the call
   765→    # text is `json.NewEncoder(w).Encode(` — find the open paren after
   766→    # `Encode`.
   767→    encode_idx = raw_body.find('Encode(', masked_call_start)
   768→    if encode_idx == -1:
   769→        return None
   770→    open_paren_idx = raw_body.find('(', encode_idx)
   771→    if open_paren_idx == -1:
   772→        return None
   773→    raw_args, _ = _find_matching_paren_raw(raw_body, open_paren_idx)
   774→    if raw_args is None:
   775→        return None
   776→    return raw_args.strip()

   778→
   779→def _find_matching_paren_raw(text: str, open_idx: int) -> Tuple[Optional[str], int]:
   780→    """Like ``find_matching_paren`` but skips over string literals so
   781→    parens inside strings don't confuse the matcher.
   782→    """
   783→    if open_idx < 0 or open_idx >= len(text) or text[open_idx] != '(':
   784→        return None, -1
   785→    depth = 0
   786→    i = open_idx
   787→    n = len(text)
   788→    in_str = False
   789→    in_rune = False
   790→    in_line_comment = False
   791→    in_block_comment = False
   792→    while i < n:
   793→        c = text[i]
   794→        nxt = text[i + 1] if i + 1 < n else ''
   795→        if in_line_comment:
   796→            if c == '\n':
   797→                in_line_comment = False
   798→            i += 1
   799→            continue
   800→        if in_block_comment:
   801→            if c == '*' and nxt == '/':
   802→                in_block_comment = False
   803→                i += 2
   804→                continue
   805→            i += 1
   806→            continue
   807→        if in_str:
   808→            if c == '\\' and nxt:
   809→                i += 2
   810→                continue
   811→            if c == '"':
   812→                in_str = False
   813→            i += 1
   814→            continue
   815→        if in_rune:
   816→            if c == '\\' and nxt:
   817→                i += 2
   818→                continue
   819→            if c == "'":
   820→                in_rune = False
   821→            i += 1
   822→            continue
   823→        if c == '/' and nxt == '/':
   824→            in_line_comment = True
   825→            i += 2
   826→            continue
   827→        if c == '/' and nxt == '*':
   828→            in_block_comment = True
   829→            i += 2
   830→            continue
   831→        if c == '"':
   832→            in_str = True
   833→            i += 1
   834→            continue
   835→        if c == "'":
   836→            in_rune = True
   837→            i += 1
   838→            continue
   839→        if c == '(':
   840→            depth += 1
   841→        elif c == ')':
   842→            depth -= 1
   843→            if depth == 0:
   844→                return text[open_idx + 1:i], i + 1
   845→        i += 1
   846→    return None, -1

   848→
   849→def _collect_var_types(body: str) -> Dict[str, str]:
   850→    """Build a ``{var_name: type_str}`` map by scanning the handler
   851→    body for variable declarations. Best-effort — used to resolve the
   852→    types of bare identifiers in map literals.
   853→    """
   854→    out: Dict[str, str] = {}
   855→    # `var name Type` declarations (most reliable — captures the type
   856→    # explicitly). Try the named-variant regex first since it handles
   857→    # package-qualified types like `models.ConfigVar`.
   858→    for m in RE_VAR_DECL_NAMED.finditer(body):
   859→        t = m.group('type')
   860→        if t.endswith('[]'):
   861→            out[m.group('name')] = t
   862→        else:
   863→            out[m.group('name')] = t
   864→    # Fallback: the looser `var name X` regex catches things the named
   865→    # variant misses (e.g. `var name map[string]X`).
   866→    for m in RE_VAR_DECL.finditer(body):
   867→        out.setdefault(m.group('name'), m.group('type'))
   868→    for m in RE_VAR_SLICE.finditer(body):
   869→        out[m.group('name')] = f'[]{m.group("type")}'
   870→    for m in RE_MAKE_SLICE.finditer(body):
   871→        out[m.group('name')] = f'[]{m.group("type")}'
   872→    for m in RE_SHORT_DECL_LITERAL.finditer(body):
   873→        name = m.group('name')
   874→        maketype = m.group('maketype')
   875→        slice_type = m.group('type')
   876→        vtype = m.group('vtype')
   877→        if maketype:
   878→            out[name] = f'[]{maketype}'
   879→        elif slice_type:
   880→            out[name] = f'[]{slice_type}'
   881→        elif vtype:
   882→            out[name] = vtype
   883→    return out

   885→
   886→def _extract_third_arg(args: str) -> Optional[str]:
   887→    """Given the comma-separated argument list of a
   888→    ``respondWithJSON(w, status, payload)`` call, return the payload
   889→    (third argument) as a string.
   890→    """
   891→    depth = 0
   892→    comma_positions: List[int] = []
   893→    for i, c in enumerate(args):
   894→        if c in '([{':
   895→            depth += 1
   896→        elif c in ')]}':
   897→            depth -= 1
   898→        elif c == ',' and depth == 0:
   899→            comma_positions.append(i)
   900→    if len(comma_positions) < 2:
   901→        return None
   902→    payload = args[comma_positions[1] + 1:].strip()
   903→    return payload

   906→def _classify_payload(payload: str,
   907→                      var_types: Dict[str, str]) -> GoResponseShape:
   908→    """Classify a response payload expression into a GoResponseShape."""
   909→    payload = payload.strip()
   910→    raw = payload[:200]
   911→    if not payload:
   912→        return GoResponseShape(kind='unknown', raw=raw)
   913→
   914→    # Strip a trailing cast like `.(someType)` or method chain — keep
   915→    # only the outermost expression.
   916→
   917→    # Case 1: struct literal `Name{...}`.
   918→    m = RE_STRUCT_LITERAL.match(payload)
   919→    if m:
   920→        return GoResponseShape(
   921→            kind='struct_literal',
   922→            struct_name=m.group('name'),
   923→            raw=raw,
   924→        )
   925→
   926→    # Case 2: map literal `map[...]...{...}`.
   927→    if RE_MAP_LITERAL.match(payload):
   928→        return _parse_map_literal(payload, var_types, raw)
   929→
   930→    # Case 3: slice of structs `[]Name{...}`.
   931→    if payload.startswith('[]'):
   932→        # e.g. `[]MemberResponse{...}` — rare for top-level responses.
   933→        inner = payload[2:].strip()
   934→        m2 = RE_STRUCT_LITERAL.match(inner)
   935→        if m2:
   936→            return GoResponseShape(
   937→                kind='struct_literal',
   938→                struct_name=m2.group('name'),
   939→                raw=raw,
   940→            )
   941→        return GoResponseShape(kind='unknown', raw=raw)
   942→
   943→    # Case 4: bare identifier (a variable holding a struct).
   944→    if re.match(r'^[a-zA-Z_]\w*$', payload):
   945→        t = var_types.get(payload)
   946→        if t:
   947→            return _make_shape_from_type(t, raw)
   948→        return GoResponseShape(kind='bare_identifier', raw=raw)
   949→
   950→    # Case 5: address-of struct `&Name{...}`.
   951→    if payload.startswith('&'):
   952→        inner = payload[1:].strip()
   953→        m3 = RE_STRUCT_LITERAL.match(inner)
   954→        if m3:
   955→            return GoResponseShape(
   956→                kind='struct_literal',
   957→                struct_name=m3.group('name'),
   958→                raw=raw,
   959→            )
   960→
   961→    return GoResponseShape(kind='unknown', raw=raw)
   962→
   963→
   964→def _make_shape_from_type(type_str: str, raw: str) -> GoResponseShape:
   965→    """Build a GoResponseShape from a resolved variable type string.
   966→    Used when a payload is a bare identifier whose type we know.
   967→    """
   968→    is_array = type_str.startswith('[]')
   969→    if is_array:
   970→        type_str = type_str[2:]
   971→    struct_ref = _strip_package(type_str)
   972→    return GoResponseShape(
   973→        kind='struct_literal' if struct_ref else 'unknown',
   974→        struct_name=struct_ref or None,
   975→        fields=[],
   976→        raw=raw,
   977→    )

   979→
   980→def _parse_map_literal(payload: str,
   981→                       var_types: Dict[str, str],
   982→                       raw: str) -> GoResponseShape:
   983→    """Parse a ``map[string]interface{}{...}`` literal into a list of
   984→    GoResponseField entries (one per top-level key).
   985→    """
   986→    # Find the opening brace of the literal body. This is tricky because
   987→    # `map[string]interface{}{...}` contains `interface{}` which has its
   988→    # own `{}` pair. We scan from the start, tracking depth, and find
   989→    # the first `{` at depth 0 (after the `map[...]` and value-type
   990→    # declarations).
   991→    brace_idx = _find_map_body_brace(payload)
   992→    if brace_idx == -1:
   993→        # Fall back to the simple find — may pick the wrong brace but
   994→        # at least we won't crash.
   995→        brace_idx = payload.find('{')
   996→    if brace_idx == -1:
   997→        return GoResponseShape(kind='unknown', raw=raw)
   998→    body, _ = find_matching_brace(payload, brace_idx, '{', '}')
   999→    if body is None:
  1000→        return GoResponseShape(kind='unknown', raw=raw)
  1001→    fields: List[GoResponseField] = []
  1002→    # Walk the body, splitting on top-level commas.
  1003→    entries = _split_top_level_commas(body)
  1004→    for entry in entries:
  1005→        entry = entry.strip()
  1006→        if not entry:
  1007→            continue
  1008→        # Each entry is `"key": value`.
  1009→        cm = re.match(r'^"(?P<key>[^"]*)"\s*:\s*(?P<val>.*)$', entry, re.DOTALL)
  1010→        if not cm:
  1011→            # Single-string map (e.g. `map[string]string{"message": "..."}`)
  1012→            # is already in the same shape — the regex above handles it.
  1013→            continue
  1014→        key = cm.group('key')
  1015→        val = cm.group('val').strip()
  1016→        f = _classify_map_value(val, var_types)
  1017→        f.name = key
  1018→        fields.append(f)
  1019→    return GoResponseShape(
  1020→        kind='map_literal',
  1021→        fields=fields,
  1022→        raw=raw,
  1023→    )
  1024→
  1025→
  1026→def _find_map_body_brace(payload: str) -> int:
  1027→    """Find the index of the opening brace of a ``map[...]...{...}``
  1028→    literal's body. Skips past the ``map[...]`` type declaration and
  1029→    any nested ``interface{}`` / ``struct{}`` in the value type.
  1030→
  1031→    The body brace is the ``{`` that comes AFTER the full type
  1032→    declaration. For ``map[string]interface{}{...}`` the type is
  1033→    ``interface{}`` (which has its own ``{}`` pair we must skip);
  1034→    for ``map[string]string{...}`` the type is just ``string``.
  1035→    """
  1036→    s = payload.lstrip()
  1037→    offset = len(payload) - len(s)
  1038→    if not s.startswith('map['):
  1039→        return payload.find('{')
  1040→    # Walk past the `map[...]` part, tracking bracket depth.
  1041→    i = payload.find('[')
  1042→    if i == -1:
  1043→        return payload.find('{')
  1044→    depth = 1
  1045→    i += 1
  1046→    n = len(payload)
  1047→    while i < n and depth > 0:
  1048→        c = payload[i]
  1049→        if c == '[':
  1050→            depth += 1
  1051→        elif c == ']':
  1052→            depth -= 1
  1053→        i += 1
  1054→    # `i` is now just past the closing `]` of `map[...]`.
  1055→    # Skip whitespace.
  1056→    while i < n and payload[i] in ' \t\n\r':
  1057→        i += 1
  1058→    # Skip the value type. It may be:
  1059→    #   - `interface{}` or `struct{}` (with their own `{}`)
  1060→    #   - `*Type` (pointer)
  1061→    #   - `[]Type` (slice)
  1062→    #   - `map[...]Type` (recursive — track depth)
  1063→    #   - `pkg.Type` or `Type` (named type)
  1064→    # We loop until we find the body's `{`.
  1065→    while i < n:
  1066→        c = payload[i]
  1067→        if c == '{':
  1068→            return i
  1069→        if c == '*':
  1070→            i += 1
  1071→            continue
  1072→        if c == '[':
  1073→            # Skip `[]` or `[N]`.
  1074→            depth = 1
  1075→            i += 1
  1076→            while i < n and depth > 0:
  1077→                if payload[i] == '[':
  1078→                    depth += 1
  1079→                elif payload[i] == ']':
  1080→                    depth -= 1
  1081→                i += 1
  1082→            continue
  1083→        if c.isalpha() or c == '_':
  1084→            # Skip an identifier (possibly with `.` and `[]` suffixes).
  1085→            while i < n and (payload[i].isalnum() or payload[i] in '._'):
  1086→                i += 1
  1087→            # After the identifier, check for `interface{}` / `struct{}`
  1088→            # — actually those are keywords, not identifiers, so they're
  1089→            # already skipped by the identifier loop above. Now check
  1090→            # for a following `{}` (e.g. `interface{}`).
  1091→            if i < n and payload[i] == '{':
  1092→                # Skip the `{}` pair.
  1093→                depth = 1
  1094→                i += 1
  1095→                while i < n and depth > 0:
  1096→                    if payload[i] == '{':
  1097→                        depth += 1
  1098→                    elif payload[i] == '}':
  1099→                        depth -= 1
  1100→                    i += 1
  1101→            continue
  1102→        # Skip whitespace and other separators.
  1103→        i += 1
  1104→    return -1
  1105→
  1106→
  1107→def _split_top_level_commas(text: str) -> List[str]:
  1108→    """Split ``text`` on top-level commas (ignoring those inside
  1109→    brackets, braces, or parens).
  1110→    """
  1111→    out: List[str] = []
  1112→    depth = 0
  1113→    start = 0
  1114→    for i, c in enumerate(text):
  1115→        if c in '([{':
  1116→            depth += 1
  1117→        elif c in ')]}':
  1118→            depth -= 1
  1119→        elif c == ',' and depth == 0:
  1120→            out.append(text[start:i])
  1121→            start = i + 1
  1122→    if start < len(text):
  1123→        out.append(text[start:])
  1124→    return out

  1127→def _classify_map_value(val: str,
  1128→                        var_types: Dict[str, str]) -> GoResponseField:
  1129→    """Given a value expression from a map literal, determine its type."""
  1130→    val = val.strip()
  1131→    # Strip trailing commas.
  1132→    while val.endswith(','):
  1133→        val = val[:-1].strip()
  1134→    # Struct literal: `Name{...}`.
  1135→    m = RE_STRUCT_LITERAL.match(val)
  1136→    if m:
  1137→        name = m.group('name')
  1138→        return GoResponseField(
  1139→            name='', go_type=name, struct_ref=name,
  1140→        )
  1141→    # Slice of struct literals: `[]Name{...}`.
  1142→    if val.startswith('[]'):
  1143→        inner = val[2:].strip()
  1144→        m2 = RE_STRUCT_LITERAL.match(inner)
  1145→        if m2:
  1146→            name = m2.group('name')
  1147→            return GoResponseField(
  1148→                name='', go_type=f'[]{name}', struct_ref=name,
  1149→                is_array=True,
  1150→            )
  1151→        # `[]pkg.Name{...}` — also a slice (fallback regex).
  1152→        m3 = re.match(r'^(?:(?P<pkg>[a-z]\w*)\.)?(?P<name>[A-Za-z]\w*)\s*\{',
  1153→                      inner)
  1154→        if m3:
  1155→            name = m3.group('name')
  1156→            return GoResponseField(
  1157→                name='', go_type=f'[]{name}', struct_ref=name,
  1158→                is_array=True,
  1159→            )
  1160→    # Bare identifier (variable reference).
  1161→    if re.match(r'^[a-zA-Z_]\w*$', val):
  1162→        t = var_types.get(val)
  1163→        if t:
  1164→            is_array = t.startswith('[]')
  1165→            base = t[2:] if is_array else t
  1166→            return GoResponseField(
  1167→                name='', go_type=t,
  1168→                struct_ref=_strip_package(base) or None,
  1169→                is_array=is_array,
  1170→            )
  1171→        return GoResponseField(name='', go_type='unknown')
  1172→    # Primitive literals.
  1173→    if val.startswith('"'):
  1174→        return GoResponseField(name='', go_type='string')
  1175→    if val in {'true', 'false'}:
  1176→        return GoResponseField(name='', go_type='bool')
  1177→    if re.match(r'^-?\d+(\.\d+)?$', val):
  1178→        return GoResponseField(name='', go_type='number')
  1179→    # Function call or complex expression — unknown.
  1180→    return GoResponseField(name='', go_type='unknown')
  1181→

  1183→def _strip_package(type_str: str) -> str:
  1184→    """``models.User`` -> ``User``; ``User`` -> ``User``."""
  1185→    if '.' in type_str:
  1186→        return type_str.rsplit('.', 1)[-1]
  1187→    return type_str

  1191→# Go routes parser
  1192→# ---------------------------------------------------------------------------
  1193→
  1194→# `<var> := <parent>.PathPrefix("<prefix>").Subrouter()` — captures
  1195→# the new var, the parent var, and the prefix.
  1196→RE_SUBROUTER = re.compile(
  1197→    r'(\w+)\s*:?=\s*(\w+)\.PathPrefix\(\s*"([^"]*)"\s*\)\.Subrouter\(\)'
  1198→)
  1199→
  1200→# `<router>.HandleFunc("<path>", <handler>)` — captures router, path.
  1201→RE_ROUTER_CALL = re.compile(
  1202→    r'(\w+)\.(HandleFunc|Handle)\(\s*"([^"]*)"\s*,'
  1203→)
  1204→
  1205→# `.Methods("GET")` after a HandleFunc call.
  1206→RE_METHODS = re.compile(r'\.Methods\(\s*"([A-Z]+)"\s*\)')
  1207→
  1208→# `<var> := <pkg>.NewXHandler(...)` — captures the var and the
  1209→# constructor name (we strip the leading `New` to get the struct name).
  1210→# Matches both `handlers.NewAuthHandler(...)` and `billing.NewFooHandler(...)`.
  1211→RE_HANDLER_VAR = re.compile(
  1212→    r'(\w+)\s*:?=\s*\w+\.New(\w+Handler)\s*\('
  1213→)
  1214→
  1215→# Wrapper functions to peel off when extracting the handler reference.
  1216→WRAPPER_FUNCS = {
  1217→    'RateLimitHandler', 'RequireAuth', 'RequireRole', 'RequireTenant',
  1218→    'RequireRootTenant', 'RequireActiveBilling', 'HandlerFunc',
  1219→}
  1220→
  1221→# Packages that appear in `pkg.Func` form inside handler expressions
  1222→# but are NOT handlers themselves.
  1223→NON_HANDLER_PKGS = {
  1224→    'http', 'middleware', 'rateLimiter', 'models', 'mux', 'r', 'fmt',
  1225→    'strings', 'context', 'time', 'os', 'syscall', 'signal', 'path',
  1226→    'filepath', 'bson', 'options', 'primitive', 'slog', 'json',
  1227→    'mux', 'cors', 'email', 'auth', 'events', 'syslog', 'stripe',
  1228→    'configstore', 'config', 'health', 'telemetry', 'testutil',
  1229→    'validation', 'version',
  1230→}

  1233→def parse_go_routes(content: str, file: str,
  1234→                    handler_vars: Optional[Dict[str, str]] = None
  1235→                    ) -> List[GoRoute]:
  1236→    """Parse a Go file's router setup and return a list of GoRoute.
  1237→
  1238→    ``content`` should be the RAW (unmasked) source so path strings
  1239→    are preserved. The patterns we match (``HandleFunc``, ``PathPrefix``,
  1240→    ``Subrouter``, ``Methods``) don't appear inside Go string literals
  1241→    in this codebase, so masking isn't needed for route parsing.
  1242→    """
  1243→    routes: List[GoRoute] = []
  1244→    # 1. Collect handler var -> struct name mappings from constructor
  1245→    #    calls (e.g. `authHandler := handlers.NewAuthHandler(...)`).
  1246→    if handler_vars is None:
  1247→        handler_vars = {}
  1248→    for m in RE_HANDLER_VAR.finditer(content):
  1249→        var = m.group(1)
  1250→        ctor = m.group(2)
  1251→        # `NewAuthHandler` -> `AuthHandler` is already correct since
  1252→        # the regex captures `(\w+Handler)` after `New`. But the
  1253→        # constructor name might be `NewX` without the `Handler`
  1254→        # suffix — in that case we just use the captured name as-is.
  1255→        struct = ctor if ctor.endswith('Handler') else ctor
  1256→        handler_vars.setdefault(var, struct)
  1257→
  1258→    # 2. Collect subrouter var -> (parent_var, prefix).
  1259→    raw_subrouters: List[Tuple[str, str, str, int]] = []
  1260→    for m in RE_SUBROUTER.finditer(content):
  1261→        var = m.group(1)
  1262→        parent = m.group(2)
  1263→        prefix = m.group(3)
  1264→        raw_subrouters.append((var, parent, prefix, m.start()))
  1265→
  1266→    # Resolve to absolute prefixes (parent's prefix + this prefix)
  1267→    # via fixpoint iteration — subrouters may reference other subrouters.
  1268→    resolved: Dict[str, Tuple[str, str]] = {}  # var -> (root, full_prefix)
  1269→    known_routers = {'router', 'mux', 'r', 'mainRouter', 'http'}
  1270→    for v in known_routers:
  1271→        resolved[v] = (v, '')
  1272→    # The top-level `api` subrouter is mounted at /api — seed it too.
  1273→    # (We'll let the iteration pick it up via the PathPrefix chain.)
  1274→
  1275→    changed = True
  1276→    iterations = 0
  1277→    while changed and iterations < 10:
  1278→        changed = False
  1279→        iterations += 1
  1280→        for var, parent, prefix, _ in raw_subrouters:
  1281→            if parent in resolved:
  1282→                parent_root, parent_prefix = resolved[parent]
  1283→                full = parent_prefix + prefix
  1284→                if var not in resolved or resolved[var][1] != full:
  1285→                    resolved[var] = (parent_root, full)
  1286→                    changed = True
  1287→    subrouters = {v: p for v, p in resolved.items() if v not in known_routers}
  1288→
  1289→    # 3. Scan for HandleFunc / Handle calls.
  1290→    for m in RE_ROUTER_CALL.finditer(content):
  1291→        router_var = m.group(1)
  1292→        path_str = m.group(3)
  1293→        # Find the open paren of the args list (between the call kind
  1294→        # and the path string).
  1295→        open_paren_idx = content.find('(', m.start(2))
  1296→        if open_paren_idx == -1 or open_paren_idx > m.end():
  1297→            continue
  1298→        args, close_idx = find_matching_paren(content, open_paren_idx)
  1299→        if args is None:
  1300→            continue
  1301→        # The args look like: `"/path", <handler_expr>` — extract the
  1302→        # handler expr by stripping the leading path string.
  1303→        am = re.match(r'\s*"' + re.escape(path_str) + r'"\s*,\s*(.*)',
  1304→                      args, re.DOTALL)
  1305→        if not am:
  1306→            am = re.match(r'\s*"[^"]*"\s*,\s*(.*)', args, re.DOTALL)
  1307→            if not am:
  1308→                continue
  1309→        handler_expr = am.group(1).strip()
  1310→
  1311→        # Look for `.Methods("X")` after the closing paren.
  1312→        method = 'GET'  # default
  1313→        tail = content[close_idx:close_idx + 200]
  1314→        mm = RE_METHODS.search(tail)
  1315→        if mm and mm.start() < 60:
  1316→            method = mm.group(1)
  1317→
  1318→        # Resolve the handler reference.
  1319→        handler_func = _extract_handler_name(handler_expr, handler_vars)
  1320→        if not handler_func:
  1321→            # Fall back to the raw expr so the route isn't lost.
  1322→            handler_func = handler_expr[:60] or '(unknown)'
  1323→
  1324→        # Build the full path.
  1325→        prefix_info = subrouters.get(router_var)
  1326→        full_prefix = prefix_info[1] if prefix_info else ''
  1327→        full_path = full_prefix + path_str
  1328→        full_path = re.sub(r'/+', '/', full_path)
  1329→        if not full_path.startswith('/'):
  1330→            full_path = '/' + full_path
  1331→
  1332→        routes.append(GoRoute(
  1333→            method=method,
  1334→            path=full_path,
  1335→            handler_func=handler_func,
  1336→            file=file,
  1337→            line=line_of(content, m.start()),
  1338→        ))
  1339→    return routes
  1340→
  1341→
  1342→def _extract_handler_name(expr: str,
  1343→                          handler_vars: Dict[str, str]) -> Optional[str]:
  1344→    """Extract a ``StructType.Method`` label from a handler expression.
  1345→
  1346→    Handles:
  1347→      * ``authHandler.Register`` -> ``AuthHandler.Register`` (when
  1348→        ``handler_vars`` maps ``authHandler`` -> ``AuthHandler``)
  1349→      * ``handlers.DocsHTML`` -> ``handlers.DocsHTML`` (package-level
  1350→        function reference)
  1351→      * ``rateLimiter.RateLimitHandler(cfg, fn, authHandler.Login)`` ->
  1352→        ``AuthHandler.Login`` (peels wrappers, prefers known handler
  1353→        vars)
  1354→      * ``authMiddleware.RequireAuth(http.HandlerFunc(
  1355→        plansHandler.ListPlansPublic))`` -> ``PlansHandler.ListPlansPublic``
  1356→    """
  1357→    expr = expr.strip()
  1358→    # Find all `var.Method` references in order of appearance.
  1359→    refs = list(re.finditer(r'\b([a-zA-Z_]\w*)\.([A-Z]\w*)\b', expr))
  1360→    if not refs:
  1361→        # Maybe a bare function name: `MyHandler` passed as a value.
  1362→        bare = re.match(r'^\s*([A-Z]\w*)\s*[\),]', expr)
  1363→        if bare:
  1364→            return bare.group(1)
  1365→        return None
  1366→    # Filter out non-handler package refs and known wrappers.
  1367→    candidates: List[Tuple[str, str]] = []
  1368→    for rm in refs:
  1369→        var = rm.group(1)
  1370→        method = rm.group(2)
  1371→        if method in WRAPPER_FUNCS:
  1372→            continue
  1373→        if var in NON_HANDLER_PKGS:
  1374→            continue
  1375→        candidates.append((var, method))
  1376→    if not candidates:
  1377→        # Fall back to all refs.
  1378→        candidates = [(rm.group(1), rm.group(2)) for rm in refs]
  1379→    # Prefer the first candidate whose var is a known handler var.
  1380→    for var, method in candidates:
  1381→        if var in handler_vars:
  1382→            return f'{handler_vars[var]}.{method}'
  1383→    # Fall back to the LAST candidate (rate-limited routes pass the
  1384→    # real handler as the final positional arg).
  1385→    var, method = candidates[-1]
  1386→    if var in handler_vars:
  1387→        return f'{handler_vars[var]}.{method}'
  1388→    # Heuristic: title-case the first letter of the var.
  1389→    struct = var[0].upper() + var[1:] if var else var
  1390→    return f'{struct}.{method}'
  1391→

  1396→
  1397→# `interface Name {` or `interface Name extends X, Y {`
  1398→RE_INTERFACE_OPEN = re.compile(
  1399→    r'\binterface\s+(?P<name>[A-Z]\w*)\s*(?:extends\s+[^{]+)?\{'
  1400→)
  1401→
  1402→# `type Name = ...` — captures name. The body may be an object literal,
  1403→# a union, or a primitive. We only care about object literals here.
  1404→RE_TYPE_ALIAS = re.compile(
  1405→    r'\btype\s+(?P<name>[A-Z]\w*)\s*=\s*'
  1406→)
  1407→
  1408→# A field inside an interface body: `name: type;` or `name?: type;`.
  1409→# Type may be a union, an array, a generic, or a nested object.
  1410→RE_TS_FIELD = re.compile(
  1411→    r'^\s*(?P<name>[A-Za-z_]\w*)\s*(?P<opt>\?)?\s*:\s*(?P<type>[^;]+?)\s*;?\s*$'
  1412→)

  1414→
  1415→def parse_ts_types(content: str, file: str) -> List[TSType]:
  1416→    """Parse every interface and object-literal type alias in a .ts file."""
  1417→    out: List[TSType] = []
  1418→    for m in RE_INTERFACE_OPEN.finditer(content):
  1419→        name = m.group('name')
  1420→        line = line_of(content, m.start())
  1421→        brace_idx = content.find('{', m.start())
  1422→        if brace_idx == -1:
  1423→            continue
  1424→        body, _ = find_matching_brace(content, brace_idx, '{', '}')
  1425→        if body is None:
  1426→            continue
  1427→        fields = _parse_ts_fields(body)
  1428→        out.append(TSType(
  1429→            name=name, file=file, line=line, kind='interface',
  1430→            fields=fields, raw=body,
  1431→        ))
  1432→    for m in RE_TYPE_ALIAS.finditer(content):
  1433→        name = m.group('name')
  1434→        line = line_of(content, m.start())
  1435→        # Look at what follows the `=`. If it starts with `{`, it's an
  1436→        # object literal type. If it's a string/union literal, capture
  1437→        # the raw type text. Otherwise skip.
  1438→        rest = content[m.end():]
  1439→        stripped = rest.lstrip()
  1440→        if stripped.startswith('{'):
  1441→            # Find the matching closing brace.
  1442→            brace_idx = m.end() + (len(rest) - len(stripped))
  1443→            body, _ = find_matching_brace(content, brace_idx, '{', '}')
  1444→            if body is None:
  1445→                continue
  1446→            fields = _parse_ts_fields(body)
  1447→            out.append(TSType(
  1448→                name=name, file=file, line=line, kind='type_alias',
  1449→                fields=fields, raw=body,
  1450→            ))
  1451→        else:
  1452→            # String/union alias like `type LogSeverity = 'a' | 'b'`.
  1453→            # Capture the raw value up to the next newline or `;`.
  1454→            end = re.search(r'[;\n]', rest)
  1455→            raw = rest[:end.start()] if end else rest
  1456→            out.append(TSType(
  1457→                name=name, file=file, line=line, kind='type_alias',
  1458→                fields=[], raw=raw.strip(),
  1459→            ))
  1460→    return out
  1461→
  1462→
  1463→def _parse_ts_fields(body: str) -> List[TSField]:
  1464→    """Parse the body of an interface or object-literal type alias.
  1465→
  1466→    Handles both newline-separated fields (the common case in
  1467→    ``interface X { ... }``) and semicolon-separated fields on a single
  1468→    line (the common case in inline object types like
  1469→    ``{ user: User; memberships: MembershipInfo[] }``).
  1470→    """
  1471→    fields: List[TSField] = []
  1472→    # Split the body into individual field declarations. We split on
  1473→    # top-level semicolons AND newlines (a field may be followed by
  1474→    # either). Brackets `{}<>` are tracked so semicolons inside nested
  1475→    # types or generics don't split.
  1476→    declarations = _split_ts_declarations(body)
  1477→    for decl in declarations:
  1478→        s = decl.strip()
  1479→        if not s or s.startswith('//') or s.startswith('/*'):
  1480→            continue
  1481→        # Skip closing braces of nested objects.
  1482→        if s in {'}', '};', '}),', '},', '{'}:
  1483→            continue
  1484→        m = RE_TS_FIELD.match(s)
  1485→        if not m:
  1486→            continue
  1487→        name = m.group('name')
  1488→        is_optional = bool(m.group('opt'))
  1489→        type_str = m.group('type').strip()
  1490→        # If the type is an inline object that spans multiple lines
  1491→        # (starts with `{` but doesn't end with `}`), the split logic
  1492→        # above already kept the full object — rejoin if needed.
  1493→        # Actually `_split_ts_declarations` keeps braces balanced, so
  1494→        # `type_str` should already be complete.
  1495→        struct_ref, is_array = _classify_ts_type(type_str)
  1496→        fields.append(TSField(
  1497→            name=name, ts_type=type_str, is_optional=is_optional,
  1498→            struct_ref=struct_ref, is_array=is_array,
  1499→        ))
  1500→    return fields
  1501→
  1502→
  1503→def _split_ts_declarations(body: str) -> List[str]:
  1504→    """Split a TS interface/type body into individual field declarations.
  1505→
  1506→    Splits on top-level semicolons and newlines, tracking depth in
  1507→    ``{}``, ``[]``, ``<>``, and ``()`` so nested types aren't split.
  1508→    """
  1509→    out: List[str] = []
  1510→    cur: List[str] = []
  1511→    depth = 0
  1512→    for c in body:
  1513→        if c in '{[<(':
  1514→            depth += 1
  1515→            cur.append(c)
  1516→        elif c in '}]>)':
  1517→            depth -= 1
  1518→            cur.append(c)
  1519→        elif c == ';' and depth == 0:
  1520→            out.append(''.join(cur))
  1521→            cur = []
  1522→        elif c == '\n' and depth == 0:
  1523→            out.append(''.join(cur))
  1524→            cur = []
  1525→        else:
  1526→            cur.append(c)
  1527→    if cur:
  1528→        out.append(''.join(cur))
  1529→    return out
  1530→

  1532→def _classify_ts_type(type_str: str) -> Tuple[Optional[str], bool]:
  1533→    """Return ``(struct_ref, is_array)`` for a TS type expression.
  1534→
  1535→    * ``User`` -> (``"User"``, False)
  1536→    * ``User[]`` -> (``"User"``, True)
  1537→    * ``Array<User>`` -> (``"User"``, True)
  1538→    * ``string`` / ``number`` / ``boolean`` -> (None, False)
  1539→    * ``'a' | 'b'`` -> (None, False)  (string literal union)
  1540→    * ``Record<string, X>`` -> (None, False)
  1541→    * ``{ sub: type }`` -> (None, False)  (inline object)
  1542→    """
  1543→    t = type_str.strip()
  1544→    # Array forms.
  1545→    if t.endswith('[]'):
  1546→        inner = t[:-2].strip()
  1547→        ref = _extract_named_ref(inner)
  1548→        return (ref, True)
  1549→    m = re.match(r'^Array<(.+)>$', t)
  1550→    if m:
  1551→        inner = m.group(1).strip()
  1552→        ref = _extract_named_ref(inner)
  1553→        return (ref, True)
  1554→    # Plain named type.
  1555→    ref = _extract_named_ref(t)
  1556→    return (ref, False)
  1557→
  1558→
  1559→def _extract_named_ref(type_str: str) -> Optional[str]:
  1560→    """If ``type_str`` is a single named identifier (e.g. ``User``,
  1561→    ``MembershipInfo``), return it. Otherwise return None.
  1562→    """
  1563→    t = type_str.strip()
  1564→    # Strip parens.
  1565→    while t.startswith('(') and t.endswith(')'):
  1566→        t = t[1:-1].strip()
  1567→    if re.match(r'^[A-Z]\w*$', t):
  1568→        return t
  1569→    # `import('...').X` form — extract X.
  1570→    m = re.match(r'^import\([^)]*\)\.([A-Z]\w*)$', t)
  1571→    if m:
  1572→        return m.group(1)
  1573→    return None

  1576→# ---------------------------------------------------------------------------
  1577→# TS client parser
  1578→# ---------------------------------------------------------------------------
  1579→
  1580→# `api.<method><<Type>>(url, ...)` — captures method, optional Type,
  1581→# and the URL (string or template literal).
  1582→RE_API_CALL = re.compile(
  1583→    r'\bapi\.(?P<method>get|post|put|patch|delete)'

  1595→def parse_ts_client(content: str, file: str) -> List[TSEndpoint]:
  1596→    """Parse ``client.ts`` and extract every API endpoint.
  1597→
  1598→    For each ``api.<method><<Type>>(url, ...)`` call we capture:
  1599→      * HTTP method (uppercased)
  1600→      * URL (with ``${var}`` normalised to ``{var}`` and ``/api`` prepended)
  1601→      * Response type (raw text)
  1602→    """
  1603→    endpoints: List[TSEndpoint] = []
  1604→    # We need to also capture the surrounding API group and method name
  1605→    # for context. Walk the file looking for `export const xxxApi = {`
  1606→    # blocks and attribute each api.* call to the most recently seen
  1607→    # group.
  1608→    current_group = ''
  1609→    current_method = ''
  1610→    lines = content.splitlines()
  1611→    # Build a map of line -> (group, method) for context attribution.
  1612→    context_by_line: Dict[int, Tuple[str, str]] = {}
  1613→    line_no = 0
  1614→    in_group = False
  1615→    for line in lines:
  1616→        line_no += 1
  1617→        # `export const authApi = {`
  1618→        gm = re.match(r'^\s*export\s+const\s+(\w+Api)\s*=\s*\{', line)
  1619→        if gm:
  1620→            current_group = gm.group(1)
  1621→            current_method = ''
  1622→            in_group = True
  1623→            continue
  1624→        if in_group:
  1625→            # `methodName: (args) =>` — capture method name.
  1626→            mm = re.match(r'^\s*(\w+)\s*:\s*(?:async\s*)?\(', line)
  1627→            if mm:
  1628→                current_method = mm.group(1)
  1629→            # Detect end of group: a line with just `};` at column 0.
  1630→            if re.match(r'^\};?\s*$', line):
  1631→                in_group = False
  1632→                current_group = ''
  1633→                current_method = ''
  1634→            else:
  1635→                context_by_line[line_no] = (current_group, current_method)
  1636→
  1637→    for m in RE_API_CALL.finditer(content):
  1638→        method = m.group('method').upper()
  1639→        type_str = (m.group('type') or '').strip()
  1640→        line = line_of(content, m.start())
  1641→        # The URL follows the open paren.
  1642→        rest = content[m.end():]
  1643→        url = _extract_url(rest)
  1644→        if url is None:
  1645→            continue
  1646→        normalised = _normalise_ts_url(url)
  1647→        # Classify the response type.
  1648→        response_kind, response_fields = _classify_ts_response(type_str)
  1649→        group, method_name = context_by_line.get(line, ('', ''))
  1650→        endpoints.append(TSEndpoint(
  1651→            method=method,
  1652→            path=normalised,
  1653→            response_type=type_str or '(unknown)',
  1654→            response_kind=response_kind,
  1655→            response_fields=response_fields,
  1656→            file=file,
  1657→            line=line,
  1658→            api_group=group,
  1659→            method_name=method_name,
  1660→        ))
  1661→    return endpoints

  1664→def _extract_url(rest: str) -> Optional[str]:
  1665→    """Extract the URL string (or template literal) from the start of
  1666→    ``rest`` (the text immediately after ``api.method<TYPE>(``).
  1667→    """
  1668→    # Try a simple quoted string first.
  1669→    m = RE_URL_STRING.match(rest)
  1670→    if m:
  1671→        return m.group('url')
  1672→    # Try a template literal.
  1673→    m = RE_URL_TEMPLATE.match(rest)
  1674→    if m:
  1675→        return m.group('url')
  1676→    # Backtick template that may span multiple lines — find the next
  1677→    # backtick.
  1678→    if rest.lstrip().startswith('`'):
  1679→        idx = rest.find('`')
  1680→        end = rest.find('`', idx + 1)
  1681→        if end != -1:
  1682→            return rest[idx + 1:end]

  1686→def _normalise_ts_url(url: str) -> str:
  1687→    """Normalise a TS URL to match the Go route form.
  1688→
  1689→    * Convert ``${var}`` to ``{var}`` (gorilla/mux syntax).
  1690→    * Prepend ``/api`` if not already present.
  1691→    * Collapse duplicate slashes.
  1692→    """
  1693→    # Replace ${var} with {var}.
  1694→    url = re.sub(r'\$\{(\w+)\}', r'{\1}', url)
  1695→    # Prepend /api if missing.
  1696→    if not url.startswith('/api'):
  1697→        if url.startswith('/'):
  1698→            url = '/api' + url
  1699→        else:
  1700→            url = '/api/' + url
  1701→    # Collapse duplicate slashes (but keep the leading double-slash if any).
  1702→    url = re.sub(r'/+', '/', url)
  1703→    return url

  1706→def _classify_ts_response(type_str: str) -> Tuple[str, List[TSField]]:
  1707→    """Classify a TS response type string and parse inline fields.
  1708→
  1709→    Returns ``(kind, fields)`` where kind is one of
  1710→    ``"named"``, ``"inline"``, ``"unknown"``.
  1711→    """
  1712→    if not type_str:
  1713→        return ('unknown', [])
  1714→    t = type_str.strip()
  1715→    # Strip a union like `AuthResponse | MFARequiredResponse` — take
  1716→    # the first named branch.
  1717→    if '|' in t and not t.startswith('{'):
  1718→        bits = [b.strip() for b in t.split('|')]
  1719→        for b in bits:
  1720→            if re.match(r'^[A-Z]\w*$', b):
  1721→                return ('named', [])
  1722→        return ('unknown', [])
  1723→    # Inline object literal: `{ user: User; memberships: MembershipInfo[] }`.
  1724→    if t.startswith('{'):
  1725→        # Parse the fields.
  1726→        body = t[1:]
  1727→        # Find the matching closing brace (the type string may have
  1728→        # been truncated by the `<...>` regex if it contained `>` —
  1729→        # best-effort: strip a trailing `}` if present).
  1730→        if body.endswith('}'):
  1731→            body = body[:-1]
  1732→        fields = _parse_ts_fields(body)
  1733→        return ('inline', fields)
  1734→    # Named type.
  1735→    if re.match(r'^[A-Z]\w*$', t):
  1736→        return ('named', [])
  1737→    return ('unknown', [])

  1811→def normalise_go_type(go_type: str) -> str:
  1812→    """Normalise a Go type to a canonical form for comparison with TS.
  1813→
  1814→    Examples:
  1815→      ``string``           -> ``string``
  1816→      ``*string``          -> ``string?``
  1817→      ``int64``            -> ``number``
  1818→      ``*time.Time``       -> ``string?``
  1819→      ``primitive.ObjectID`` -> ``string``
  1820→      ``models.User``      -> ``User``
  1821→      ``[]string``         -> ``string[]``
  1822→      ``[]models.User``    -> ``User[]``
  1823→      ``map[string]X``     -> ``Record<string, X>``
  1824→      ``MemberRole``       -> ``string`` (if ``type MemberRole string``
  1825→      was found in the codebase)
  1826→    """
  1827→    t = go_type.strip()
  1828→    # Strip a leading `*` (pointer).
  1829→    optional = False
  1830→    if t.startswith('*'):
  1831→        optional = True
  1832→        t = t[1:].strip()
  1833→    # Slice.
  1834→    if t.startswith('[]'):
  1835→        inner = t[2:].strip()
  1836→        return normalise_go_type(inner) + '[]'
  1837→    # Map (best-effort — we don't recurse into the value type).
  1838→    if t.startswith('map['):
  1839→        # Extract the value type.
  1840→        m = re.match(r'^map\[[^\]]+\](.+)$', t)
  1841→        if m:
  1842→            val = normalise_go_type(m.group(1).strip())
  1843→            return f'Record<string, {val}>'
  1844→    # String-serialising types (check BEFORE stripping the package
  1845→    # prefix — the set contains fully-qualified names like
  1846→    # `time.Time` and `primitive.ObjectID`).
  1847→    if t in GO_STRING_TYPES:
  1848→        return 'string' + ('?' if optional else '')
  1849→    # Package-qualified type.
  1850→    if '.' in t:
  1851→        base = t.rsplit('.', 1)[-1]
  1852→        # Check if the base is a known primitive alias.
  1853→        if base in GO_PRIMITIVE_ALIASES:
  1854→            underlying = GO_PRIMITIVE_ALIASES[base]
  1855→            return normalise_go_type(underlying) + ('?' if optional else '')
  1856→        return base + ('?' if optional else '')
  1857→    # Primitive alias (e.g. ``MemberRole`` -> ``string``).
  1858→    if t in GO_PRIMITIVE_ALIASES:
  1859→        underlying = GO_PRIMITIVE_ALIASES[t]
  1860→        return normalise_go_type(underlying) + ('?' if optional else '')
  1861→    # Primitive mappings.
  1862→    if t in GO_NUMERIC_TYPES:
  1863→        return 'number' + ('?' if optional else '')
  1864→    if t == 'bool':
  1865→        return 'boolean' + ('?' if optional else '')
  1866→    if t == 'interface{}' or t == 'any':
  1867→        return 'unknown'
  1868→    return t + ('?' if optional else '')

  1871→def normalise_ts_type(ts_type: str) -> str:
  1872→    """Normalise a TS type to a canonical form for comparison with Go.
  1873→
  1874→    Examples:
  1875→      ``string``           -> ``string``
  1876→      ``string | undefined`` -> ``string?``
  1877→      ``number``           -> ``number``
  1878→      ``User``             -> ``User``
  1879→      ``User[]``           -> ``User[]``
  1880→      ``'a' | 'b'``        -> ``string``  (literal union collapses to string)
  1881→      ``Record<string, X>`` -> ``Record<string, X>``
  1882→      ``ConfigVarType``    -> ``string``  (if the alias resolves to a
  1883→      string literal union)
  1884→    """
  1885→    t = ts_type.strip()
  1886→    # Strip parens.
  1887→    while t.startswith('(') and t.endswith(')'):
  1888→        t = t[1:-1].strip()
  1889→    # Named alias that resolves to a primitive (e.g. ``ConfigVarType``
  1890→    # whose definition is ``'string' | 'numeric' | 'enum' | 'template'``).
  1891→    if re.match(r'^[A-Z]\w*$', t) and t in TS_PRIMITIVE_ALIASES:
  1892→        return TS_PRIMITIVE_ALIASES[t]
  1893→    # Union with undefined/null -> optional.
  1894→    optional = False
  1895→    if '|' in t:
  1896→        bits = [b.strip() for b in t.split('|')]
  1897→        if 'undefined' in bits:
  1898→            optional = True
  1899→            bits = [b for b in bits if b != 'undefined']
  1900→        # If all remaining bits are string literals, collapse to string.
  1901→        if bits and all(re.match(r"^['\"]", b) for b in bits):
  1902→            return 'string' + ('?' if optional else '')
  1903→        # If all remaining bits are number literals, collapse to number.
  1904→        if bits and all(re.match(r'^-?\d+(\.\d+)?$', b) for b in bits):
  1905→            return 'number' + ('?' if optional else '')
  1906→        # If exactly one non-null branch remains, recurse.
  1907→        if len(bits) == 1:
  1908→            t = bits[0]
  1909→        else:
  1910→            # Mixed union — keep as a sorted union string.
  1911→            return ' | '.join(sorted(bits)) + ('?' if optional else '')
  1912→    # Array.
  1913→    if t.endswith('[]'):
  1914→        inner = t[:-2].strip()
  1915→        return normalise_ts_type(inner) + '[]'
  1916→    m = re.match(r'^Array<(.+)>$', t)
  1917→    if m:
  1918→        return normalise_ts_type(m.group(1).strip()) + '[]'
  1919→    # String literal.
  1920→    if t.startswith("'") or t.startswith('"'):
  1921→        return 'string' + ('?' if optional else '')
  1922→    # Boolean literal.
  1923→    if t in {'true', 'false'}:
  1924→        return 'boolean' + ('?' if optional else '')
  1925→    # Number literal.
  1926→    if re.match(r'^-?\d+(\.\d+)?$', t):
  1927→        return 'number' + ('?' if optional else '')
  1928→    return t + ('?' if optional else '')

  1931→def types_compatible(go_type: str, ts_type: str) -> Tuple[bool, str]:
  1932→    """Return ``(is_compatible, note)`` for a Go↔TS type pair.
  1933→
  1934→    The note explains the mismatch when not compatible.
  1935→
  1936→    A Go type of ``unknown`` (used when the response payload contains a
  1937→    value the auditor can't statically resolve — e.g. a package-qualified
  1938→    constant like ``version.Current``) is treated as compatible with any
  1939→    TS type, since we can't know the actual type without type info.
  1940→    """
  1941→    g = normalise_go_type(go_type)
  1942→    t = normalise_ts_type(ts_type)
  1943→    # `unknown` Go types are compatible with anything — we can't know
  1944→    # the actual type without full Go type inference.
  1945→    if g == 'unknown' or go_type.strip() == 'unknown':
  1946→        return (True, 'Go type unknown — not statically resolvable')
  1947→    if g == t:
  1948→        return (True, '')
  1949→    # Strip optional markers for the comparison — optionality is
  1950→    # handled separately (a Go field is optional if it's a pointer or
  1951→    # omitempty; a TS field is optional if marked with `?`).
  1952→    g_base = g.rstrip('?')
  1953→    t_base = t.rstrip('?')
  1954→    if g_base == t_base:
  1955→        return (True, 'optional differs')
  1956→    # Record<string, X> on both sides — recurse on X (best-effort: skip).
  1957→    if g_base.startswith('Record<string,') and t_base.startswith('Record<string,'):
  1958→        return (True, 'record values not deeply compared')
  1959→    # Known mismatch patterns:
  1960→    # Go int64 / number vs TS string — classic JS precision bug for IDs.
  1961→    if g_base == 'number' and t_base == 'string':
  1962→        return (False, f'Go {go_type} (number) vs TS {ts_type} (string) — possible ID precision bug')
  1963→    if g_base == 'string' and t_base == 'number':
  1964→        return (False, f'Go {go_type} (string) vs TS {ts_type} (number)')
  1965→    if g_base == 'boolean' and t_base != 'boolean':
  1966→        return (False, f'Go {go_type} (boolean) vs TS {ts_type}')
  1967→    # Named struct references that differ in name.
  1968→    if g_base and t_base and g_base[0].isupper() and t_base[0].isupper():
  1969→        if g_base != t_base:
  1970→            return (False, f'struct name mismatch: Go {g_base} vs TS {t_base}')
  1971→    return (False, f'Go {g} vs TS {t}')
