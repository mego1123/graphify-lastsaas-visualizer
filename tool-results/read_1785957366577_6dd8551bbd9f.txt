  1667→def _extract_pipeline_match_fields(text: str) -> list[str]:
  1668→    """Extract field names from ``$match`` stages in an aggregate pipeline.
  1669→
  1670→    A ``$match`` stage looks like ``{"$match": bson.M{...}}`` or
  1671→    ``bson.D{{Key: "$match", Value: bson.D{...}}}``. We locate the inner
  1672→    filter literal and delegate to :func:`extract_filter_fields_from_text`,
  1673→    passing enough surrounding context (the ``bson.M`` / ``bson.D`` prefix)
  1674→    that the BSON regexes can match.
  1675→    """
  1676→    fields: list[str] = []
  1677→    match_re = re.compile(
  1678→        r'"\$match"\s*:\s*(bson\.M|bson\.D)\s*\{', re.IGNORECASE
  1679→    )
  1680→    for m in match_re.finditer(text):
  1681→        # Find the closing brace of the ``bson.M{...}`` block.
  1682→        brace_idx = m.end() - 1
  1683→        end = find_block_end(text, brace_idx, "{", "}")
  1684→        if end == -1:
  1685→            continue
  1686→        # Include the ``bson.M`` prefix in the slice so that
  1687→        # ``BSON_M_RE`` / ``BSON_D_RE`` (which require the prefix) match.
  1688→        bson_prefix_start = m.start(1)
  1689→        inner = text[bson_prefix_start : end + 1]
  1690→        fields.extend(extract_filter_fields_from_text(inner))
  1691→    # Dedupe.
  1692→    seen: set[str] = set()
  1693→    out: list[str] = []
  1694→    for f in fields:
  1695→        if f and not f.startswith("$") and f not in seen:
  1696→            seen.add(f)
  1697→            out.append(f)
  1698→    return out

  1701→def _extract_struct_name(text: str) -> Optional[str]:
  1702→    """Best-effort: extract the struct name from an argument like
  1703→    ``models.User{...}`` or ``&models.User{...}`` or ``User{...}``.
  1704→    """
  1705→    m = re.search(r"&?([\w\.]+)\s*\{", text)
  1706→    if m:
  1707→        # Take the last component after any dots.
  1708→        name = m.group(1).split(".")[-1]
  1709→        return name
  1710→    return None

  1722→def _accessor_to_collection(accessor: str) -> str:
  1723→    """Convert a PascalCase accessor name (e.g. ``Users``) to its snake_case
  1724→    collection name (e.g. ``users``). Best-effort.
  1725→    """
  1726→    # Handle common acronyms first.

  1728→    cur = ""
  1729→    for ch in accessor:
  1730→        if ch.isupper():
  1731→            if cur:
  1732→                out.append(cur)
  1733→                cur = ch.lower()
  1734→            else:
  1735→                cur = ch.lower()
  1736→        else:
  1737→            cur += ch
  1738→    if cur:
  1739→        out.append(cur)

  1753→def _build_snippet(raw_lines: list[str], line_no: int, filter_text: str) -> str:
  1754→    """Build a compact human-readable snippet for the report.
  1755→
  1756→    Uses the raw (unmasked) source lines so the snippet shows the actual code
  1757→    the developer wrote (with strings/comments intact).
  1758→    """
  1759→    if 1 <= line_no <= len(raw_lines):
  1760→        first = raw_lines[line_no - 1].strip()
  1761→    else:
  1762→        first = ""
  1763→    # Include up to 3 following lines if the call clearly continues.
  1764→    snippet_lines = [first]
  1765→    depth_paren = first.count("(") - first.count(")")
  1766→    depth_brace = first.count("{") - first.count("}")
  1767→    i = line_no + 1
  1768→    while (depth_paren > 0 or depth_brace > 0) and i <= len(raw_lines) and (i - line_no) < 5:
  1769→        nxt = raw_lines[i - 1].rstrip()
  1770→        snippet_lines.append(nxt)
  1771→        depth_paren += nxt.count("(") - nxt.count(")")
  1772→        depth_brace += nxt.count("{") - nxt.count("}")
  1773→        i += 1
  1774→    snippet = "\n".join(snippet_lines).strip()
  1775→    if len(snippet) > 400:
  1776→        snippet = snippet[:397] + "..."
  1777→    return snippet

  1785→def build_summary(
  1786→    queries: list[Query],
  1787→    file_index: dict[str, FileInfo],
  1788→    accessor_map: dict[str, str],
  1789→    repo_root: Path,
  1790→    collection_struct_map: Optional[dict[str, str]] = None,
  1791→    user_scoped_collections: Optional[set[str]] = None,
  1792→    global_collections: Optional[set[str]] = None,
  1793→    route_classifier: Optional[dict[str, set[str]]] = None,
  1794→) -> dict:
  1795→    """Build the JSON-serialisable summary object."""
  1796→    total = len(queries)
  1797→    with_tenant = sum(1 for q in queries if q.has_tenant_id)
  1798→    without_tenant = total - with_tenant
  1799→    violations = [q for q in queries if q.is_violation]
  1800→    critical = [q for q in violations if q.risk_level == "CRITICAL"]
  1801→    high = [q for q in violations if q.risk_level == "HIGH"]
  1802→    medium = [q for q in queries if q.risk_level == "MEDIUM"]
  1803→    ok = [q for q in queries if q.risk_level == "OK"]
  1804→    safe_key_violations = [q for q in violations if q.safe_key_filter]
  1805→    real_violations = [q for q in violations if not q.safe_key_filter]
  1806→
  1807→    # Suppression breakdown — how many queries were downgraded from a
  1808→    # potential violation to MEDIUM informational by each heuristic.
  1809→    suppression_counts: Counter = Counter()
  1810→    for q in queries:
  1811→        if q.suppression_reason:
  1812→            suppression_counts[q.suppression_reason] += 1
  1813→    route_scope_counts: Counter = Counter(q.route_scope for q in queries)
  1814→
  1815→    by_collection: Counter = Counter()
  1816→    by_operation: Counter = Counter()
  1817→    violations_by_collection: Counter = Counter()
  1818→    violations_by_file: Counter = Counter()
  1819→    for q in queries:
  1820→        by_collection[q.collection] += 1
  1821→        by_operation[q.operation] += 1
  1822→        if q.is_violation:
  1823→            violations_by_collection[q.collection] += 1
  1824→            violations_by_file[q.file] += 1
  1825→
  1826→    # Top files by violation count.
  1827→    top_files = [
  1828→        {"file": f, "violations": v, "total": file_index[f].total_queries}
  1829→        for f, v in violations_by_file.most_common(20)
  1830→    ]
  1831→
  1832→    return {
  1833→        "repo_root": str(repo_root),
  1834→        "total_queries": total,
  1835→        "queries_with_tenant_id": with_tenant,
  1836→        "queries_without_tenant_id": without_tenant,
  1837→        "violations": len(violations),
  1838→        "critical_violations": len(critical),
  1839→        "high_violations": len(high),
  1840→        "medium_global_collection_queries": len(medium),
  1841→        "ok_queries": len(ok),
  1842→        "safe_key_violations": len(safe_key_violations),
  1843→        "real_violations_needing_review": len(real_violations),
  1844→        "pct_with_tenant_id": (
  1845→            round(100.0 * with_tenant / total, 2) if total else 0.0
  1846→        ),
  1847→        "queries_by_collection": [
  1848→            {"collection": c, "queries": n}
  1849→            for c, n in by_collection.most_common()
  1850→        ],
  1851→        "queries_by_operation": [
  1852→            {"operation": op, "count": n}
  1853→            for op, n in by_operation.most_common()
  1854→        ],
  1855→        "violations_by_collection": [
  1856→            {"collection": c, "violations": n}
  1857→            for c, n in violations_by_collection.most_common()
  1858→        ],
  1859→        "top_files_by_violations": top_files,
  1860→        "exempt_collections": sorted(EXEMPT_COLLECTIONS),
  1861→        "tenant_id_fields": sorted(TENANT_ID_FIELDS),
  1862→        "safe_unique_keys": sorted(SAFE_UNIQUE_KEYS),
  1863→        "accessor_map_size": len(accessor_map),
  1864→        "suppression_breakdown": [
  1865→            {"reason": r, "count": n}
  1866→            for r, n in suppression_counts.most_common()
  1867→        ],
  1868→        "route_scope_breakdown": [
  1869→            {"scope": s, "count": n}
  1870→            for s, n in route_scope_counts.most_common()

  1872→        "user_scoped_collections": sorted(user_scoped_collections or set()),
  1873→        "global_collections": sorted(global_collections or set()),
  1874→        "collection_struct_map": collection_struct_map or {},
