     1→#!/usr/bin/env python3
     2→"""graphify_tenant_audit — tenant-isolation security auditor for Go MongoDB code.
     3→
     4→Scans all .go files under the target path and finds MongoDB queries that are
     5→missing a ``tenantId`` filter — the #1 cause of cross-tenant data leakage in a
     6→multi-tenant SaaS.
     7→
     8→For every detected query (``Find`` / ``FindOne`` / ``InsertOne`` /
     9→``UpdateOne`` / ``UpdateMany`` / ``DeleteOne`` / ``DeleteMany`` / ``Aggregate``
    10→/ ``CountDocuments`` / ``FindOneAndUpdate`` / ``FindOneAndDelete`` /
    11→``FindOneAndReplace`` / ``ReplaceOne`` / ``UpdateByID`` / ``InsertMany`` /
    12→``BulkWrite``), the script:
    13→
    14→  1. Resolves the underlying MongoDB collection (via direct ``.Collection("x")``
    15→     calls or accessor methods like ``h.db.Users()`` defined in the codebase).
    16→  2. Extracts the filter argument — handling inline ``bson.M{}`` / ``bson.D{}``
    17→     literals, ``nil``, and *variable* filters by tracing the variable back to
    18→     its definition (including dynamic field additions like
    19→     ``filter["tenantId"] = tenant.ID``).
    20→  3. Checks whether the filter contains ``tenantId`` or ``tenant_id``.
    21→  4. Classifies the query as OK / MEDIUM / HIGH / CRITICAL:
    22→       - OK       — has a ``tenantId`` filter.
    23→       - MEDIUM   — on a "global" collection (``tenants``, ``plans``,
    24→                    ``system_config``, ``system_logs``, ``users``) that
    25→                    legitimately doesn't need tenant filtering (or is
    26→                    cross-tenant by nature).
    27→       - HIGH     — read op (Find/FindOne/Aggregate/CountDocuments) without
    28→                    ``tenantId`` on a tenant-scoped collection.
    29→       - CRITICAL — write op (InsertOne/UpdateOne/DeleteOne/...) without
    30→                    ``tenantId`` on a tenant-scoped collection.
    31→
    32→  5. Records each query with file, line, collection, function, operation,
    33→     filter fields, the raw snippet, a safe-key heuristic flag (queries that
    34→     filter on a globally-unique key like ``_id`` / ``tokenHash`` / ``slug``
    35→     are noted as likely false positives), and a human-readable note.
    36→
    37→A "violation" is any query classified as HIGH or CRITICAL. MEDIUM queries on
    38→global collections are reported separately for context.
    39→
    40→Usage:
    41→    python graphify_tenant_audit.py [path] [--out report.md] [--json]
    42→
    43→Outputs:
    44→    - JSON written to /home/z/my-project/public/tenant-audit.json (best effort)
    45→    - Markdown written to /home/z/my-project/public/TENANT_AUDIT.md (best effort)
    46→    - Markdown written to --out path if specified
    47→    - JSON to stdout if --json given (without a value)
    48→
    49→Test target: /home/z/my-project/repos/lastsaas/backend
    50→"""

   480→def mask_strings_and_comments(text: str) -> str:
   481→    """Replace string literals and comments with spaces (preserving length and
   482→    newlines) so brace/paren matching is robust against braces inside strings.
   483→
   484→    Go has raw strings (```...```), interpreted strings (``"..."``), rune
   485→    literals (``'...'``), line comments (``//...``), and block comments
   486→    (``/*...*/``). All are masked here.
   487→    """
   488→    out: list[str] = []
   489→    i = 0
   490→    n = len(text)
   491→    while i < n:
   492→        c = text[i]
   493→        # Block comment
   494→        if c == "/" and i + 1 < n and text[i + 1] == "*":
   495→            end = text.find("*/", i + 2)
   496→            if end == -1:
   497→                # Unterminated — mask to end of file.
   498→                for j in range(i, n):
   499→                    out.append("\n" if text[j] == "\n" else " ")
   500→                return "".join(out)
   501→            for j in range(i, end + 2):
   502→                out.append("\n" if text[j] == "\n" else " ")
   503→            i = end + 2
   504→            continue
   505→        # Line comment
   506→        if c == "/" and i + 1 < n and text[i + 1] == "/":
   507→            end = text.find("\n", i + 2)
   508→            if end == -1:
   509→                end = n
   510→            for j in range(i, end):
   511→                out.append(" ")
   512→            i = end
   513→            continue
   514→        # Raw string literal
   515→        if c == "`":
   516→            end = text.find("`", i + 1)
   517→            if end == -1:
   518→                end = n
   519→            for j in range(i, end + 1 if end < n else n):
   520→                out.append("\n" if text[j] == "\n" else " ")
   521→            i = end + 1
   522→            continue
   523→        # Interpreted string literal
   524→        if c == '"':
   525→            j = i + 1
   526→            while j < n:
   527→                if text[j] == "\\" and j + 1 < n:
   528→                    out.append(" ")
   529→                    out.append(" ")
   530→                    j += 2
   531→                    continue
   532→                if text[j] == '"':
   533→                    break
   534→                if text[j] == "\n":
   535→                    # Unterminated string at EOL — bail.
   536→                    break
   537→                out.append(" ")
   538→                j += 1
   539→            # Mask the opening & closing quotes too.
   540→            out.append(" ")  # opening quote already added via outer loop? no.
   541→            # We didn't append the opening quote yet — fix that.
   542→            # Actually the outer loop hasn't appended c yet, so we handle it here.
   543→            i = j + 1
   544→            # Append the closing quote as a space (if present).
   545→            if j < n and text[j] == '"':
   546→                out.append(" ")
   547→            continue
   548→        # Rune literal (single-char, possibly escaped).
   549→        if c == "'":
   550→            j = i + 1
   551→            while j < n:
   552→                if text[j] == "\\" and j + 1 < n:
   553→                    j += 2
   554→                    continue
   555→                if text[j] == "'":
   556→                    break
   557→                if text[j] == "\n":
   558→                    break
   559→                j += 1
   560→            out.append(" ")  # opening '
   561→            for _ in range(i + 1, min(j + 1, n)):
   562→                out.append(" ")
   563→            i = j + 1
   564→            continue
   565→        out.append(c)
   566→        i += 1
   567→    return "".join(out)

   570→def build_accessor_map(repo_root: Path) -> dict[str, str]:
   571→    """Build {accessor_method_name: collection_name}.

   574→    return <expr>.Collection("name") }``.

   576→    accessor_map: dict[str, str] = {}

   592→def collect_struct_fields(repo_root: Path) -> dict[str, set[str]]:
   593→    """Build {struct_name: set_of_bson_field_names}.
   594→
   595→    Walks every Go struct declaration and extracts the bson field names from
   596→    struct-tag annotations (e.g. ``bson:"tenantId,omitempty"`` → ``tenantId``).
   597→    Used to check whether an Insert target struct carries a tenantId field.
   598→    """
   599→    struct_re = re.compile(r"^type\s+([A-Z]\w*)\s+struct\s*\{")
   600→    # Matches the bson tag in a struct field annotation:
   601→    #   `json:"x" bson:"tenantId,omitempty" validate:"required"`
   602→    # captures ``tenantId`` (up to the first ``,`` or ``"``).
   603→    tag_re = re.compile(r'bson:"([^",]+)')
   604→    structs: dict[str, set[str]] = {}
   605→    for path in repo_root.rglob("*.go"):
   606→        if "graphify-out" in path.parts:
   607→            continue
   608→        try:
   609→            text = path.read_text(encoding="utf-8", errors="replace")
   610→        except (OSError, UnicodeDecodeError):
   611→            continue
   612→        lines = text.splitlines()
   613→        i = 0
   614→        n = len(lines)
   615→        while i < n:
   616→            m = struct_re.match(lines[i].strip())
   617→            if not m:
   618→                i += 1
   619→                continue
   620→            name = m.group(1)
   621→            fields: set[str] = set()
   622→            i += 1
   623→            depth = 1
   624→            while i < n and depth > 0:
   625→                line = lines[i]
   626→                depth += line.count("{") - line.count("}")
   627→                if depth <= 0:
   628→                    break
   629→                # Extract the bson tag (first field name, ignoring omitempty etc.)
   630→                tm = tag_re.search(line)
   631→                if tm:
   632→                    fname = tm.group(1).split(",")[0]
   633→                    if fname and fname != "-":
   634→                        fields.add(fname)
   635→                i += 1
   636→            structs.setdefault(name, set())
   637→            structs[name].update(fields)
   638→    return structs

   641→def _collection_to_struct_name(collection: str) -> str:
   642→    """Convert a snake_case collection name to a PascalCase struct name.
   643→
   644→    Heuristic:
   645→      - "ies" suffix → "y"     (e.g. webhook_deliveries → WebhookDelivery)
   646→      - "s"  suffix  → ""      (e.g. users → User, tenant_memberships → TenantMembership)
   647→      - otherwise   → unchanged (e.g. system_config → SystemConfig)
   648→
   649→    Then snake_case → PascalCase (split on ``_``, capitalise each segment).
   650→    """
   651→    name = collection
   652→    if name.endswith("ies"):
   653→        name = name[:-3] + "y"
   654→    elif name.endswith("s"):
   655→        name = name[:-1]
   656→    parts = [p for p in name.split("_") if p]
   657→    return "".join(p[:1].upper() + p[1:] for p in parts)

   660→def build_collection_struct_map(
   661→    accessor_map: dict[str, str],
   662→    struct_fields: dict[str, set[str]],
   663→) -> dict[str, str]:
   664→    """Build {collection_name: struct_name} using the snake→Pascal heuristic.
   665→
   666→    For each collection referenced by an accessor, derive the candidate
   667→    struct name and confirm it exists in ``struct_fields``. If the heuristic
   668→    doesn't match, fall back to checking known special cases (e.g.
   669→    ``counters`` → ``InvoiceCounter``).
   670→    """
   671→    # Hardcoded overrides for cases the naming heuristic can't resolve
   672→    # (collections whose struct name uses acronym capitalization like
   673→    # ``APIKey`` instead of ``ApiKey``).
   674→    overrides: dict[str, str] = {
   675→        "counters": "InvoiceCounter",
   676→        "api_keys": "APIKey",
   677→        "oauth_states": "OAuthState",
   678→        "sso_connections": "SSOConnection",
   679→        "webauthn_credentials": "WebAuthnCredential",
   680→    }
   681→    coll_to_struct: dict[str, str] = {}
   682→    collections = set(accessor_map.values()) | set(overrides.keys())
   683→    for coll in collections:
   684→        if coll in overrides and overrides[coll] in struct_fields:
   685→            coll_to_struct[coll] = overrides[coll]
   686→            continue
   687→        candidate = _collection_to_struct_name(coll)
   688→        if candidate in struct_fields:
   689→            coll_to_struct[coll] = candidate
   690→            continue
   691→        # Try without singularisation (some collections are already
   692→        # singular in the DB but use a plural struct name in Go).
   693→        if coll in struct_fields:
   694→            coll_to_struct[coll] = coll
   695→    return coll_to_struct

   735→# --------------------------------------------------------------------------- #
   736→# Route classification — parse main.go to map handler functions to route scope
   737→# --------------------------------------------------------------------------- #
   738→
   739→# Matches ``<router>.HandleFunc("/path", HANDLER)`` or ``<router>.Handle("/path", ...)``
   740→# capturing the router variable. The handler can be either a bare method value
   741→# (``handler.Foo``) or wrapped in ``rateLimiter.RateLimitHandler(...,
   742→# handler.Foo,)``.
   743→ROUTER_HANDLE_RE = re.compile(
   744→    r'\b(\w+)\.(?:HandleFunc|Handle)\s*\(\s*"[^"]*"'
   745→)
   746→
   747→# Matches a handler method value like ``authHandler.Login`` or
   748→# ``plansHandler.ListPlans`` (last identifier is the function name).
   749→HANDLER_METHOD_RE = re.compile(
   750→    r'\b(\w+Handler)\.([A-Z]\w*)\b'
   751→)
   752→
   753→# Matches ``http.HandlerFunc(handler.X)`` — the X is the actual handler.
   754→HTTPHANDLERFUNC_RE = re.compile(
   755→    r'\bhttp\.HandlerFunc\s*\(\s*(\w+Handler)\.([A-Z]\w*)\s*\)'
   756→)
   757→
   758→
   759→def build_route_classifier(repo_root: Path) -> dict[str, set[str]]:
   760→    """Parse ``cmd/server/main.go`` to extract handler-function → route-scope.
   761→
   762→    Returns a dict with three sets of handler function names:
   763→      - ``admin``   — registered on adminAPI/adminWrite/adminOwner (sees ALL tenants)
   764→      - ``tenant``  — registered on tenantAPI/usageAPI/telemetryAPI/billingAPI
   765→      - ``auth``    — registered on guarded/protectedAuth (auth flow, user-scoped)
   766→      - ``public``  — registered on api/router (no auth)
   767→
   768→    Falls back to an empty dict if main.go cannot be found.
   769→    """
   770→    result: dict[str, set[str]] = {
   771→        "admin": set(),
   772→        "tenant": set(),
   773→        "auth": set(),
   774→        "public": set(),
   775→    }
   776→    # Look for cmd/server/main.go under the repo root.
   777→    candidates = [
   778→        repo_root / "cmd" / "server" / "main.go",
   779→        repo_root / "backend" / "cmd" / "server" / "main.go",
   780→    ]
   781→    main_go: Optional[Path] = None
   782→    for c in candidates:
   783→        if c.is_file():
   784→            main_go = c
   785→            break
   786→    if main_go is None:
   787→        # Last resort: rglob.
   788→        for p in repo_root.rglob("main.go"):
   789→            if "cmd" in p.parts and "server" in p.parts:
   790→                main_go = p
   791→                break
   792→    if main_go is None or not main_go.is_file():
   793→        return result
   794→
   795→    try:
   796→        text = main_go.read_text(encoding="utf-8", errors="replace")
   797→    except OSError:
   798→        return result
   799→
   800→    masked = mask_strings_and_comments(text)
   801→    # Walk HandleFunc/Handle calls. We match the router.HandleFunc("/path", ...)
   802→    # pattern on the RAW text (so the string literal "/path" is visible), but
   803→    # use the MASKED text to find the closing paren of the call (so parens
   804→    # inside string literals don't confuse depth tracking).
   805→    for m in ROUTER_HANDLE_RE.finditer(text):
   806→        router = m.group(1)
   807→        # Find the corresponding index in the masked text (same offset since
   808→        # mask_strings_and_comments preserves length).
   809→        open_paren = masked.find("(", m.start())
   810→        if open_paren == -1:
   811→            continue
   812→        close_paren = find_block_end(masked, open_paren, "(", ")")
   813→        if close_paren == -1:
   814→            continue
   815→        call_span = text[m.start():close_paren]
   816→        # Find all handler.Foo references in the call span. The LAST one is
   817→        # the actual handler (rateLimiter.RateLimitHandler takes the handler
   818→        # as its last positional argument).
   819→        handlers = HANDLER_METHOD_RE.findall(call_span)
   820→        if not handlers:
   821→            # Try http.HandlerFunc(handler.X) form.
   822→            hfd = HTTPHANDLERFUNC_RE.findall(call_span)
   823→            handlers = hfd
   824→        if not handlers:
   825→            continue
   826→        # Take the LAST handler reference (in rate-limited form, the actual
   827→        # handler is the last positional argument).
   828→        _, func_name = handlers[-1]
   829→        if router in ADMIN_ROUTERS:
   830→            result["admin"].add(func_name)
   831→        elif router in TENANT_ROUTERS:
   832→            result["tenant"].add(func_name)
   833→        elif router in ("guarded", "protectedAuth", "api", "router"):
   834→            # Auth flow + public — the heuristic below refines further.
   835→            if func_name in PUBLIC_ENDPOINT_FUNCS:
   836→                result["public"].add(func_name)
   837→            else:
   838→                result["auth"].add(func_name)
   839→    return result
   840→

   842→def extract_filter_fields_from_text(text: str) -> list[str]:
   843→    """Pull field names out of bson.M / bson.D / map literals in the given
   844→    text. Returns a de-duplicated list (preserving first-seen order).
   845→    """
   846→    fields: list[str] = []
   847→
   848→    # bson.D blocks
   849→    for block in BSON_D_BLOCK_RE.finditer(text):
   850→        inner = block.group(1)
   851→        for m in BSON_D_KEYVAL_NAMED_RE.finditer(inner):
   852→            fields.append(m.group(1))
   853→        for m in BSON_D_KEYVAL_POS_RE.finditer(inner):
   854→            fields.append(m.group(1))
   855→
   856→    # bson.M blocks
   857→    for m in BSON_M_RE.finditer(text):
   858→        for km in BSON_M_KEY_RE.finditer(m.group(1)):
   859→            fields.append(km.group(1))
   860→
   861→    # map[string]interface{} blocks
   862→    for m in MAP_LITERAL_RE.finditer(text):
   863→        for km in MAP_KEY_RE.finditer(m.group(1)):
   864→            fields.append(km.group(1))
   865→
   866→    # Dedupe, drop operators and empty strings.
   867→    seen: set[str] = set()
   868→    out: list[str] = []
   869→    for f in fields:
   870→        if not f or f.startswith("$") or f in seen:
   871→            continue
   872→        seen.add(f)
   873→        out.append(f)
   874→    return out

   877→def find_block_end(text: str, start: int, open_ch: str, close_ch: str) -> int:
   878→    """Return the index of the matching closing bracket starting at ``start``
   879→    (which must point at ``open_ch``). Returns -1 if no match is found.
   880→    """
   881→    assert text[start] == open_ch
   882→    depth = 0
   883→    i = start
   884→    n = len(text)
   885→    while i < n:
   886→        c = text[i]
   887→        if c == open_ch:
   888→            depth += 1
   889→        elif c == close_ch:
   890→            depth -= 1
   891→            if depth == 0:
   892→                return i
   893→        i += 1
   894→    return -1

   897→def extract_call_args(
   898→    masked: str, raw: str, open_paren_idx: int
   899→) -> list[str]:
   900→    """Given the index of the opening ``(`` of a call in ``masked`` (and the
   901→    corresponding ``raw`` text of identical length), return the list of
   902→    top-level argument source strings (split on commas at depth 1).
   903→
   904→    Depth tracking uses ``masked`` (so commas/braces/parens inside string
   905→    literals don't confuse the splitter), but the returned argument text is
   906→    sliced from ``raw`` so that filter field extraction can see the actual
   907→    string keys (e.g. ``"tenantId"``).
   908→    """
   909→    if open_paren_idx >= len(masked) or masked[open_paren_idx] != "(":
   910→        return []
   911→    close_idx = find_block_end(masked, open_paren_idx, "(", ")")
   912→    if close_idx == -1:
   913→        return []
   914→    # The inner text spans [open_paren_idx+1, close_idx) in both masked & raw.
   915→    args: list[str] = []
   916→    depth_paren = 0
   917→    depth_brace = 0
   918→    depth_brack = 0
   919→    buf_start = open_paren_idx + 1
   920→    i = buf_start
   921→    n = close_idx
   922→    while i < n:
   923→        c = masked[i]
   924→        if c == "(":
   925→            depth_paren += 1
   926→        elif c == ")":
   927→            depth_paren -= 1
   928→        elif c == "{":
   929→            depth_brace += 1
   930→        elif c == "}":
   931→            depth_brace -= 1
   932→        elif c == "[":
   933→            depth_brack += 1
   934→        elif c == "]":
   935→            depth_brack -= 1
   936→        elif c == "," and depth_paren == 0 and depth_brace == 0 and depth_brack == 0:
   937→            args.append(raw[buf_start:i].strip())
   938→            buf_start = i + 1
   939→        i += 1
   940→    tail = raw[buf_start:close_idx].strip()
   941→    if tail:
   942→        args.append(tail)
   943→    return args

   951→def _collect_brace_window(
   952→    masked_lines: list[str],
   953→    raw_lines: list[str],
   954→    start_idx: int,
   955→    start_col: int,
   956→    max_lines: int = 80,
   957→) -> str:
   958→    """Build a multi-line text window starting at
   959→    ``raw_lines[start_idx][start_col:]`` and extending until the braces
   960→    opened on the first line are balanced.
   961→
   962→    Depth tracking uses ``masked_lines`` (so braces/parens inside string
   963→    literals don't prematurely close the window), but the returned text is
   964→    sliced from ``raw_lines`` so that filter field extraction can see the
   965→    actual string keys (e.g. ``"tenantId"``).
   966→
   967→    The two line arrays must have identical length and newline positions
   968→    (which is guaranteed when both are derived from the same source via
   969→    :func:`mask_strings_and_comments`).
   970→    """
   971→    n = len(masked_lines)
   972→    if start_idx >= n:
   973→        return ""
   974→    first_masked = masked_lines[start_idx][start_col:]
   975→    first_raw = raw_lines[start_idx][start_col:]
   976→    window_lines = [first_raw]
   977→    depth = 0
   978→    for ch in first_masked:
   979→        if ch == "{":
   980→            depth += 1
   981→        elif ch == "}":
   982→            depth -= 1
   983→    i = start_idx + 1
   984→    while depth > 0 and i < n and (i - start_idx) < max_lines:
   985→        nxt_masked = masked_lines[i]
   986→        window_lines.append(raw_lines[i])
   987→        for ch in nxt_masked:
   988→            if ch == "{":
   989→                depth += 1
   990→            elif ch == "}":
   991→                depth -= 1
   992→        i += 1
   993→    return "\n".join(window_lines)

   996→def scan_file(
   997→    path: Path,
   998→    repo_root: Path,
   999→    accessor_map: dict[str, str],
  1000→    struct_fields: dict[str, set[str]],
  1001→    queries: list[Query],
  1002→    file_index: dict[str, FileInfo],
  1003→    route_classifier: Optional[dict[str, set[str]]] = None,
  1004→    collection_struct_map: Optional[dict[str, str]] = None,
  1005→    user_scoped_collections: Optional[set[str]] = None,
  1006→    global_collections: Optional[set[str]] = None,
  1007→) -> None:
  1008→    """Scan a single Go file for MongoDB queries and classify each."""
  1009→    try:
  1010→        raw = path.read_text(encoding="utf-8", errors="replace")
  1011→    except OSError:
  1012→        return
  1013→    masked = mask_strings_and_comments(raw)
  1014→    lines = masked.splitlines()
  1015→    raw_lines = raw.splitlines()
  1016→
  1017→    rel = _relative(repo_root, path)
  1018→    finfo = file_index.setdefault(rel, FileInfo(path=rel))
  1019→
  1020→    # Pre-compute per-file context flags.
  1021→    is_cli_file = CLI_PATH_PARTS[0] in path.parts and CLI_PATH_PARTS[1] in path.parts
  1022→    is_testutil = "testutil" in path.parts or path.name == "testutil.go"
  1023→    is_test_file = path.name.endswith("_test.go") or is_testutil
  1024→    route_classifier = route_classifier or {}
  1025→    collection_struct_map = collection_struct_map or {}
  1026→    user_scoped_collections = user_scoped_collections or set()
  1027→    global_collections = global_collections or set()
  1028→
  1029→    # Per-file alias + filter-variable tracking.
  1030→    #   aliases[var_name]      = collection_name           (collection-typed vars)
  1031→    #   filter_vars[var_name]  = {fields, def_line, ...}   (bson.M/bson.D/map vars)
  1032→    #   pipeline_vars[var_name]= {fields, def_line, ...}   ([]bson.M / mongo.Pipeline)
  1033→    #   struct_vars[var_name]  = struct_name               (models.Invitation, etc.)
  1034→    aliases: dict[str, str] = {}
  1035→    filter_vars: dict[str, dict] = {}
  1036→    pipeline_vars: dict[str, dict] = {}
  1037→    struct_vars: dict[str, str] = {}
  1038→
  1039→    current_func = "<package-init>"
  1040→    n = len(lines)
  1041→    masked_full = masked
  1042→    raw_full = raw
  1043→
  1044→    # Single pass: walk line-by-line, tracking function context and variable
  1045→    # assignments. When a MongoDB operation call is found on the current
  1046→    # line, extract its arguments from the full masked/raw text (so multi-
  1047→    # line calls are handled) and classify the query.
  1048→    #
  1049→    # NOTE: function/alias/var-def tracking uses the MASKED line (so braces
  1050→    # inside strings don't confuse depth tracking), but dynamic field-addition
  1051→    # detection (``filter["tenantId"] = ...``) uses the RAW line because the
  1052→    # string key must be visible to the regex.
  1053→    for idx in range(n):
  1054→        line = lines[idx]
  1055→        raw_line = raw_lines[idx]
  1056→        line_no = idx + 1
  1057→        stripped = line.lstrip()
  1058→
  1059→        # ---- Track current function (and reset per-function state) ----
  1060→        if stripped.startswith("func"):
  1061→            fm = FUNC_RE.match(stripped)
  1062→            if fm:
  1063→                current_func = fm.group(1)
  1064→                filter_vars.clear()
  1065→                pipeline_vars.clear()
  1066→                aliases.clear()
  1067→                struct_vars.clear()
  1068→                continue
  1069→
  1070→        # ---- Track collection-alias assignments ----
  1071→        for m in ALIAS_LITERAL_RE.finditer(line):
  1072→            aliases[m.group(1)] = m.group(2)
  1073→        for m in ALIAS_ACCESSOR_RE.finditer(line):
  1074→            var_name = m.group(1)
  1075→            accessor = m.group(2)
  1076→            end = m.end()
  1077→            if end < len(line) and line[end] == ".":
  1078→                continue  # chained call, not a bare alias
  1079→            if accessor in accessor_map:
  1080→                aliases[var_name] = accessor_map[accessor]
  1081→
  1082→        # ---- Track filter-variable definitions ----
  1083→        # ``filter := bson.M{...}`` — capture the multi-line literal that
  1084→        # follows, starting at the variable name so the ``bson.M`` prefix is
  1085→        # preserved (the field-extraction regexes require it).
  1086→        for m in FILTER_VAR_DEF_RE.finditer(line):
  1087→            var_name = m.group(1)
  1088→            window = _collect_brace_window(lines, raw_lines, idx, m.start(), max_lines=30)
  1089→            fields = extract_filter_fields_from_text(window)
  1090→            filter_vars[var_name] = {
  1091→                "fields": set(fields),
  1092→                "def_line": line_no,
  1093→                "source": "inline-literal",
  1094→            }
  1095→
  1096→        # ---- Track filter-variable dynamic field additions ----
  1097→        # ``filter["tenantId"] = ...`` — uses the RAW line so the string key
  1098→        # is visible (the masked line has the key's characters replaced with
  1099→        # spaces).
  1100→        for m in FILTER_VAR_SET_RE.finditer(raw_line):
  1101→            var_name = m.group(1)
  1102→            field_name = m.group(2)
  1103→            if var_name in filter_vars:
  1104→                filter_vars[var_name]["fields"].add(field_name)
  1105→            else:
  1106→                # Variable defined elsewhere (function param, loop var, etc.)
  1107→                filter_vars[var_name] = {
  1108→                    "fields": {field_name},
  1109→                    "def_line": line_no,
  1110→                    "source": "dynamic-only",
  1111→                }
  1112→
  1113→        # ---- Track pipeline-variable definitions (for Aggregate) ----
  1114→        for m in PIPELINE_VAR_DEF_RE.finditer(line):
  1115→            var_name = m.group(1)
  1116→            window = _collect_brace_window(lines, raw_lines, idx, m.start(), max_lines=80)
  1117→            fields = _extract_pipeline_match_fields(window)
  1118→            pipeline_vars[var_name] = {
  1119→                "fields": set(fields),
  1120→                "def_line": line_no,
  1121→                "source": "inline-literal",
  1122→                "window": window,
  1123→            }
  1124→
  1125→        # ---- Track struct-literal variable assignments ----
  1126→        # ``invitation := models.Invitation{...}`` — records the struct name
  1127→        # so InsertOne(ctx, invitation) can be checked for tenantId presence
  1128→        # via the struct's bson field tags. Uses the RAW line so that the
  1129→        # struct name (which is a Go identifier, not a string) is visible.
  1130→        for m in STRUCT_VAR_DEF_RE.finditer(raw_line):
  1131→            var_name = m.group(1)
  1132→            struct_name = m.group(3)
  1133→            # Only track if the struct is known (has bson field tags). This
  1134→            # filters out generic-looking assignments like ``x := Foo{}``
  1135→            # where Foo isn't a tracked struct.
  1136→            if struct_name in struct_fields:
  1137→                struct_vars[var_name] = struct_name
  1138→
  1139→        # ---- Find operation calls on this line ----
  1140→        line_offset = sum(len(lines[k]) + 1 for k in range(idx))  # +1 for \n
  1141→        for op_m in OP_RE.finditer(line):
  1142→            operation = op_m.group(1)
  1143→            # OP_RE matches ``.Find(`` etc. — op_m.end() is just past the "(".
  1144→            open_paren_idx_in_line = op_m.end() - 1
  1145→            open_paren_idx_full = line_offset + open_paren_idx_in_line
  1146→
  1147→            # Skip option-builder calls (``options.Find()`` etc.). Uses the
  1148→            # RAW prefix so the literal ``options`` token is reliably visible.
  1149→            raw_prefix = raw_line[: op_m.start()]
  1150→            if OPTIONS_BUILDER_RE.search(raw_prefix):
  1151→                continue
  1152→
  1153→            # ---- Resolve the collection ----
  1154→            # The prefix search uses the RAW line so that collection literals
  1155→            # like ``db.Collection("users")`` (which contain a string) are
  1156→            # visible. Depth tracking isn't needed here — we're just looking
  1157→            # for the last accessor call or collection literal before ``.Op(``.
  1158→            collection: Optional[str] = None
  1159→            accessor_used: Optional[str] = None
  1160→
  1161→            lit_m = COLLECTION_LITERAL_RE.search(raw_prefix)
  1162→            if lit_m:
  1163→                collection = lit_m.group(1)
  1164→            else:
  1165→                acc_m = re.search(r"([A-Z]\w*)\(\s*\)\s*$", raw_prefix)
  1166→                if acc_m:
  1167→                    accessor_used = acc_m.group(1)
  1168→                    collection = accessor_map.get(accessor_used)
  1169→                else:
  1170→                    var_m = re.search(r"([A-Za-z_]\w*)\s*$", raw_prefix)
  1171→                    if var_m:
  1172→                        collection = aliases.get(var_m.group(1))
  1173→
  1174→            # Fallback: look back a couple of lines for a collection literal
  1175→            # or accessor call (handles calls split across lines). Uses RAW
  1176→            # lines for the same reason as above.
  1177→            if collection is None:
  1178→                for back in range(idx - 1, max(idx - 4, -1), -1):
  1179→                    if back < 0 or back >= n:
  1180→                        continue
  1181→                    prev_raw = raw_lines[back]
  1182→                    lit_m2 = COLLECTION_LITERAL_RE.search(prev_raw)
  1183→                    if lit_m2:
  1184→                        collection = lit_m2.group(1)
  1185→                        break
  1186→                    acc_m2 = re.search(r"([A-Z]\w*)\(\s*\)", prev_raw)
  1187→                    if acc_m2 and acc_m2.group(1) in accessor_map:
  1188→                        collection = accessor_map[acc_m2.group(1)]
  1189→                        if accessor_used is None:
  1190→                            accessor_used = acc_m2.group(1)
  1191→                        break
  1192→
  1193→            if collection is None:
  1194→                # Last resort: if the accessor name is in the exempt set,
  1195→                # use it as the collection name (so we can still classify).
  1196→                if accessor_used and accessor_used in EXEMPT_ACCESSORS:
  1197→                    collection = _accessor_to_collection(accessor_used)
  1198→                else:
  1199→                    collection = "<unknown>"
  1200→
  1201→            # ---- Extract the call arguments ----
  1202→            args = extract_call_args(masked_full, raw_full, open_paren_idx_full)
  1203→
  1204→            # Determine which argument holds the filter/document/pipeline.
  1205→            filter_text = ""
  1206→            filter_source = "unknown"
  1207→            filter_fields: list[str] = []
  1208→            has_tenant_id = False
  1209→
  1210→            # MongoDB driver convention: first arg is always ctx (or t in
  1211→            # tests). The filter/document/pipeline is the second arg.
  1212→            if operation == "EstimatedDocumentCount":
  1213→                # No filter argument.
  1214→                filter_source = "no-filter"
  1215→            elif operation == "BulkWrite":
  1216→                # BulkWrite(ctx, []mongo.WriteModel{...}) — the models slice
  1217→                # is too complex to analyse statically.
  1218→                filter_source = "bulk-write-models"
  1219→                if len(args) >= 2:
  1220→                    filter_text = args[1]
  1221→            elif operation == "InsertMany":
  1222→                # InsertMany(ctx, []interface{}{...}) — the slice contains
  1223→                # the documents to insert.
  1224→                if len(args) >= 2:
  1225→                    filter_text = args[1]
  1226→                    filter_fields = extract_filter_fields_from_text(filter_text)
  1227→                    filter_source = "inline-document-slice"
  1228→                else:
  1229→                    filter_source = "missing-arg"
  1230→            elif operation == "InsertOne":
  1231→                # InsertOne(ctx, document) — document may be a bson.M literal,
  1232→                # an inline struct literal (``models.Foo{...}``), or a
  1233→                # variable holding a struct literal (``foo`` assigned from
  1234→                # ``models.Foo{...}`` earlier in the function).
  1235→                if len(args) >= 2:
  1236→                    filter_text = args[1]
  1237→                    if "bson.M" in filter_text or "bson.D" in filter_text or "map[string]" in filter_text:
  1238→                        filter_fields = extract_filter_fields_from_text(filter_text)
  1239→                        filter_source = "inline-document"
  1240→                    else:
  1241→                        # Try inline struct literal first.
  1242→                        struct_name = _extract_struct_name(filter_text)
  1243→                        # Then fall back to a struct-typed variable alias.
  1244→                        if struct_name is None or struct_name not in struct_fields:
  1245→                            var_m = re.match(r"^&?(\w+)\s*$", filter_text.strip())
  1246→                            if var_m and var_m.group(1) in struct_vars:
  1247→                                struct_name = struct_vars[var_m.group(1)]
  1248→                                filter_source = f"struct-var:{var_m.group(1)}:{struct_name}"
  1249→                        if struct_name and struct_name in struct_fields:
  1250→                            if not filter_source.startswith("struct-var:"):
  1251→                                filter_source = f"struct:{struct_name}"
  1252→                            bson_fields = struct_fields[struct_name]
  1253→                            # Don't pollute filter_fields with every struct
  1254→                            # field — just record the ones relevant for
  1255→                            # tenant filtering. We expose the full field set
  1256→                            # for context, but only treat tenantId presence
  1257→                            # as "has_tenant_id" if the struct HAS a
  1258→                            # tenantId bson field.
  1259→                            filter_fields = sorted(
  1260→                                f for f in bson_fields if not f.startswith("_")
  1261→                            )
  1262→                            if TENANT_ID_FIELDS & bson_fields:
  1263→                                has_tenant_id = True
  1264→                        else:
  1265→                            filter_source = "struct:unknown"
  1266→                else:
  1267→                    filter_source = "missing-arg"
  1268→            elif operation == "UpdateByID":
  1269→                # UpdateByID(ctx, id, update) — id is the document _id (unique).
  1270→                filter_source = "by-id"
  1271→                if len(args) >= 2:
  1272→                    filter_text = args[1]
  1273→                    # Mark as having a safe unique key.
  1274→                    filter_fields = ["_id"]
  1275→            elif operation == "Aggregate":
  1276→                # Aggregate(ctx, pipeline) — pipeline is a slice of stages.
  1277→                if len(args) >= 2:
  1278→                    filter_text = args[1]
  1279→                    if (
  1280→                        filter_text.startswith("bson.M")
  1281→                        or filter_text.startswith("bson.D")
  1282→                        or filter_text.startswith("bson.A")
  1283→                        or filter_text.startswith("[]bson")
  1284→                        or filter_text.startswith("mongo.Pipeline")
  1285→                    ):
  1286→                        # Inline pipeline literal.
  1287→                        filter_fields = _extract_pipeline_match_fields(filter_text)
  1288→                        filter_source = "inline-pipeline"
  1289→                    else:
  1290→                        # Variable pipeline — look up in pipeline_vars.
  1291→                        var_m = re.match(r"^(\w+)\s*$", filter_text.strip())
  1292→                        if var_m and var_m.group(1) in pipeline_vars:
  1293→                            info = pipeline_vars[var_m.group(1)]
  1294→                            filter_fields = sorted(info["fields"])
  1295→                            filter_source = f"variable-pipeline:{var_m.group(1)}"
  1296→                        else:
  1297→                            filter_source = "variable-pipeline:unknown"
  1298→                else:
  1299→                    filter_source = "missing-arg"
  1300→            elif operation in FILTER_AT_POS_2:
  1301→                # Standard filter-bearing call: <Op>(ctx, filter, ...).
  1302→                if len(args) >= 2:
  1303→                    filter_text = args[1]
  1304→                    if filter_text == "nil":
  1305→                        filter_source = "nil"
  1306→                        filter_fields = []
  1307→                    elif filter_text.startswith("bson.M") or filter_text.startswith("bson.D") or filter_text.startswith("map[string]"):
  1308→                        filter_fields = extract_filter_fields_from_text(filter_text)
  1309→                        filter_source = "inline"
  1310→                    else:
  1311→                        # Variable filter — look up in filter_vars.
  1312→                        var_m = re.match(r"^(\w+)\s*$", filter_text.strip())
  1313→                        if var_m and var_m.group(1) in filter_vars:
  1314→                            info = filter_vars[var_m.group(1)]
  1315→                            filter_fields = sorted(info["fields"])
  1316→                            filter_source = f"variable:{var_m.group(1)}"
  1317→                        else:
  1318→                            filter_source = f"variable:unknown:{filter_text.strip()[:40]}"
  1319→                else:
  1320→                    filter_source = "missing-arg"
  1321→            else:
  1322→                # Should not happen — every op is handled above.
  1323→                filter_source = "unhandled"
  1324→
  1325→            # ---- Determine has_tenant_id ----
  1326→            if filter_fields:
  1327→                field_set = set(filter_fields)
  1328→                if TENANT_ID_FIELDS & field_set:
  1329→                    has_tenant_id = True
  1330→
  1331→            # ---- Classify ----
  1332→            is_exempt = _is_exempt_collection(collection, accessor_used)
  1333→            safe_key = _has_safe_unique_key(filter_fields, operation)
  1334→            has_user_id = "userId" in filter_fields or "user_id" in filter_fields
  1335→
  1336→            # ---- Compute context heuristics (false-positive suppression) ----
  1337→            # These are checked in priority order; the first match wins and
  1338→            # downgrades the query from a violation to MEDIUM informational.
  1339→            # Auth flow handlers are user-scoped (operate on refresh_tokens,
  1340→            # verification_tokens, etc.) — not admin handlers.
  1341→            is_auth_flow = (
  1342→                current_func in route_classifier.get("auth", set())
  1343→                or current_func in AUTH_FLOW_FUNCS
  1344→            )
  1345→            is_admin_handler = (
  1346→                current_func in route_classifier.get("admin", set())
  1347→                or current_func in ADMIN_SERVICE_FUNCS
  1348→            )
  1349→            is_public_endpoint = (
  1350→                current_func in route_classifier.get("public", set())
  1351→                or current_func in PUBLIC_ENDPOINT_FUNCS
  1352→            )
  1353→            is_background_task = current_func in BACKGROUND_TASK_FUNCS
  1354→            # Resolve the underlying struct for this collection.
  1355→            struct_name = collection_struct_map.get(collection)
  1356→            coll_has_tid: Optional[bool] = None
  1357→            if struct_name and struct_name in struct_fields:
  1358→                coll_has_tid = bool(
  1359→                    TENANT_ID_FIELDS & struct_fields[struct_name]
  1360→                )
  1361→            # ``collection_is_user_scoped`` is True when the struct has a
  1362→            # ``userId`` bson field — for these, a ``userId`` filter is a
  1363→            # valid scope (the user's data spans tenants).
  1364→            coll_is_user_scoped = collection in user_scoped_collections
  1365→            # ``collection_is_global`` is True when the struct does NOT
  1366→            # declare a tenantId field — these collections are global by
  1367→            # design (e.g. ``webhooks``, ``plans``, ``system_nodes``).
  1368→            collection_is_global = (
  1369→                collection in global_collections
  1370→                or coll_has_tid is False
  1371→            )
  1372→            # Decide the route scope label (for the report).
  1373→            if is_testutil:
  1374→                route_scope = "testutil"
  1375→            elif is_cli_file:
  1376→                route_scope = "cli"
  1377→            elif is_background_task:
  1378→                route_scope = "background"
  1379→            elif is_public_endpoint:
  1380→                route_scope = "public"
  1381→            elif is_admin_handler:
  1382→                route_scope = "admin"
  1383→            elif is_auth_flow:
  1384→                route_scope = "auth"
  1385→            elif current_func in route_classifier.get("tenant", set()):
  1386→                route_scope = "tenant"
  1387→            else:
  1388→                route_scope = "unknown"
  1389→
  1390→            suppression_reason: str = ""
  1391→
  1392→            # ---- Apply false-positive suppressions in priority order ----
  1393→            if has_tenant_id:
  1394→                risk_level = "OK"
  1395→                is_violation = False
  1396→                note = "Filter contains tenantId."
  1397→            elif is_testutil:
  1398→                risk_level = "MEDIUM"
  1399→                is_violation = False
  1400→                suppression_reason = "test-util"
  1401→                note = (
  1402→                    "Query is in a test-utility file (internal/testutil/) — "
  1403→                    "test helpers reset/clean test databases between test "
  1404→                    "runs. Not production code."
  1405→                )
  1406→            elif is_cli_file:
  1407→                risk_level = "MEDIUM"
  1408→                is_violation = False
  1409→                suppression_reason = "cli-tool"
  1410→                note = (
  1411→                    "Query is in a CLI tool (cmd/lastsaas/) — CLI tools have "
  1412→                    "no HTTP request context and operate on system-wide "
  1413→                    "collections by design."
  1414→                )
  1415→            elif is_background_task:
  1416→                risk_level = "MEDIUM"
  1417→                is_violation = False
  1418→                suppression_reason = "background-task"
  1419→                note = (
  1420→                    f"Function '{current_func}' is a background system task "
  1421→                    "(no request context, no tenantId available). Operates "
  1422→                    "on system-wide collections by design."
  1423→                )
  1424→            elif is_admin_handler:
  1425→                risk_level = "MEDIUM"
  1426→                is_violation = False
  1427→                suppression_reason = "admin-handler"
  1428→                note = (
  1429→                    f"Function '{current_func}' is an admin handler "
  1430→                    "(registered on /api/admin/*). Admin handlers see ALL "
  1431→                    "tenants by design — no tenantId filter needed."
  1432→                )
  1433→            elif is_public_endpoint:
  1434→                risk_level = "MEDIUM"
  1435→                is_violation = False
  1436→                suppression_reason = "public-endpoint"
  1437→                note = (
  1438→                    f"Function '{current_func}' is a public endpoint (no "
  1439→                    "auth) — returns published/global content to everyone."
  1440→                )
  1441→            elif collection_is_global:
  1442→                risk_level = "MEDIUM"
  1443→                is_violation = False
  1444→                suppression_reason = "global-by-design"
  1445→                if struct_name:
  1446→                    note = (
  1447→                        f"Collection '{collection}' maps to struct "
  1448→                        f"'{struct_name}' which does NOT declare a tenantId "
  1449→                        f"bson field — this collection is global by design."
  1450→                    )
  1451→                else:
  1452→                    note = (
  1453→                        f"Collection '{collection}' is in the global list "
  1454→                        f"(no tenantId field) — global by design."
  1455→                    )
  1456→            elif coll_is_user_scoped and has_user_id:
  1457→                risk_level = "OK"
  1458→                is_violation = False
  1459→                suppression_reason = "user-scoped"
  1460→                note = (
  1461→                    f"Collection '{collection}' is user-scoped (has userId "
  1462→                    f"field) and the filter contains userId — a user's data "
  1463→                    f"spans tenants, so userId is a valid scope."
  1464→                )
  1465→            elif is_auth_flow and (has_user_id or safe_key):
  1466→                risk_level = "MEDIUM"
  1467→                is_violation = False
  1468→                suppression_reason = "auth-flow"
  1469→                note = (
  1470→                    f"Auth-flow handler '{current_func}' operates on user-"
  1471→                    f"scoped collections (the user hasn't selected a tenant "
  1472→                    f"yet during auth). Filter on userId/token/tokenHash is "
  1473→                    f"a valid scope."
  1474→                )
  1475→            elif safe_key:
  1476→                risk_level = "MEDIUM"
  1477→                is_violation = False
  1478→                suppression_reason = "safe-unique-key"
  1479→                note = (
  1480→                    f"Filter contains a globally-unique key "
  1481→                    f"({', '.join(f for f in filter_fields if f in SAFE_UNIQUE_KEYS)}) "
  1482→                    f"— the query will only ever return one document "
  1483→                    f"regardless of tenant."
  1484→                )
  1485→            elif is_exempt:
  1486→                risk_level = "MEDIUM"
  1487→                is_violation = False
  1488→                suppression_reason = "exempt-collection"
  1489→                note = (
  1490→                    f"Collection '{collection}' is in the exempt (global) "
  1491→                    f"list; tenant filtering is not strictly required, but "
  1492→                    f"queries should still be reviewed for appropriate "
  1493→                    f"scoping."
  1494→                )
  1495→            elif operation in ("InsertOne", "InsertMany") and (
  1496→                filter_source in ("inline-document", "inline-document-slice")
  1497→                or filter_source == "missing-arg"
  1498→            ) and coll_has_tid is True:
  1499→                # The inserted struct (or slice of structs) declares a
  1500→                # ``tenantId`` bson field. The auditor can't statically
  1501→                # verify the value is set at runtime, but the struct
  1502→                # SUPPORTS tenant scoping — caller is responsible. Common
  1503→                # case: a tenant-scoped HTTP handler populates the struct
  1504→                # from request context before calling the service method.
  1505→                risk_level = "MEDIUM"
  1506→                is_violation = False
  1507→                suppression_reason = "struct-supports-tenant-id"
  1508→                note = (
  1509→                    f"{operation} inserts a document (or slice of documents) "
  1510→                    f"whose struct '{struct_name}' declares a tenantId bson "
  1511→                    f"field. The value cannot be verified statically — caller "
  1512→                    f"is responsible for setting it (typically the tenant-"
  1513→                    f"scoped HTTP handler reads tenantId from request context "
  1514→                    f"and populates the struct before calling this method)."
  1515→                )
  1516→            elif operation in WRITE_OPS:
  1517→                risk_level = "CRITICAL"
  1518→                is_violation = True
  1519→                if filter_source == "by-id":
  1520→                    note = (
  1521→                        "UpdateByID uses the document _id (globally unique) — "
  1522→                        "tenant filtering is implicit. Likely safe."
  1523→                    )
  1524→                elif filter_source.startswith("struct:"):
  1525→                    struct_name_src = filter_source.split(":", 1)[1]
  1526→                    if struct_name_src == "unknown":
  1527→                        note = (
  1528→                            "InsertOne with an unrecognised struct value; "
  1529→                            "could not statically verify tenantId presence. "
  1530→                            "Manual review required."
  1531→                        )
  1532→                    else:
  1533→                        note = (
  1534→                            f"InsertOne with a {struct_name_src} struct "
  1535→                            f"value; tenantId presence inferred from the "
  1536→                            f"struct definition — verify the value is "
  1537→                            f"actually set at runtime."
  1538→                        )
  1539→                elif filter_source.startswith("struct-var:"):
  1540→                    parts = filter_source.split(":")
  1541→                    var_name = parts[1] if len(parts) > 1 else "?"
  1542→                    struct_name_src = parts[2] if len(parts) > 2 else "?"
  1543→                    note = (

  1667→def _extract_pipeline_match_fields(text: str) -> list[str]:
  1668→    """Extract field names from ``$match`` stages in an aggregate pipeline.
  1669→
  1670→    A ``$match`` stage looks like ``{"$match": bson.M{...}}`` or
  1671→    ``bson.D{{Key: "$match", Value: bson.D{...}}}``. We locate the inner
  1672→    filter literal and delegate to :func:`extract_filter_fields_from_text`,
  1673→    passing enough surrounding context (the ``bson.M`` / ``bson.D`` prefix)
  1674→    that the BSON regexes can match.
  1675→    """
  1676→    fields: list[str] = []
  1677→    match_re = re.compile(
  1678→        r'"\$match"\s*:\s*(bson\.M|bson\.D)\s*\{', re.IGNORECASE
  1679→    )
  1680→    for m in match_re.finditer(text):
  1681→        # Find the closing brace of the ``bson.M{...}`` block.
  1682→        brace_idx = m.end() - 1
  1683→        end = find_block_end(text, brace_idx, "{", "}")
  1684→        if end == -1:
  1685→            continue
  1686→        # Include the ``bson.M`` prefix in the slice so that
  1687→        # ``BSON_M_RE`` / ``BSON_D_RE`` (which require the prefix) match.
  1688→        bson_prefix_start = m.start(1)
  1689→        inner = text[bson_prefix_start : end + 1]
  1690→        fields.extend(extract_filter_fields_from_text(inner))
  1691→    # Dedupe.
  1692→    seen: set[str] = set()
  1693→    out: list[str] = []
  1694→    for f in fields:
  1695→        if f and not f.startswith("$") and f not in seen:
  1696→            seen.add(f)
  1697→            out.append(f)
  1698→    return out

  1753→def _build_snippet(raw_lines: list[str], line_no: int, filter_text: str) -> str:
  1754→    """Build a compact human-readable snippet for the report.
  1755→
  1756→    Uses the raw (unmasked) source lines so the snippet shows the actual code
  1757→    the developer wrote (with strings/comments intact).
  1758→    """
  1759→    if 1 <= line_no <= len(raw_lines):
  1760→        first = raw_lines[line_no - 1].strip()
  1761→    else:
  1762→        first = ""
  1763→    # Include up to 3 following lines if the call clearly continues.
  1764→    snippet_lines = [first]
  1765→    depth_paren = first.count("(") - first.count(")")
  1766→    depth_brace = first.count("{") - first.count("}")
  1767→    i = line_no + 1
  1768→    while (depth_paren > 0 or depth_brace > 0) and i <= len(raw_lines) and (i - line_no) < 5:
  1769→        nxt = raw_lines[i - 1].rstrip()
  1770→        snippet_lines.append(nxt)
  1771→        depth_paren += nxt.count("(") - nxt.count(")")
  1772→        depth_brace += nxt.count("{") - nxt.count("}")
  1773→        i += 1
  1774→    snippet = "\n".join(snippet_lines).strip()
  1775→    if len(snippet) > 400:
  1776→        snippet = snippet[:397] + "..."
