    20→// ---------------------------------------------------------------------------
    21→// HTTP client for proxying read-only requests to the LastSaaS API
    22→// ---------------------------------------------------------------------------
    23→
    24→type mcpClient struct {
    25→        baseURL    string
    26→        apiKey     string
    27→        httpClient *http.Client
    28→}
    29→
    30→func newMCPClient(baseURL, apiKey string) *mcpClient {
    31→        return &mcpClient{
    32→                baseURL: baseURL,
    33→                apiKey:  apiKey,
    34→                httpClient: &http.Client{
    35→                        Timeout: 30 * time.Second,
    36→                },
    37→        }
    38→}
    39→
    40→func (c *mcpClient) get(ctx context.Context, path string) ([]byte, error) {
    41→        req, err := http.NewRequestWithContext(ctx, http.MethodGet, c.baseURL+path, nil)
    42→        if err != nil {
    43→                return nil, fmt.Errorf("failed to create request: %w", err)
    44→        }
    45→        req.Header.Set("Authorization", "Bearer "+c.apiKey)
    46→
    47→        resp, err := c.httpClient.Do(req)
    48→        if err != nil {
    49→                return nil, fmt.Errorf("request failed: %w", err)
    50→        }
    51→        defer resp.Body.Close()
    52→
    53→        body, err := io.ReadAll(resp.Body)
    54→        if err != nil {
    55→                return nil, fmt.Errorf("failed to read response: %w", err)
    56→        }
    57→
    58→        if resp.StatusCode < 200 || resp.StatusCode >= 300 {
    59→                return nil, fmt.Errorf("API error (HTTP %d): %s", resp.StatusCode, string(body))
    60→        }
    61→        return body, nil
    62→}

    68→func prettyJSON(data []byte) string {
    69→        var buf bytes.Buffer
    70→        if err := json.Indent(&buf, data, "", "  "); err != nil {
    71→                return string(data)
    72→        }
    73→        return buf.String()
    74→}
    75→
    76→func buildQuery(params map[string]string) string {
    77→        v := url.Values{}
    78→        for key, val := range params {
    79→                if val != "" {
    80→                        v.Set(key, val)
    81→                }
    82→        }
    83→        if len(v) == 0 {
    84→                return ""
    85→        }
    86→        return "?" + v.Encode()
    87→}

    89→// ---------------------------------------------------------------------------
    90→// Entry point
    91→// ---------------------------------------------------------------------------
    92→
    93→func cmdMCP() {
    94→        baseURL := os.Getenv("LASTSAAS_URL")
    95→        apiKey := os.Getenv("LASTSAAS_API_KEY")
    96→
    97→        if baseURL == "" {
    98→                fmt.Fprintln(os.Stderr, "LASTSAAS_URL environment variable is required (e.g. http://localhost:3000)")
    99→                os.Exit(1)
   100→        }
   101→        if apiKey == "" {
   102→                fmt.Fprintln(os.Stderr, "LASTSAAS_API_KEY environment variable is required (e.g. lsk_xxxxx)")
   103→                os.Exit(1)
   104→        }
   105→
   106→        client := newMCPClient(baseURL, apiKey)
   107→
   108→        s := server.NewMCPServer(
   109→                "lastsaas-admin",
   110→                version.Current,
   111→                server.WithToolCapabilities(false),
   112→                server.WithResourceCapabilities(false, false),
   113→                server.WithRecovery(),
   114→        )
   115→
   116→        registerAboutTools(s, client)
   117→        registerDashboardTools(s, client)
   118→        registerTenantTools(s, client)
   119→        registerUserTools(s, client)
   120→        registerFinancialTools(s, client)
   121→        registerLogTools(s, client)
   122→        registerHealthTools(s, client)
   123→        registerConfigTools(s, client)
   124→        registerPlanTools(s, client)
   125→        registerAnnouncementTools(s, client)
   126→        registerPromotionTools(s, client)
   127→        registerSecurityTools(s, client)
   128→        registerWebhookTools(s, client)
   129→        registerPMTools(s, client)
   130→        registerResources(s, client)
   131→
   132→        if err := server.ServeStdio(s); err != nil {
   133→                fmt.Fprintf(os.Stderr, "MCP server error: %v\n", err)
   134→                os.Exit(1)
   135→        }
   136→}
