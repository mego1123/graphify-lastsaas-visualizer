    39→// spaHandler serves a single-page application from a static directory.
    40→// For files that exist on disk, it serves them directly. For all other
    41→// paths it serves index.html so the SPA router can handle them.
    42→// When serving index.html, it replaces {{APP_NAME}} with the actual app name
    43→// to prevent a title flicker while JavaScript loads branding data.
    44→type spaHandler struct {
    45→        staticPath string
    46→        indexPath  string
    47→        getAppName func() string
    48→}
    49→
    50→func (h spaHandler) ServeHTTP(w http.ResponseWriter, r *http.Request) {
    51→        // Never serve the SPA for /api/ paths — those should be handled by API routes
    52→        if strings.HasPrefix(r.URL.Path, "/api/") || r.URL.Path == "/api" {
    53→                http.NotFound(w, r)
    54→                return
    55→        }
    56→
    57→        path := filepath.Join(h.staticPath, filepath.Clean(r.URL.Path))
    58→
    59→        fi, err := os.Stat(path)
    60→        if os.IsNotExist(err) || (err == nil && fi.IsDir()) {
    61→                // Serve index.html with no-store so browsers always fetch the latest version.
    62→                // Replace the {{APP_NAME}} placeholder with the real app name so the
    63→                // browser tab shows the correct title immediately, before JS loads.
    64→                w.Header().Set("Cache-Control", "no-store, no-cache, must-revalidate")
    65→                indexFile := filepath.Join(h.staticPath, h.indexPath)
    66→                data, readErr := os.ReadFile(indexFile)
    67→                if readErr != nil {
    68→                        http.Error(w, "Internal Server Error", http.StatusInternalServerError)
    69→                        return
    70→                }
    71→                appName := "App"
    72→                if h.getAppName != nil {
    73→                        if name := h.getAppName(); name != "" {
    74→                                appName = name
    75→                        }
    76→                }
    77→                html := strings.Replace(string(data), "{{APP_NAME}}", appName, 1)
    78→                w.Header().Set("Content-Type", "text/html; charset=utf-8")
    79→                if _, err := w.Write([]byte(html)); err != nil {
    80→                        slog.Warn("server: failed to write index.html", "error", err)
    81→                }
    82→                return
    83→        }
    84→        if err != nil {
    85→                http.Error(w, "Internal Server Error", http.StatusInternalServerError)
    86→                return
    87→        }
    88→
    89→        http.FileServer(http.Dir(h.staticPath)).ServeHTTP(w, r)
    90→}
