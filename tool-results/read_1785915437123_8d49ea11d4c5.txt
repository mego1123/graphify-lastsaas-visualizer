     1→#!/usr/bin/env python3
     2→"""graphify_tenant_audit — tenant-isolation security auditor for Go MongoDB code.
     3→
     4→Scans all .go files under the target path and finds MongoDB queries that are
     5→missing a ``tenantId`` filter — the #1 cause of cross-tenant data leakage in a
     6→multi-tenant SaaS.
     7→
     8→For every detected query (``Find`` / ``FindOne`` / ``InsertOne`` /
     9→``UpdateOne`` / ``UpdateMany`` / ``DeleteOne`` / ``DeleteMany`` / ``Aggregate``
    10→/ ``CountDocuments`` / ``FindOneAndUpdate`` / ``FindOneAndDelete`` /
    11→``FindOneAndReplace`` / ``ReplaceOne`` / ``UpdateByID`` / ``InsertMany`` /
    12→``BulkWrite``), the script:
    13→
    14→  1. Resolves the underlying MongoDB collection (via direct ``.Collection("x")``
    15→     calls or accessor methods like ``h.db.Users()`` defined in the codebase).
    16→  2. Extracts the filter argument — handling inline ``bson.M{}`` / ``bson.D{}``
    17→     literals, ``nil``, and *variable* filters by tracing the variable back to
    18→     its definition (including dynamic field additions like
    19→     ``filter["tenantId"] = tenant.ID``).
    20→  3. Checks whether the filter contains ``tenantId`` or ``tenant_id``.
    21→  4. Classifies the query as OK / MEDIUM / HIGH / CRITICAL:
    22→       - OK       — has a ``tenantId`` filter.
    23→       - MEDIUM   — on a "global" collection (``tenants``, ``plans``,
    24→                    ``system_config``, ``system_logs``, ``users``) that
    25→                    legitimately doesn't need tenant filtering (or is
    26→                    cross-tenant by nature).
    27→       - HIGH     — read op (Find/FindOne/Aggregate/CountDocuments) without
    28→                    ``tenantId`` on a tenant-scoped collection.
    29→       - CRITICAL — write op (InsertOne/UpdateOne/DeleteOne/...) without
    30→                    ``tenantId`` on a tenant-scoped collection.
    31→
    32→  5. Records each query with file, line, collection, function, operation,
    33→     filter fields, the raw snippet, a safe-key heuristic flag (queries that
    34→     filter on a globally-unique key like ``_id`` / ``tokenHash`` / ``slug``
    35→     are noted as likely false positives), and a human-readable note.
    36→
    37→A "violation" is any query classified as HIGH or CRITICAL. MEDIUM queries on
    38→global collections are reported separately for context.
    39→
    40→Usage:
    41→    python graphify_tenant_audit.py [path] [--out report.md] [--json]
    42→
    43→Outputs:
    44→    - JSON written to /home/z/my-project/public/tenant-audit.json (best effort)
    45→    - Markdown written to /home/z/my-project/public/TENANT_AUDIT.md (best effort)
    46→    - Markdown written to --out path if specified
    47→    - JSON to stdout if --json given (without a value)
    48→
    49→Test target: /home/z/my-project/repos/lastsaas/backend
    50→"""

   309→def mask_strings_and_comments(text: str) -> str:
   310→    """Replace string literals and comments with spaces (preserving length and
   311→    newlines) so brace/paren matching is robust against braces inside strings.
   312→
   313→    Go has raw strings (```...```), interpreted strings (``"..."``), rune
   314→    literals (``'...'``), line comments (``//...``), and block comments
   315→    (``/*...*/``). All are masked here.
   316→    """
   317→    out: list[str] = []
   318→    i = 0
   319→    n = len(text)
   320→    while i < n:
   321→        c = text[i]
   322→        # Block comment
   323→        if c == "/" and i + 1 < n and text[i + 1] == "*":
   324→            end = text.find("*/", i + 2)
   325→            if end == -1:
   326→                # Unterminated — mask to end of file.
   327→                for j in range(i, n):
   328→                    out.append("\n" if text[j] == "\n" else " ")
   329→                return "".join(out)
   330→            for j in range(i, end + 2):
   331→                out.append("\n" if text[j] == "\n" else " ")
   332→            i = end + 2
   333→            continue
   334→        # Line comment
   335→        if c == "/" and i + 1 < n and text[i + 1] == "/":
   336→            end = text.find("\n", i + 2)
   337→            if end == -1:
   338→                end = n
   339→            for j in range(i, end):
   340→                out.append(" ")
   341→            i = end
   342→            continue
   343→        # Raw string literal
   344→        if c == "`":
   345→            end = text.find("`", i + 1)
   346→            if end == -1:
   347→                end = n
   348→            for j in range(i, end + 1 if end < n else n):
   349→                out.append("\n" if text[j] == "\n" else " ")
   350→            i = end + 1
   351→            continue
   352→        # Interpreted string literal
   353→        if c == '"':
   354→            j = i + 1
   355→            while j < n:
   356→                if text[j] == "\\" and j + 1 < n:
   357→                    out.append(" ")
   358→                    out.append(" ")
   359→                    j += 2
   360→                    continue
   361→                if text[j] == '"':
   362→                    break
   363→                if text[j] == "\n":
   364→                    # Unterminated string at EOL — bail.
   365→                    break
   366→                out.append(" ")
   367→                j += 1
   368→            # Mask the opening & closing quotes too.
   369→            out.append(" ")  # opening quote already added via outer loop? no.
   370→            # We didn't append the opening quote yet — fix that.
   371→            # Actually the outer loop hasn't appended c yet, so we handle it here.
   372→            i = j + 1
   373→            # Append the closing quote as a space (if present).
   374→            if j < n and text[j] == '"':
   375→                out.append(" ")
   376→            continue
   377→        # Rune literal (single-char, possibly escaped).
   378→        if c == "'":
   379→            j = i + 1
   380→            while j < n:
   381→                if text[j] == "\\" and j + 1 < n:
   382→                    j += 2
   383→                    continue
   384→                if text[j] == "'":
   385→                    break
   386→                if text[j] == "\n":
   387→                    break
   388→                j += 1
   389→            out.append(" ")  # opening '
   390→            for _ in range(i + 1, min(j + 1, n)):
   391→                out.append(" ")
   392→            i = j + 1
   393→            continue
   394→        out.append(c)
   395→        i += 1
   396→    return "".join(out)

   399→def build_accessor_map(repo_root: Path) -> dict[str, str]:
   400→    """Build {accessor_method_name: collection_name}.
   401→
   402→    Scans every .go file for ``func (recv) Name() *mongo.Collection {
   403→    return <expr>.Collection("name") }``.
   404→    """
   405→    accessor_map: dict[str, str] = {}
   406→    for path in repo_root.rglob("*.go"):
   407→        if "graphify-out" in path.parts:
   408→            continue
   409→        try:
   410→            text = path.read_text(encoding="utf-8", errors="replace")
   411→        except (OSError, UnicodeDecodeError):
   412→            continue
   413→        # NOTE: do NOT mask strings here — the collection name is itself a
   414→        # string literal that we need to capture.
   415→        for m in ACCESSOR_DEF_RE.finditer(text):
   416→            name, coll = m.group(1), m.group(2)
   417→            accessor_map.setdefault(name, coll)
   418→    return accessor_map

   421→def collect_struct_fields(repo_root: Path) -> dict[str, set[str]]:
   422→    """Build {struct_name: set_of_bson_field_names}.
   423→
   424→    Walks every Go struct declaration and extracts the bson field names from
   425→    struct-tag annotations (e.g. ``bson:"tenantId,omitempty"`` → ``tenantId``).
   426→    Used to check whether an Insert target struct carries a tenantId field.
   427→    """
   428→    struct_re = re.compile(r"^type\s+([A-Z]\w*)\s+struct\s*\{")
   429→    # Matches the bson tag in a struct field annotation:
   430→    #   `json:"x" bson:"tenantId,omitempty" validate:"required"`
   431→    # captures ``tenantId`` (up to the first ``,`` or ``"``).
   432→    tag_re = re.compile(r'bson:"([^",]+)')
   433→    structs: dict[str, set[str]] = {}
   434→    for path in repo_root.rglob("*.go"):
   435→        if "graphify-out" in path.parts:
   436→            continue
   437→        try:
   438→            text = path.read_text(encoding="utf-8", errors="replace")
   439→        except (OSError, UnicodeDecodeError):
   440→            continue
   441→        lines = text.splitlines()
   442→        i = 0
   443→        n = len(lines)
   444→        while i < n:
   445→            m = struct_re.match(lines[i].strip())
   446→            if not m:
   447→                i += 1
   448→                continue
   449→            name = m.group(1)
   450→            fields: set[str] = set()
   451→            i += 1
   452→            depth = 1
   453→            while i < n and depth > 0:
   454→                line = lines[i]
   455→                depth += line.count("{") - line.count("}")
   456→                if depth <= 0:
   457→                    break
   458→                # Extract the bson tag (first field name, ignoring omitempty etc.)
   459→                tm = tag_re.search(line)
   460→                if tm:
   461→                    fname = tm.group(1).split(",")[0]
   462→                    if fname and fname != "-":
   463→                        fields.add(fname)
   464→                i += 1
   465→            structs.setdefault(name, set())
   466→            structs[name].update(fields)
   467→    return structs

   470→def extract_filter_fields_from_text(text: str) -> list[str]:
   471→    """Pull field names out of bson.M / bson.D / map literals in the given
   472→    text. Returns a de-duplicated list (preserving first-seen order).
   473→    """
   474→    fields: list[str] = []
   475→
   476→    # bson.D blocks
   477→    for block in BSON_D_BLOCK_RE.finditer(text):
   478→        inner = block.group(1)
   479→        for m in BSON_D_KEYVAL_NAMED_RE.finditer(inner):
   480→            fields.append(m.group(1))
   481→        for m in BSON_D_KEYVAL_POS_RE.finditer(inner):
   482→            fields.append(m.group(1))
   483→
   484→    # bson.M blocks
   485→    for m in BSON_M_RE.finditer(text):
   486→        for km in BSON_M_KEY_RE.finditer(m.group(1)):
   487→            fields.append(km.group(1))
   488→
   489→    # map[string]interface{} blocks
   490→    for m in MAP_LITERAL_RE.finditer(text):
   491→        for km in MAP_KEY_RE.finditer(m.group(1)):
   492→            fields.append(km.group(1))
   493→
   494→    # Dedupe, drop operators and empty strings.
   495→    seen: set[str] = set()
   496→    out: list[str] = []
   497→    for f in fields:
   498→        if not f or f.startswith("$") or f in seen:
   499→            continue
   500→        seen.add(f)
   501→        out.append(f)
   502→    return out

   505→def find_block_end(text: str, start: int, open_ch: str, close_ch: str) -> int:
   506→    """Return the index of the matching closing bracket starting at ``start``
   507→    (which must point at ``open_ch``). Returns -1 if no match is found.
   508→    """
   509→    assert text[start] == open_ch
   510→    depth = 0
   511→    i = start
   512→    n = len(text)
   513→    while i < n:
   514→        c = text[i]
   515→        if c == open_ch:
   516→            depth += 1
   517→        elif c == close_ch:
   518→            depth -= 1
   519→            if depth == 0:
   520→                return i
   521→        i += 1
   522→    return -1

   525→def extract_call_args(
   526→    masked: str, raw: str, open_paren_idx: int
   527→) -> list[str]:
   528→    """Given the index of the opening ``(`` of a call in ``masked`` (and the
   529→    corresponding ``raw`` text of identical length), return the list of
   530→    top-level argument source strings (split on commas at depth 1).
   531→
   532→    Depth tracking uses ``masked`` (so commas/braces/parens inside string
   533→    literals don't confuse the splitter), but the returned argument text is
   534→    sliced from ``raw`` so that filter field extraction can see the actual
   535→    string keys (e.g. ``"tenantId"``).
   536→    """
   537→    if open_paren_idx >= len(masked) or masked[open_paren_idx] != "(":
   538→        return []
   539→    close_idx = find_block_end(masked, open_paren_idx, "(", ")")
   540→    if close_idx == -1:
   541→        return []
   542→    # The inner text spans [open_paren_idx+1, close_idx) in both masked & raw.
   543→    args: list[str] = []
   544→    depth_paren = 0
   545→    depth_brace = 0
   546→    depth_brack = 0
   547→    buf_start = open_paren_idx + 1
   548→    i = buf_start
   549→    n = close_idx
   550→    while i < n:
   551→        c = masked[i]
   552→        if c == "(":
   553→            depth_paren += 1
   554→        elif c == ")":
   555→            depth_paren -= 1
   556→        elif c == "{":
   557→            depth_brace += 1
   558→        elif c == "}":
   559→            depth_brace -= 1
   560→        elif c == "[":
   561→            depth_brack += 1
   562→        elif c == "]":
   563→            depth_brack -= 1
   564→        elif c == "," and depth_paren == 0 and depth_brace == 0 and depth_brack == 0:
   565→            args.append(raw[buf_start:i].strip())
   566→            buf_start = i + 1
   567→        i += 1
   568→    tail = raw[buf_start:close_idx].strip()
   569→    if tail:
   570→        args.append(tail)
   571→    return args
   572→

   575→# Per-file scanner
   576→# --------------------------------------------------------------------------- #

   578→
   579→def _collect_brace_window(
   580→    masked_lines: list[str],
   581→    raw_lines: list[str],
   582→    start_idx: int,
   583→    start_col: int,
   584→    max_lines: int = 80,
   585→) -> str:
   586→    """Build a multi-line text window starting at
   587→    ``raw_lines[start_idx][start_col:]`` and extending until the braces
   588→    opened on the first line are balanced.
   589→
   590→    Depth tracking uses ``masked_lines`` (so braces/parens inside string
   591→    literals don't prematurely close the window), but the returned text is
   592→    sliced from ``raw_lines`` so that filter field extraction can see the
   593→    actual string keys (e.g. ``"tenantId"``).
   594→
   595→    The two line arrays must have identical length and newline positions
   596→    (which is guaranteed when both are derived from the same source via
   597→    :func:`mask_strings_and_comments`).
   598→    """
   599→    n = len(masked_lines)
   600→    if start_idx >= n:
   601→        return ""
   602→    first_masked = masked_lines[start_idx][start_col:]
   603→    first_raw = raw_lines[start_idx][start_col:]
   604→    window_lines = [first_raw]
   605→    depth = 0
   606→    for ch in first_masked:
   607→        if ch == "{":
   608→            depth += 1
   609→        elif ch == "}":
   610→            depth -= 1
   611→    i = start_idx + 1
   612→    while depth > 0 and i < n and (i - start_idx) < max_lines:
   613→        nxt_masked = masked_lines[i]
   614→        window_lines.append(raw_lines[i])
   615→        for ch in nxt_masked:
   616→            if ch == "{":
   617→                depth += 1
   618→            elif ch == "}":
   619→                depth -= 1
   620→        i += 1
   621→    return "\n".join(window_lines)
   622→
   623→
   624→def scan_file(
   625→    path: Path,
   626→    repo_root: Path,
   627→    accessor_map: dict[str, str],
   628→    struct_fields: dict[str, set[str]],
   629→    queries: list[Query],
   630→    file_index: dict[str, FileInfo],
   631→) -> None:
   632→    """Scan a single Go file for MongoDB queries and classify each."""
   633→    try:
   634→        raw = path.read_text(encoding="utf-8", errors="replace")
   635→    except OSError:
   636→        return
   637→    masked = mask_strings_and_comments(raw)
   638→    lines = masked.splitlines()
   639→    raw_lines = raw.splitlines()
   640→
   641→    rel = _relative(repo_root, path)
   642→    finfo = file_index.setdefault(rel, FileInfo(path=rel))
   643→
   644→    # Per-file alias + filter-variable tracking.
   645→    #   aliases[var_name]      = collection_name           (collection-typed vars)
   646→    #   filter_vars[var_name]  = {fields, def_line, ...}   (bson.M/bson.D/map vars)
   647→    #   pipeline_vars[var_name]= {fields, def_line, ...}   ([]bson.M / mongo.Pipeline)
   648→    #   struct_vars[var_name]  = struct_name               (models.Invitation, etc.)
   649→    aliases: dict[str, str] = {}
   650→    filter_vars: dict[str, dict] = {}
   651→    pipeline_vars: dict[str, dict] = {}
   652→    struct_vars: dict[str, str] = {}
   653→
   654→    current_func = "<package-init>"
   655→    n = len(lines)
   656→    masked_full = masked
   657→    raw_full = raw
   658→
   659→    # Single pass: walk line-by-line, tracking function context and variable
   660→    # assignments. When a MongoDB operation call is found on the current
   661→    # line, extract its arguments from the full masked/raw text (so multi-
   662→    # line calls are handled) and classify the query.
   663→    #
   664→    # NOTE: function/alias/var-def tracking uses the MASKED line (so braces
   665→    # inside strings don't confuse depth tracking), but dynamic field-addition
   666→    # detection (``filter["tenantId"] = ...``) uses the RAW line because the
   667→    # string key must be visible to the regex.
   668→    for idx in range(n):
   669→        line = lines[idx]
   670→        raw_line = raw_lines[idx]
   671→        line_no = idx + 1
   672→        stripped = line.lstrip()
   673→
   674→        # ---- Track current function (and reset per-function state) ----
   675→        if stripped.startswith("func"):
   676→            fm = FUNC_RE.match(stripped)
   677→            if fm:
   678→                current_func = fm.group(1)
   679→                filter_vars.clear()
   680→                pipeline_vars.clear()
   681→                aliases.clear()
   682→                struct_vars.clear()
   683→                continue
   684→
   685→        # ---- Track collection-alias assignments ----
   686→        for m in ALIAS_LITERAL_RE.finditer(line):
   687→            aliases[m.group(1)] = m.group(2)
   688→        for m in ALIAS_ACCESSOR_RE.finditer(line):
   689→            var_name = m.group(1)
   690→            accessor = m.group(2)
   691→            end = m.end()
   692→            if end < len(line) and line[end] == ".":
   693→                continue  # chained call, not a bare alias
   694→            if accessor in accessor_map:
   695→                aliases[var_name] = accessor_map[accessor]
   696→
   697→        # ---- Track filter-variable definitions ----
   698→        # ``filter := bson.M{...}`` — capture the multi-line literal that
   699→        # follows, starting at the variable name so the ``bson.M`` prefix is
   700→        # preserved (the field-extraction regexes require it).
   701→        for m in FILTER_VAR_DEF_RE.finditer(line):
   702→            var_name = m.group(1)
   703→            window = _collect_brace_window(lines, raw_lines, idx, m.start(), max_lines=30)
   704→            fields = extract_filter_fields_from_text(window)
   705→            filter_vars[var_name] = {
   706→                "fields": set(fields),
   707→                "def_line": line_no,
   708→                "source": "inline-literal",
   709→            }
   710→
   711→        # ---- Track filter-variable dynamic field additions ----
   712→        # ``filter["tenantId"] = ...`` — uses the RAW line so the string key
   713→        # is visible (the masked line has the key's characters replaced with
   714→        # spaces).
   715→        for m in FILTER_VAR_SET_RE.finditer(raw_line):
   716→            var_name = m.group(1)
   717→            field_name = m.group(2)
   718→            if var_name in filter_vars:
   719→                filter_vars[var_name]["fields"].add(field_name)
   720→            else:
   721→                # Variable defined elsewhere (function param, loop var, etc.)
   722→                filter_vars[var_name] = {
   723→                    "fields": {field_name},
   724→                    "def_line": line_no,
   725→                    "source": "dynamic-only",
   726→                }
   727→
   728→        # ---- Track pipeline-variable definitions (for Aggregate) ----
   729→        for m in PIPELINE_VAR_DEF_RE.finditer(line):
   730→            var_name = m.group(1)
   731→            window = _collect_brace_window(lines, raw_lines, idx, m.start(), max_lines=80)
   732→            fields = _extract_pipeline_match_fields(window)
   733→            pipeline_vars[var_name] = {
   734→                "fields": set(fields),
   735→                "def_line": line_no,
   736→                "source": "inline-literal",
   737→                "window": window,
   738→            }
   739→
   740→        # ---- Track struct-literal variable assignments ----
   741→        # ``invitation := models.Invitation{...}`` — records the struct name
   742→        # so InsertOne(ctx, invitation) can be checked for tenantId presence
   743→        # via the struct's bson field tags. Uses the RAW line so that the
   744→        # struct name (which is a Go identifier, not a string) is visible.
   745→        for m in STRUCT_VAR_DEF_RE.finditer(raw_line):
   746→            var_name = m.group(1)
   747→            struct_name = m.group(3)
   748→            # Only track if the struct is known (has bson field tags). This
   749→            # filters out generic-looking assignments like ``x := Foo{}``
   750→            # where Foo isn't a tracked struct.
   751→            if struct_name in struct_fields:
   752→                struct_vars[var_name] = struct_name
   753→
   754→        # ---- Find operation calls on this line ----
   755→        line_offset = sum(len(lines[k]) + 1 for k in range(idx))  # +1 for \n
   756→        for op_m in OP_RE.finditer(line):
   757→            operation = op_m.group(1)
   758→            # OP_RE matches ``.Find(`` etc. — op_m.end() is just past the "(".
   759→            open_paren_idx_in_line = op_m.end() - 1
   760→            open_paren_idx_full = line_offset + open_paren_idx_in_line
   761→
   762→            # Skip option-builder calls (``options.Find()`` etc.). Uses the
   763→            # RAW prefix so the literal ``options`` token is reliably visible.
   764→            raw_prefix = raw_line[: op_m.start()]
   765→            if OPTIONS_BUILDER_RE.search(raw_prefix):
   766→                continue
   767→
   768→            # ---- Resolve the collection ----
   769→            # The prefix search uses the RAW line so that collection literals
   770→            # like ``db.Collection("users")`` (which contain a string) are
   771→            # visible. Depth tracking isn't needed here — we're just looking
   772→            # for the last accessor call or collection literal before ``.Op(``.
   773→            collection: Optional[str] = None
   774→            accessor_used: Optional[str] = None
   775→
   776→            lit_m = COLLECTION_LITERAL_RE.search(raw_prefix)
   777→            if lit_m:
   778→                collection = lit_m.group(1)
   779→            else:
   780→                acc_m = re.search(r"([A-Z]\w*)\(\s*\)\s*$", raw_prefix)
   781→                if acc_m:
   782→                    accessor_used = acc_m.group(1)
   783→                    collection = accessor_map.get(accessor_used)
   784→                else:
   785→                    var_m = re.search(r"([A-Za-z_]\w*)\s*$", raw_prefix)
   786→                    if var_m:
   787→                        collection = aliases.get(var_m.group(1))
   788→
   789→            # Fallback: look back a couple of lines for a collection literal
   790→            # or accessor call (handles calls split across lines). Uses RAW
   791→            # lines for the same reason as above.
   792→            if collection is None:
   793→                for back in range(idx - 1, max(idx - 4, -1), -1):
   794→                    if back < 0 or back >= n:
   795→                        continue
   796→                    prev_raw = raw_lines[back]
   797→                    lit_m2 = COLLECTION_LITERAL_RE.search(prev_raw)
   798→                    if lit_m2:
   799→                        collection = lit_m2.group(1)
   800→                        break
   801→                    acc_m2 = re.search(r"([A-Z]\w*)\(\s*\)", prev_raw)
   802→                    if acc_m2 and acc_m2.group(1) in accessor_map:
   803→                        collection = accessor_map[acc_m2.group(1)]
   804→                        if accessor_used is None:
   805→                            accessor_used = acc_m2.group(1)
   806→                        break
   807→
   808→            if collection is None:
   809→                # Last resort: if the accessor name is in the exempt set,
   810→                # use it as the collection name (so we can still classify).
   811→                if accessor_used and accessor_used in EXEMPT_ACCESSORS:
   812→                    collection = _accessor_to_collection(accessor_used)
   813→                else:
   814→                    collection = "<unknown>"
   815→
   816→            # ---- Extract the call arguments ----
   817→            args = extract_call_args(masked_full, raw_full, open_paren_idx_full)
   818→
   819→            # Determine which argument holds the filter/document/pipeline.
   820→            filter_text = ""
   821→            filter_source = "unknown"
   822→            filter_fields: list[str] = []
   823→            has_tenant_id = False
   824→
   825→            # MongoDB driver convention: first arg is always ctx (or t in
   826→            # tests). The filter/document/pipeline is the second arg.
   827→            if operation == "EstimatedDocumentCount":
   828→                # No filter argument.
   829→                filter_source = "no-filter"
   830→            elif operation == "BulkWrite":
   831→                # BulkWrite(ctx, []mongo.WriteModel{...}) — the models slice
   832→                # is too complex to analyse statically.
   833→                filter_source = "bulk-write-models"
   834→                if len(args) >= 2:
   835→                    filter_text = args[1]
   836→            elif operation == "InsertMany":
   837→                # InsertMany(ctx, []interface{}{...}) — the slice contains
   838→                # the documents to insert.
   839→                if len(args) >= 2:
   840→                    filter_text = args[1]
   841→                    filter_fields = extract_filter_fields_from_text(filter_text)
   842→                    filter_source = "inline-document-slice"
   843→                else:
   844→                    filter_source = "missing-arg"
   845→            elif operation == "InsertOne":
   846→                # InsertOne(ctx, document) — document may be a bson.M literal,
   847→                # an inline struct literal (``models.Foo{...}``), or a
   848→                # variable holding a struct literal (``foo`` assigned from
   849→                # ``models.Foo{...}`` earlier in the function).
   850→                if len(args) >= 2:
   851→                    filter_text = args[1]
   852→                    if "bson.M" in filter_text or "bson.D" in filter_text or "map[string]" in filter_text:
   853→                        filter_fields = extract_filter_fields_from_text(filter_text)
   854→                        filter_source = "inline-document"
   855→                    else:
   856→                        # Try inline struct literal first.
   857→                        struct_name = _extract_struct_name(filter_text)
   858→                        # Then fall back to a struct-typed variable alias.
   859→                        if struct_name is None or struct_name not in struct_fields:
   860→                            var_m = re.match(r"^&?(\w+)\s*$", filter_text.strip())
   861→                            if var_m and var_m.group(1) in struct_vars:
   862→                                struct_name = struct_vars[var_m.group(1)]
   863→                                filter_source = f"struct-var:{var_m.group(1)}:{struct_name}"
   864→                        if struct_name and struct_name in struct_fields:
   865→                            if not filter_source.startswith("struct-var:"):
   866→                                filter_source = f"struct:{struct_name}"
   867→                            bson_fields = struct_fields[struct_name]
   868→                            # Don't pollute filter_fields with every struct
   869→                            # field — just record the ones relevant for
   870→                            # tenant filtering. We expose the full field set
   871→                            # for context, but only treat tenantId presence
   872→                            # as "has_tenant_id" if the struct HAS a
   873→                            # tenantId bson field.
   874→                            filter_fields = sorted(
   875→                                f for f in bson_fields if not f.startswith("_")
   876→                            )
   877→                            if TENANT_ID_FIELDS & bson_fields:
   878→                                has_tenant_id = True
   879→                        else:
   880→                            filter_source = "struct:unknown"
   881→                else:
   882→                    filter_source = "missing-arg"
   883→            elif operation == "UpdateByID":
   884→                # UpdateByID(ctx, id, update) — id is the document _id (unique).
   885→                filter_source = "by-id"
   886→                if len(args) >= 2:
   887→                    filter_text = args[1]
   888→                    # Mark as having a safe unique key.
   889→                    filter_fields = ["_id"]
   890→            elif operation == "Aggregate":
   891→                # Aggregate(ctx, pipeline) — pipeline is a slice of stages.
   892→                if len(args) >= 2:
   893→                    filter_text = args[1]
   894→                    if (
   895→                        filter_text.startswith("bson.M")
   896→                        or filter_text.startswith("bson.D")
   897→                        or filter_text.startswith("bson.A")
   898→                        or filter_text.startswith("[]bson")
   899→                        or filter_text.startswith("mongo.Pipeline")
   900→                    ):
   901→                        # Inline pipeline literal.
   902→                        filter_fields = _extract_pipeline_match_fields(filter_text)
   903→                        filter_source = "inline-pipeline"
   904→                    else:
   905→                        # Variable pipeline — look up in pipeline_vars.
   906→                        var_m = re.match(r"^(\w+)\s*$", filter_text.strip())
   907→                        if var_m and var_m.group(1) in pipeline_vars:
   908→                            info = pipeline_vars[var_m.group(1)]
   909→                            filter_fields = sorted(info["fields"])
   910→                            filter_source = f"variable-pipeline:{var_m.group(1)}"
   911→                        else:
   912→                            filter_source = "variable-pipeline:unknown"
   913→                else:
   914→                    filter_source = "missing-arg"
   915→            elif operation in FILTER_AT_POS_2:
   916→                # Standard filter-bearing call: <Op>(ctx, filter, ...).
   917→                if len(args) >= 2:
   918→                    filter_text = args[1]
   919→                    if filter_text == "nil":
   920→                        filter_source = "nil"
   921→                        filter_fields = []
   922→                    elif filter_text.startswith("bson.M") or filter_text.startswith("bson.D") or filter_text.startswith("map[string]"):
   923→                        filter_fields = extract_filter_fields_from_text(filter_text)
   924→                        filter_source = "inline"
   925→                    else:
   926→                        # Variable filter — look up in filter_vars.
   927→                        var_m = re.match(r"^(\w+)\s*$", filter_text.strip())
   928→                        if var_m and var_m.group(1) in filter_vars:
   929→                            info = filter_vars[var_m.group(1)]
   930→                            filter_fields = sorted(info["fields"])
   931→                            filter_source = f"variable:{var_m.group(1)}"
   932→                        else:
   933→                            filter_source = f"variable:unknown:{filter_text.strip()[:40]}"
   934→                else:
   935→                    filter_source = "missing-arg"
   936→            else:
   937→                # Should not happen — every op is handled above.
   938→                filter_source = "unhandled"
   939→
   940→            # ---- Determine has_tenant_id ----
   941→            if filter_fields:
   942→                field_set = set(filter_fields)
   943→                if TENANT_ID_FIELDS & field_set:
   944→                    has_tenant_id = True
   945→
   946→            # ---- Classify ----
   947→            is_exempt = _is_exempt_collection(collection, accessor_used)
   948→            safe_key = _has_safe_unique_key(filter_fields, operation)
   949→
   950→            if has_tenant_id:
   951→                risk_level = "OK"
   952→                is_violation = False
   953→                note = "Filter contains tenantId."
   954→            elif is_exempt:
   955→                risk_level = "MEDIUM"
   956→                is_violation = False
   957→                note = (
   958→                    f"Collection '{collection}' is in the exempt (global) list; "
   959→                    "tenant filtering is not strictly required, but queries "
   960→                    "should still be reviewed for appropriate scoping."
   961→                )
   962→            elif operation in WRITE_OPS:
   963→                risk_level = "CRITICAL"
   964→                is_violation = True
   965→                if safe_key:
   966→                    note = (
   967→                        "Write operation on a tenant-scoped collection without "
   968→                        "tenantId in the filter, but the filter contains a "
   969→                        "globally-unique key — likely safe, manual review "
   970→                        "recommended."
   971→                    )
   972→                elif filter_source == "by-id":
   973→                    note = (
   974→                        "UpdateByID uses the document _id (globally unique) — "
   975→                        "tenant filtering is implicit. Likely safe."
   976→                    )
   977→                elif filter_source.startswith("struct:"):
   978→                    struct_name = filter_source.split(":", 1)[1]
   979→                    if struct_name == "unknown":
   980→                        note = (
   981→                            "InsertOne with an unrecognised struct value; "
   982→                            "could not statically verify tenantId presence. "
   983→                            "Manual review required."
   984→                        )
   985→                    else:
   986→                        note = (
   987→                            f"InsertOne with a {struct_name} struct value; "
   988→                            "tenantId presence inferred from the struct "
   989→                            "definition — verify the value is actually set at "
   990→                            "runtime."
   991→                        )
   992→                elif filter_source.startswith("struct-var:"):
   993→                    # ``foo := models.Foo{...}; InsertOne(ctx, foo)``
   994→                    # — struct DOES NOT have a tenantId bson field.
   995→                    parts = filter_source.split(":")
   996→                    var_name = parts[1] if len(parts) > 1 else "?"
   997→                    struct_name = parts[2] if len(parts) > 2 else "?"
   998→                    note = (
   999→                        f"InsertOne with {var_name} (a {struct_name} struct); "
  1000→                        f"the struct definition does NOT declare a tenantId "
  1001→                        f"bson field. CRITICAL: inserted document will be "
  1002→                        f"orphaned across tenants."
  1003→                    )
  1004→                elif operation in ("InsertOne", "InsertMany") and (
  1005→                    filter_source in ("inline-document", "inline-document-slice")
  1006→                    or filter_source == "missing-arg"
  1007→                ):
  1008→                    note = (
  1009→                        f"{operation} inserts a document with no tenantId "
  1010→                        "field. CRITICAL: the inserted document will be "
  1011→                        "orphaned — readable/modifiable by any tenant that "
  1012→                        "queries without a tenantId filter, or invisible to "
  1013→                        "queries that DO filter by tenantId."
  1014→                    )
  1015→                elif operation == "BulkWrite":
  1016→                    note = (
  1017→                        "BulkWrite with a slice of WriteModel operations — "
  1018→                        "individual model filters cannot be analysed "
  1019→                        "statically. CRITICAL: manual review required to "
  1020→                        "confirm every UpdateOne/DeleteOne/ReplaceOne model "
  1021→                        "in the slice includes a tenantId filter."
  1022→                    )
  1023→                elif filter_source == "nil":
  1024→                    note = (
  1025→                        f"{operation} with nil filter — affects ALL "
  1026→                        "documents in the collection. CRITICAL: this can "
  1027→                        "modify/delete data across all tenants."
  1028→                    )
  1029→                elif not filter_fields and operation in (
  1030→                    "UpdateOne", "UpdateMany", "DeleteOne", "DeleteMany",
  1031→                    "FindOneAndUpdate", "FindOneAndDelete", "FindOneAndReplace",
  1032→                    "ReplaceOne",
  1033→                ):
  1034→                    note = (
  1035→                        f"{operation} with an empty filter — affects ALL "
  1036→                        "documents. CRITICAL: cross-tenant data corruption "
  1037→                        "risk."
  1038→                    )
  1039→                else:
  1040→                    note = (
  1041→                        f"{operation} on a tenant-scoped collection without "
  1042→                        "tenantId in the filter. CRITICAL: this can "
  1043→                        "modify/delete data belonging to other tenants."
  1044→                    )
  1045→            else:
  1046→                # Read operation.
  1047→                risk_level = "HIGH"
  1048→                is_violation = True
  1049→                if safe_key:
  1050→                    note = (
  1051→                        "Read operation without tenantId, but the filter "
  1052→                        "contains a globally-unique key — likely safe, "
  1053→                        "manual review recommended."
  1054→                    )
  1055→                elif filter_source == "no-filter":
  1056→                    note = (
  1057→                        f"{operation} returns aggregate info about ALL "
  1058→                        "documents in the collection (no filter is taken). "
  1059→                        "HIGH: leaks the total document count across all "
  1060→                        "tenants."
  1061→                    )
  1062→                elif filter_source == "nil":
  1063→                    note = (
  1064→                        "Read operation with nil filter — returns ALL "
  1065→                        "documents in the collection. HIGH: cross-tenant "
  1066→                        "data leakage risk."
  1067→                    )
  1068→                elif not filter_fields:
  1069→                    note = (

  1118→def _extract_pipeline_match_fields(text: str) -> list[str]:
  1119→    """Extract field names from ``$match`` stages in an aggregate pipeline.
  1120→
  1121→    A ``$match`` stage looks like ``{"$match": bson.M{...}}`` or
  1122→    ``bson.D{{Key: "$match", Value: bson.D{...}}}``. We locate the inner
  1123→    filter literal and delegate to :func:`extract_filter_fields_from_text`,
  1124→    passing enough surrounding context (the ``bson.M`` / ``bson.D`` prefix)
  1125→    that the BSON regexes can match.
  1126→    """
  1127→    fields: list[str] = []
  1128→    match_re = re.compile(
  1129→        r'"\$match"\s*:\s*(bson\.M|bson\.D)\s*\{', re.IGNORECASE
  1130→    )
  1131→    for m in match_re.finditer(text):
  1132→        # Find the closing brace of the ``bson.M{...}`` block.
  1133→        brace_idx = m.end() - 1
  1134→        end = find_block_end(text, brace_idx, "{", "}")
  1135→        if end == -1:
  1136→            continue
  1137→        # Include the ``bson.M`` prefix in the slice so that
  1138→        # ``BSON_M_RE`` / ``BSON_D_RE`` (which require the prefix) match.
  1139→        bson_prefix_start = m.start(1)
  1140→        inner = text[bson_prefix_start : end + 1]
  1141→        fields.extend(extract_filter_fields_from_text(inner))
  1142→    # Dedupe.
  1143→    seen: set[str] = set()
  1144→    out: list[str] = []
  1145→    for f in fields:
  1146→        if f and not f.startswith("$") and f not in seen:
  1147→            seen.add(f)
  1148→            out.append(f)
  1149→    return out

  1152→def _extract_struct_name(text: str) -> Optional[str]:
  1153→    """Best-effort: extract the struct name from an argument like
  1154→    ``models.User{...}`` or ``&models.User{...}`` or ``User{...}``.
  1155→    """
  1156→    m = re.search(r"&?([\w\.]+)\s*\{", text)
  1157→    if m:
  1158→        # Take the last component after any dots.
  1159→        name = m.group(1).split(".")[-1]
  1160→        return name
  1161→    return None

  1173→def _accessor_to_collection(accessor: str) -> str:
  1174→    """Convert a PascalCase accessor name (e.g. ``Users``) to its snake_case
  1175→    collection name (e.g. ``users``). Best-effort.
  1176→    """
  1177→    # Handle common acronyms first.
  1178→    out: list[str] = []
  1179→    cur = ""
  1180→    for ch in accessor:
  1181→        if ch.isupper():
  1182→            if cur:
  1183→                out.append(cur)
  1184→                cur = ch.lower()
  1185→            else:
  1186→                cur = ch.lower()
  1187→        else:
  1188→            cur += ch
  1189→    if cur:
  1190→        out.append(cur)
  1191→    return "_".join(out)

  1204→def _build_snippet(raw_lines: list[str], line_no: int, filter_text: str) -> str:
  1205→    """Build a compact human-readable snippet for the report.
  1206→
  1207→    Uses the raw (unmasked) source lines so the snippet shows the actual code
  1208→    the developer wrote (with strings/comments intact).
  1209→    """
  1210→    if 1 <= line_no <= len(raw_lines):
  1211→        first = raw_lines[line_no - 1].strip()
  1212→    else:
  1213→        first = ""
  1214→    # Include up to 3 following lines if the call clearly continues.
  1215→    snippet_lines = [first]
  1216→    depth_paren = first.count("(") - first.count(")")
  1217→    depth_brace = first.count("{") - first.count("}")
  1218→    i = line_no + 1
  1219→    while (depth_paren > 0 or depth_brace > 0) and i <= len(raw_lines) and (i - line_no) < 5:
  1220→        nxt = raw_lines[i - 1].rstrip()
  1221→        snippet_lines.append(nxt)
  1222→        depth_paren += nxt.count("(") - nxt.count(")")
  1223→        depth_brace += nxt.count("{") - nxt.count("}")
  1224→        i += 1
  1225→    snippet = "\n".join(snippet_lines).strip()
  1226→    if len(snippet) > 400:
  1227→        snippet = snippet[:397] + "..."
  1228→    return snippet

  1236→def build_summary(
  1237→    queries: list[Query],
  1238→    file_index: dict[str, FileInfo],
  1239→    accessor_map: dict[str, str],
  1240→    repo_root: Path,
  1241→) -> dict:
  1242→    """Build the JSON-serialisable summary object."""
  1243→    total = len(queries)
  1244→    with_tenant = sum(1 for q in queries if q.has_tenant_id)
  1245→    without_tenant = total - with_tenant
  1246→    violations = [q for q in queries if q.is_violation]
  1247→    critical = [q for q in violations if q.risk_level == "CRITICAL"]
  1248→    high = [q for q in violations if q.risk_level == "HIGH"]
  1249→    medium = [q for q in queries if q.risk_level == "MEDIUM"]
  1250→    safe_key_violations = [q for q in violations if q.safe_key_filter]
  1251→    real_violations = [q for q in violations if not q.safe_key_filter]
  1252→
  1253→    by_collection: Counter = Counter()
  1254→    by_operation: Counter = Counter()
  1255→    violations_by_collection: Counter = Counter()
  1256→    violations_by_file: Counter = Counter()
  1257→    for q in queries:
  1258→        by_collection[q.collection] += 1
  1259→        by_operation[q.operation] += 1
  1260→        if q.is_violation:
  1261→            violations_by_collection[q.collection] += 1
  1262→            violations_by_file[q.file] += 1
  1263→
  1264→    # Top files by violation count.
  1265→    top_files = [
  1266→        {"file": f, "violations": v, "total": file_index[f].total_queries}
  1267→        for f, v in violations_by_file.most_common(20)
  1268→    ]
  1269→
  1270→    return {
  1271→        "repo_root": str(repo_root),
  1272→        "total_queries": total,
  1273→        "queries_with_tenant_id": with_tenant,
  1274→        "queries_without_tenant_id": without_tenant,
  1275→        "violations": len(violations),
  1276→        "critical_violations": len(critical),
  1277→        "high_violations": len(high),
  1278→        "medium_global_collection_queries": len(medium),
  1279→        "safe_key_violations": len(safe_key_violations),
  1280→        "real_violations_needing_review": len(real_violations),
  1281→        "pct_with_tenant_id": (
  1282→            round(100.0 * with_tenant / total, 2) if total else 0.0
  1283→        ),
  1284→        "queries_by_collection": [
  1285→            {"collection": c, "queries": n}
  1286→            for c, n in by_collection.most_common()
  1287→        ],
  1288→        "queries_by_operation": [
  1289→            {"operation": op, "count": n}
  1290→            for op, n in by_operation.most_common()
  1291→        ],
  1292→        "violations_by_collection": [
  1293→            {"collection": c, "violations": n}
  1294→            for c, n in violations_by_collection.most_common()
  1295→        ],
  1296→        "top_files_by_violations": top_files,
  1297→        "exempt_collections": sorted(EXEMPT_COLLECTIONS),
  1298→        "tenant_id_fields": sorted(TENANT_ID_FIELDS),
  1299→        "safe_unique_keys": sorted(SAFE_UNIQUE_KEYS),
  1300→        "accessor_map_size": len(accessor_map),
  1301→    }
