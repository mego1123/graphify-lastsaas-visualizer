     1→     1→// graphify_filter_tracer uses golang.org/x/tools/go/ssa to trace which
     2→     2→// fields get written to bson.M / bson.D filter variables in functions
     3→     3→// that perform MongoDB operations.
     4→     4→//
     5→     5→// For each function that contains a call to one of:
     6→     6→//
     7→     7→//      Find, FindOne, FindOneAndUpdate, FindOneAndDelete, FindOneAndReplace,
     8→     8→//      UpdateOne, UpdateMany, DeleteOne, DeleteMany, InsertOne, InsertMany,
     9→     9→//      CountDocuments, Aggregate, ReplaceOne, BulkWrite
    10→    10→//
    11→    11→// it produces one or more Finding records describing how a filter (or
    12→    12→// document / update) was constructed:
    13→    13→//
    14→    14→//   - "literal"     — a bson.M{...} or bson.D{...} composite literal that
    15→    15→//                     has at least one key. Keys are extracted from the
    16→    16→//                     AST so that both map-style and bson.E-style literals
    17→    17→//                     are handled.
    18→    18→//   - "map_update"  — an explicit `filter["x"] = y` write discovered via
    19→    19→//                     ssa.MapUpdate. MapUpdates that are part of an
    20→    20→//                     inline bson.M{...} literal are deduplicated against
    21→    21→//     the "literal" pass using source positions, so they only appear once.
    22→    22→//   - "struct_type" — a Call to InsertOne / InsertMany / ReplaceOne whose
    23→    23→//                     document argument is a struct (or slice of structs).
    24→    24→//                     The reported fields are the bson tag names of the
    25→    25→//                     struct's fields (flattened through embedded structs).
    26→    26→//
    27→    27→// Output is a JSON array of FunctionReport objects, written to stdout or
    28→    28→// a file with -out.
    29→    29→//
    30→    30→// Usage:
    31→    31→//
    32→    32→//      cd /path/to/go/project && go run /path/to/graphify_filter_tracer/main.go -out filters.json

    55→    55→// Finding describes one observation of how a filter / document was
    56→    56→// constructed inside a function.
    57→    57→type Finding struct {

    59→    59→        Operation  string   `json:"operation,omitempty"`  // Find, FindOne, InsertOne, ...

   152→   152→func run(cfg Config) error {
   153→   153→        if cfg.Verbose {
   154→   154→                fmt.Fprintf(os.Stderr, "loading packages: %v\n", cfg.Patterns)
   155→   155→        }
   156→   156→
   157→   157→        mode := packages.NeedName | packages.NeedFiles | packages.NeedTypes |
   158→   158→                packages.NeedTypesInfo | packages.NeedSyntax | packages.NeedDeps |
   159→   159→                packages.NeedImports | packages.NeedModule
   160→   160→
   161→   161→        pkgs, err := packages.Load(&packages.Config{
   162→   162→                Mode:       mode,
   163→   163→                Tests:      cfg.Tests,
   164→   164→                BuildFlags: []string{"-tags=!ignore"},
   165→   165→        }, cfg.Patterns...)
   166→   166→        if err != nil {
   167→   167→                return fmt.Errorf("packages.Load: %w", err)
   168→   168→        }
   169→   169→
   170→   170→        var loadErrs int
   171→   171→        for _, p := range pkgs {
   172→   172→                for _, e := range p.Errors {
   173→   173→                        loadErrs++
   174→   174→                        if cfg.Verbose {
   175→   175→                                fmt.Fprintf(os.Stderr, "  pkg %s: %v\n", p.ID, e)
   176→   176→                        }
   177→   177→                }
   178→   178→        }
   179→   179→        if cfg.Verbose {
   180→   180→                fmt.Fprintf(os.Stderr, "loaded %d packages (%d errors)\n", len(pkgs), loadErrs)
   181→   181→        }
   182→   182→
   183→   183→        // Build SSA only for the initial packages (deps=false). Dependency
   184→   184→        // packages get placeholder SSA packages with no instructions, which
   185→   185→        // is fine — we only walk the initial set's functions.
   186→   186→        prog, ssaPkgs := ssautil.Packages(pkgs, ssa.InstantiateGenerics)
   187→   187→        prog.Build()
   188→   188→        if cfg.Verbose {
   189→   189→                fmt.Fprintf(os.Stderr, "built SSA for %d packages\n", len(ssaPkgs))
   190→   190→        }
   191→   191→
   192→   192→        // Map types.Package -> bool for the initial set, so we can filter
   193→   193→        // out functions belonging to dependencies.
   194→   194→        initial := make(map[*types.Package]bool)
   195→   195→        // Map types.Package -> *types.Info so the AST pass can do precise
   196→   196→        // type resolution (e.g. detect `map[string]interface{}` literals
   197→   197→        // that aren't typed as bson.M).
   198→   198→        infoByPkg := make(map[*types.Package]*types.Info)
   199→   199→        for _, p := range pkgs {
   200→   200→                if p.Types != nil {
   201→   201→                        initial[p.Types] = true
   202→   202→                        if p.TypesInfo != nil {
   203→   203→                                infoByPkg[p.Types] = p.TypesInfo
   204→   204→                        }
   205→   205→                }
   206→   206→        }
   207→   207→
   208→   208→        fset := prog.Fset
   209→   209→        reports := []FunctionReport{}
   210→   210→
   211→   211→        for fn := range ssautil.AllFunctions(prog) {
   212→   212→                // Skip synthetic functions (anonymous wrappers, init wrappers,
   213→   213→                // interface thunks, etc.) — they have no source-level object.
   214→   214→                if fn.Object() == nil {
   215→   215→                        continue
   216→   216→                }
   217→   217→                // Skip functions from dependency packages.
   218→   218→                pkg := fn.Package()
   219→   219→                if pkg == nil || pkg.Pkg == nil || !initial[pkg.Pkg] {
   220→   220→                        continue
   221→   221→                }
   222→   222→                // Skip functions with no body (e.g. extern declarations).
   223→   223→                if fn.Blocks == nil || len(fn.Blocks) == 0 {
   224→   224→                        continue
   225→   225→                }
   226→   226→
   227→   227→                // Determine whether this function calls any mongo op. While
   228→   228→                // doing so, collect the call sites so we can later attribute
   229→   229→                // findings to specific operations.
   230→   230→                mongoCalls := collectMongoCalls(fn)
   231→   231→                if len(mongoCalls) == 0 {
   232→   232→                        continue
   233→   233→                }
   234→   234→
   235→   235→                pos := fset.Position(fn.Pos())
   236→   236→                rpt := FunctionReport{
   237→   237→                        Function: fn.RelString(pkg.Pkg),
   238→   238→                        Package:  pkg.Pkg.Path(),
   239→   239→                        File:     relPath(pos.Filename),
   240→   240→                        Line:     pos.Line,
   241→   241→                        Findings: []Finding{},
   242→   242→                }
   243→   243→
   244→   244→                info := infoByPkg[pkg.Pkg]
   245→   245→
   246→   246→                // Pass 1: AST scan for bson.M{...} / bson.D{...} / map[string]any{}
   247→   247→                // literals. Records source ranges of all matched literals so the
   248→   248→                // MapUpdate pass can dedupe inline-construction updates.
   249→   249→                literalRanges := scanLiterals(fn, info, fset, &rpt)
   250→   250→
   251→   251→                // Pass 2: SSA scan for MapUpdates on bson.M-typed maps.
   252→   252→                scanMapUpdates(fn, fset, literalRanges, &rpt)
   253→   253→
   254→   254→                // Pass 3: SSA scan for struct-typed args to InsertOne / etc.
   255→   255→                scanStructArgs(fn, fset, mongoCalls, &rpt)
   256→   256→
   257→   257→                // Only emit reports for functions that produced at least one
   258→   258→                // finding OR explicitly call a mongo op (so callers can see
   259→   259→                // "this function uses Mongo but we couldn't extract a filter").
   260→   260→                if len(rpt.Findings) > 0 {
   261→   261→                        reports = append(reports, rpt)
   262→   262→                }
   263→   263→        }
   264→   264→
   265→   265→        // Stable ordering.
   266→   266→        sort.Slice(reports, func(i, j int) bool {
   267→   267→                if reports[i].Package != reports[j].Package {
   268→   268→                        return reports[i].Package < reports[j].Package
   269→   269→                }
   270→   270→                if reports[i].File != reports[j].File {
   271→   271→                        return reports[i].File < reports[j].File
   272→   272→                }
   273→   273→                return reports[i].Line < reports[j].Line
   274→   274→        })
   275→   275→
   276→   276→        if cfg.Verbose {
   277→   277→                total := 0
   278→   278→                for _, r := range reports {
   279→   279→                        total += len(r.Findings)
   280→   280→                }
   281→   281→                fmt.Fprintf(os.Stderr, "emitted %d findings across %d functions\n", total, len(reports))
   282→   282→        }
   283→   283→
   284→   284→        buf, err := json.MarshalIndent(reports, "", "  ")
   285→   285→        if err != nil {
   286→   286→                return fmt.Errorf("marshal: %w", err)
   287→   287→        }
   288→   288→        if cfg.Out != "" {
   289→   289→                if err := os.WriteFile(cfg.Out, buf, 0o644); err != nil {
   290→   290→                        return fmt.Errorf("write %s: %w", cfg.Out, err)
   291→   291→                }
   292→   292→                if cfg.Verbose {
   293→   293→                        fmt.Fprintf(os.Stderr, "wrote %s (%d bytes)\n", cfg.Out, len(buf))
   294→   294→                }
   295→   295→                return nil
   296→   296→        }
   297→   297→        _, _ = os.Stdout.Write(buf)
   298→   298→        _, _ = os.Stdout.Write([]byte("\n"))
   299→   299→        return nil
   300→   300→}

   302→   302→// mongoCall is a recorded call to one of the MongoDB operations.
   303→   303→type mongoCall struct {
   304→   304→        op   string         // method name, e.g. "Find"
   305→   305→        call *ssa.Call      // the SSA call instruction
   306→   306→        pos  token.Position // position of the call
   307→   307→}
   308→   308→
   309→   309→// collectMongoCalls walks every block of fn looking for *ssa.Call
   310→   310→// instructions whose callee (static or invoked) matches one of the
   311→   311→// MongoDB operation names.
   312→   312→func collectMongoCalls(fn *ssa.Function) []mongoCall {
   313→   313→        var out []mongoCall
   314→   314→        for _, b := range fn.Blocks {
   315→   315→                for _, instr := range b.Instrs {
   316→   316→                        call, ok := instr.(*ssa.Call)
   317→   317→                        if !ok {
   318→   318→                                continue
   319→   319→                        }
   320→   320→                        cc := call.Common()
   321→   321→                        name := ""
   322→   322→                        if cc.IsInvoke() {
   323→   323→                                name = cc.Method.Name()
   324→   324→                        } else if f := cc.StaticCallee(); f != nil {
   325→   325→                                name = f.Name()
   326→   326→                        }
   327→   327→                        if name == "" || !mongoOps[name] {
   328→   328→                                continue
   329→   329→                        }
   330→   330→                        out = append(out, mongoCall{
   331→   331→                                op:   name,
   332→   332→                                call: call,
   333→   333→                                pos:  fn.Prog.Fset.Position(cc.Pos()),
   334→   334→                        })
   335→   335→                }
   336→   336→        }
   337→   337→        return out
   338→   338→}
   339→   339→
   340→   340→// scanLiterals walks the AST of fn's body looking for composite literals
   341→   341→// whose type is bson.M, bson.D, or map[string]interface{} (the underlying
   342→   342→// type of bson.M). For each non-empty literal it emits a "literal"
   343→   343→// finding with the field keys. It also returns the source ranges of ALL
   344→   344→// matched literals (including empty ones) so the MapUpdate pass can skip
   345→   345→// updates that belong to a literal construction.
   346→   346→//
   347→   347→// Detection uses TypesInfo for precise type resolution when available,
   348→   348→// falling back to AST identifier matching (bson.M / bson.D) so the tool
   349→   349→// still works on packages whose TypesInfo is incomplete.
   350→   350→func scanLiterals(fn *ssa.Function, info *types.Info, fset *token.FileSet, rpt *FunctionReport) []literalRange {
   351→   351→        ranges := []literalRange{}
   352→   352→        syn := fn.Syntax()
   353→   353→        if syn == nil {
   354→   354→                return ranges
   355→   355→        }
   356→   356→        ast.Inspect(syn, func(n ast.Node) bool {
   357→   357→                lit, ok := n.(*ast.CompositeLit)
   358→   358→                if !ok {
   359→   359→                        return true
   360→   360→                }
   361→   361→                bsonKind := classifyLiteral(lit, info)
   362→   362→                if bsonKind == "" {
   363→   363→                        return true
   364→   364→                }
   365→   365→                // Record the literal's source range so the SSA MapUpdate
   366→   366→                // pass can skip updates that belong to this literal. This is
   367→   367→                // done for empty literals too, even though they have no
   368→   368→                // MapUpdates of their own — symmetry is cheap.
   369→   369→                start := fset.Position(lit.Pos())
   370→   370→                end := fset.Position(lit.End())
   371→   371→                ranges = append(ranges, literalRange{
   372→   372→                        file:  start.Filename,
   373→   373→                        start: start.Offset,
   374→   374→                        end:   end.Offset,
   375→   375→                })
   376→   376→                if len(lit.Elts) == 0 {
   377→   377→                        return true
   378→   378→                }
   379→   379→                keys := extractLiteralKeys(bsonKind, lit)
   380→   380→                if len(keys) == 0 {
   381→   381→                        return true
   382→   382→                }
   383→   383→                // Find the nearest enclosing *ast.CallExpr whose callee name
   384→   384→                // is a mongo op (best-effort).
   385→   385→                op := enclosingMongoOp(fn, lit)
   386→   386→                rpt.Findings = append(rpt.Findings, Finding{
   387→   387→                        Method:    "literal",
   388→   388→                        Operation: op,
   389→   389→                        Fields:    keys,
   390→   390→                        Position:  fmt.Sprintf("%s:%d", relPath(start.Filename), start.Line),
   391→   391→                })
   392→   392→                return true
   393→   393→        })
   394→   394→        return ranges
   395→   395→}
   396→   396→
   397→   397→// classifyLiteral returns "M" if lit's type is bson.M or
   398→   398→// map[string]interface{} (the underlying type of bson.M), "D" if it's
   399→   399→// bson.D, and "" otherwise.
   400→   400→func classifyLiteral(lit *ast.CompositeLit, info *types.Info) string {
   401→   401→        // Try precise type resolution first.
   402→   402→        if info != nil {
   403→   403→                if tv, ok := info.Types[lit]; ok {
   404→   404→                        return classifyType(tv.Type)
   405→   405→                }
   406→   406→                // Some literals don't have a Types entry but do have an
   407→   407→                // inferred type via their Type expression.
   408→   408→                if lit.Type != nil {
   409→   409→                        if tv, ok := info.Types[lit.Type]; ok {
   410→   410→                                return classifyType(tv.Type)
   411→   411→                        }
   412→   412→                }
   413→   413→        }
   414→   414→        // Fall back to identifier name matching.
   415→   415→        return matchBsonIdent(lit.Type)
   416→   416→}
   417→   417→
   418→   418→// classifyType inspects a types.Type and returns "M", "D", or "".
   419→   419→//
   420→   420→// Handles *types.Named (e.g. primitive.Regex), *types.Alias (e.g.
   421→   421→// bson.M which is `type M = map[string]interface{}`), and the raw
   422→   422→// underlying types (map[string]interface{} for M, []bson.E for D).
   423→   423→func classifyType(t types.Type) string {
   424→   424→        switch v := t.(type) {
   425→   425→        case *types.Alias:
   426→   426→                obj := v.Obj()
   427→   427→                if obj.Pkg() != nil && obj.Pkg().Path() == "go.mongodb.org/mongo-driver/bson" {
   428→   428→                        switch obj.Name() {
   429→   429→                        case "M":
   430→   430→                                return "M"
   431→   431→                        case "D":
   432→   432→                                return "D"
   433→   433→                        }
   434→   434→                }
   435→   435→                return classifyType(v.Underlying())
   436→   436→        case *types.Named:
   437→   437→                obj := v.Obj()
   438→   438→                if obj.Pkg() != nil && obj.Pkg().Path() == "go.mongodb.org/mongo-driver/bson" {
   439→   439→                        switch obj.Name() {
   440→   440→                        case "M":
   441→   441→                                return "M"
   442→   442→                        case "D":
   443→   443→                                return "D"
   444→   444→                        case "E":
   445→   445→                                // bson.E is the element type of bson.D; the literal
   446→   446→                                // itself is reported as bson.E but it's part of a
   447→   447→                                // bson.D slice. We return "" here so callers don't
   448→   448→                                // mis-classify a bare bson.E as a top-level filter.
   449→   449→                                return ""
   450→   450→                        }
   451→   451→                }
   452→   452→                return classifyType(v.Underlying())
   453→   453→        case *types.Map:
   454→   454→                // map[string]interface{} (or map[string]any) — the underlying
   455→   455→                // type of bson.M.
   456→   456→                basic, ok := v.Key().(*types.Basic)
   457→   457→                if !ok || basic.Kind() != types.String {
   458→   458→                        return ""
   459→   459→                }
   460→   460→                if iface, ok := v.Elem().(*types.Interface); ok && iface.NumMethods() == 0 {
   461→   461→                        return "M"
   462→   462→                }
   463→   463→        case *types.Slice:
   464→   464→                // []bson.E — the underlying type of bson.D.
   465→   465→                if named, ok := v.Elem().(*types.Named); ok {
   466→   466→                        obj := named.Obj()

   488→   488→// matchBsonIdent returns "M" if the AST type expression is the
   489→   489→// identifier `bson.M` (or `M` qualified by a `bson.` selector), "D" if
   490→   490→// it's `bson.D`, and "" otherwise.
   491→   491→func matchBsonIdent(typeExpr ast.Expr) string {
   492→   492→        if typeExpr == nil {
   493→   493→                return ""
   494→   494→        }
   495→   495→        switch t := typeExpr.(type) {
   496→   496→        case *ast.SelectorExpr:
   497→   497→                if ident, ok := t.X.(*ast.Ident); ok {
   498→   498→                        return matchBsonName(ident.Name, t.Sel.Name)
   499→   499→                }
   500→   500→        case *ast.Ident:
   501→   501→                return matchBsonName("", t.Name)
   502→   502→        }
   503→   503→        return ""
   504→   504→}

   506→   506→// matchBsonName returns "M" for a `bson.M` / `M` qualifier, "D" for
   507→   507→// `bson.D` / `D`, and "" otherwise.
   508→   508→func matchBsonName(pkg, name string) string {
   509→   509→        switch name {
   510→   510→        case "M":
   511→   511→                if pkg == "" || pkg == "bson" {
   512→   512→                        return "M"
   513→   513→                }
   514→   514→        case "D":
   515→   515→                if pkg == "" || pkg == "bson" {
   516→   516→                        return "D"
   517→   517→                }
   518→   518→        }
   519→   519→        return ""
   520→   520→}

   522→   522→// extractLiteralKeys pulls the field names out of a bson.M or bson.D
   523→   523→// composite literal. For bson.M (map literal), each element is a
   524→   524→// *ast.KeyValueExpr whose Key is a string literal. For bson.D (slice of
   525→   525→// bson.E), each element is itself a *ast.CompositeLit with a `Key: "..."`
   526→   526→// field.
   527→   527→func extractLiteralKeys(bsonKind string, lit *ast.CompositeLit) []string {
   528→   528→        keys := []string{}
   529→   529→        for _, el := range lit.Elts {
   530→   530→                switch bsonKind {
   531→   531→                case "M":
   532→   532→                        kv, ok := el.(*ast.KeyValueExpr)
   533→   533→                        if !ok {
   534→   534→                                continue
   535→   535→                        }
   536→   536→                        k := stringLiteral(kv.Key)
   537→   537→                        if k != "" {
   538→   538→                                keys = append(keys, k)
   539→   539→                        }
   540→   540→                case "D":
   541→   541→                        // Each element should be a *ast.CompositeLit (bson.E).
   542→   542→                        cl, ok := el.(*ast.CompositeLit)
   543→   543→                        if !ok {
   544→   544→                                // Could also be a *ast.KeyValueExpr in some
   545→   545→                                // shorthand forms — try anyway.
   546→   546→                                if kv, ok := el.(*ast.KeyValueExpr); ok {
   547→   547→                                        if k := stringLiteral(kv.Key); k != "" {
   548→   548→                                                keys = append(keys, k)
   549→   549→                                        }
   550→   550→                                }
   551→   551→                                continue
   552→   552→                        }
   553→   553→                        k := bsonEKey(cl)
   554→   554→                        if k != "" {
   555→   555→                                keys = append(keys, k)
   556→   556→                        }
   557→   557→                }
   558→   558→        }
   559→   559→        return keys
   560→   560→}
   561→   561→
   562→   562→// bsonEKey looks at a *ast.CompositeLit of type bson.E and returns the
   563→   563→// value of the `Key: "..."` field. Both `bson.E{Key: "x", Value: y}`
   564→   564→// and positional `{Key: "x", Value: y}` forms are handled.
   565→   565→func bsonEKey(cl *ast.CompositeLit) string {
   566→   566→        // Named-field form.
   567→   567→        for _, el := range cl.Elts {
   568→   568→                kv, ok := el.(*ast.KeyValueExpr)
   569→   569→                if !ok {
   570→   570→                        continue
   571→   571→                }
   572→   572→                key, ok := kv.Key.(*ast.Ident)
   573→   573→                if !ok || key.Name != "Key" {
   574→   574→                        continue
   575→   575→                }
   576→   576→                if s := stringLiteral(kv.Value); s != "" {
   577→   577→                        return s
   578→   578→                }
   579→   579→        }
   580→   580→        // Positional form: bson.E{"x", y} — Key is element 0.
   581→   581→        if len(cl.Elts) > 0 {
   582→   582→                if s := stringLiteral(cl.Elts[0]); s != "" {
   583→   583→                        return s
   584→   584→                }
   585→   585→        }
   586→   586→        return ""
   587→   587→}

   589→   589→// stringLiteral returns the unquoted string value of an *ast.BasicLit
   590→   590→// of kind STRING, or "" if expr is not a string literal. Both
   591→   591→// double-quoted ("...") and back-quoted (`...`) forms are handled.
   592→   592→func stringLiteral(expr ast.Expr) string {
   593→   593→        lit, ok := expr.(*ast.BasicLit)
   594→   594→        if !ok || lit.Kind != token.STRING {
   595→   595→                return ""
   596→   596→        }
   597→   597→        s, err := strconv.Unquote(lit.Value)
   598→   598→        if err != nil {
   599→   599→                // strconv.Unquote only handles double-quoted strings; fall
   600→   600→                // back to stripping backticks.
   601→   601→                v := lit.Value
   602→   602→                if len(v) >= 2 && v[0] == '`' && v[len(v)-1] == '`' {
   603→   603→                        return v[1 : len(v)-1]
   604→   604→                }
   605→   605→                return ""
   606→   606→        }
   607→   607→        return s
   608→   608→}

   610→   610→// enclosingMongoOp walks up the AST stack maintained by an outer
   611→   611→// ast.Inspect call to find the nearest enclosing *ast.CallExpr whose
   612→   612→// callee is a method matching one of the mongo op names. Because
   613→   613→// ast.Inspect doesn't expose the parent chain, we re-walk the whole
   614→   614→// function and check whether the literal's position is inside the call's
   615→   615→// argument list.
   616→   616→func enclosingMongoOp(fn *ssa.Function, target ast.Node) string {
   617→   617→        syn := fn.Syntax()
   618→   618→        if syn == nil {
   619→   619→                return ""
   620→   620→        }
   621→   621→        fset := fn.Prog.Fset
   622→   622→        tpos := fset.Position(target.Pos())
   623→   623→        tend := fset.Position(target.End())
   624→   624→
   625→   625→        best := ""
   626→   626→        bestStart := -1
   627→   627→
   628→   628→        ast.Inspect(syn, func(n ast.Node) bool {
   629→   629→                call, ok := n.(*ast.CallExpr)
   630→   630→                if !ok {
   631→   631→                        return true
   632→   632→                }
   633→   633→                name := callMethodName(call.Fun)
   634→   634→                if name == "" || !mongoOps[name] {
   635→   635→                        return true
   636→   636→                }
   637→   637→                cpos := fset.Position(call.Pos())
   638→   638→                cend := fset.Position(call.End())
   639→   639→                // Is the literal inside this call's argument list?
   640→   640→                if tpos.Filename == cpos.Filename &&
   641→   641→                        tpos.Offset >= cpos.Offset && tend.Offset <= cend.Offset {
   642→   642→                        // Pick the *innermost* matching call.
   643→   643→                        if bestStart == -1 || cpos.Offset > bestStart {
   644→   644→                                bestStart = cpos.Offset
   645→   645→                                best = name
   646→   646→                        }
   647→   647→                }
   648→   648→                return true
   649→   649→        })
   650→   650→        return best
   651→   651→}

   653→   653→// callMethodName returns the method name from a call's Fun expression.
   654→   654→// Handles `x.Method(...)`, `x.Method().InnerMethod(...)`, and bare
   655→   655→// `Method(...)` calls.
   656→   656→func callMethodName(fun ast.Expr) string {
   657→   657→        sel, ok := fun.(*ast.SelectorExpr)
   658→   658→        if !ok {
   659→   659→                return ""
   660→   660→        }
   661→   661→        return sel.Sel.Name
   662→   662→}

   664→   664→// scanMapUpdates walks the SSA blocks of fn and emits one "map_update"
   665→   665→// finding for each ssa.MapUpdate whose Map's type is bson.M
   666→   666→// (map[string]interface{}). MapUpdates whose position falls inside a
   667→   667→// bson.M / bson.D / map[string]interface{} literal's source range (as
   668→   668→// recorded by scanLiterals) are skipped — they are part of the literal
   669→   669→// construction and are already reported (or skipped) as literals. Only
   670→   670→// explicit user-written mutations like `filter["x"] = y` survive.
   671→   671→func scanMapUpdates(fn *ssa.Function, fset *token.FileSet, literalRanges []literalRange, rpt *FunctionReport) {
   672→   672→        for _, b := range fn.Blocks {
   673→   673→                for _, instr := range b.Instrs {
   674→   674→                        upd, ok := instr.(*ssa.MapUpdate)
   675→   675→                        if !ok {
   676→   676→                                continue
   677→   677→                        }
   678→   678→                        if !isBsonM(upd.Map.Type()) {
   679→   679→                                continue
   680→   680→                        }
   681→   681→                        pos := fset.Position(upd.Pos())
   682→   682→                        // Skip if this update is inside a literal we already reported
   683→   683→                        // or recorded as a range (e.g. `filter := bson.M{...}` whose
   684→   684→                        // construction SSA lowers into MakeMap + MapUpdates).
   685→   685→                        if inAnyRange(literalRanges, pos) {
   686→   686→                                continue
   687→   687→                        }
   688→   688→                        key := constString(upd.Key)
   689→   689→                        rpt.Findings = append(rpt.Findings, Finding{
   690→   690→                                Method:    "map_update",
   691→   691→                                Fields:    []string{key},
   692→   692→                                Variable:  valueName(upd.Map),
   693→   693→                                Position:  fmt.Sprintf("%s:%d", relPath(pos.Filename), pos.Line),
   694→   694→                        })
   695→   695→                }
   696→   696→        }
   697→   697→}

   699→   699→// scanStructArgs walks the SSA blocks of fn looking for calls to
   700→   700→// InsertOne / InsertMany / ReplaceOne (and friends) whose document
   701→   701→// argument is a struct (or slice of structs). For each, it emits a
   702→   702→// "struct_type" finding with the struct's bson-tagged field names.
   703→   703→func scanStructArgs(fn *ssa.Function, fset *token.FileSet, calls []mongoCall, rpt *FunctionReport) {
   704→   704→        for _, mc := range calls {
   705→   705→                if !methodsWhoseArg1IsDocument[mc.op] {
   706→   706→                        continue
   707→   707→                }
   708→   708→                cc := mc.call.Common()
   709→   709→                argIdx := dataArgIndex(cc)
   710→   710→                if argIdx < 0 || argIdx >= len(cc.Args) {
   711→   711→                        continue
   712→   712→                }
   713→   713→                arg := cc.Args[argIdx]
   714→   714→                t := concreteType(arg)
   715→   715→                if t == nil {
   716→   716→                        continue
   717→   717→                }
   718→   718→                // Unwrap pointer types.
   719→   719→                if ptr, ok := t.(*types.Pointer); ok {
   720→   720→                        t = ptr.Elem()
   721→   721→                }
   722→   722→                // Unwrap named types.
   723→   723→                named, _ := t.(*types.Named)
   724→   724→                var structType *types.Struct
   725→   725→                var sliceElem types.Type
   726→   726→                if s, ok := t.Underlying().(*types.Struct); ok {
   727→   727→                        structType = s
   728→   728→                } else if sl, ok := t.Underlying().(*types.Slice); ok {
   729→   729→                        sliceElem = sl.Elem()
   730→   730→                        if ptr, ok := sliceElem.(*types.Pointer); ok {
   731→   731→                                sliceElem = ptr.Elem()
   732→   732→                        }
   733→   733→                        if s, ok := sliceElem.Underlying().(*types.Struct); ok {
   734→   734→                                structType = s
   735→   735→                        }
   736→   736→                }
   737→   737→                if structType == nil {
   738→   738→                        continue
   739→   739→                }
   740→   740→                typeName := typeString(t)
   741→   741→                if typeName == "" && named != nil {
   742→   742→                        typeName = named.Obj().Pkg().Path() + "." + named.Obj().Name()
   743→   743→                }
   744→   744→                rpt.Findings = append(rpt.Findings, Finding{
   745→   745→                        Method:     "struct_type",
   746→   746→                        Operation:  mc.op,
   747→   747→                        StructType: typeName,
   748→   748→                        Fields:     structBsonFields(structType),
   749→   749→                        Position:   fmt.Sprintf("%s:%d", relPath(mc.pos.Filename), mc.pos.Line),
   750→   750→                })
   751→   751→        }
   752→   752→}

   754→   754→// concreteType returns the concrete type of v, unwrapping *ssa.MakeInterface
   755→   755→// (which boxes a concrete value into an empty interface). For other Value
   756→   756→// kinds it returns v.Type() as-is.
   757→   757→func concreteType(v ssa.Value) types.Type {
   758→   758→        if mi, ok := v.(*ssa.MakeInterface); ok {
   759→   759→                return mi.X.Type()
   760→   760→        }
   761→   761→        return v.Type()
   762→   762→}

   764→   764→// dataArgIndex returns the index in CallCommon.Args of the "data"
   765→   765→// argument (filter, document, or pipeline) for a mongo-op call. For
   766→   766→// static calls to methods, Args[0] is the receiver, Args[1] is ctx,
   767→   767→// Args[2] is the data arg. For invoke-mode calls (interface methods),
   768→   768→// Args[0] is ctx, Args[1] is the data arg.
   769→   769→func dataArgIndex(cc *ssa.CallCommon) int {
   770→   770→        if cc.IsInvoke() {
   771→   771→                return 1
   772→   772→        }
   773→   773→        if f := cc.StaticCallee(); f != nil {
   774→   774→                if f.Signature != nil && f.Signature.Recv() != nil {
   775→   775→                        return 2
   776→   776→                }
   777→   777→                return 1
   778→   778→        }
   779→   779→        return 1
   780→   780→}

   782→   782→// isBsonM reports whether t is the underlying type of bson.M, i.e.
   783→   783→// map[string]interface{} (which is map[string]<any interface>).
   784→   784→// Type aliases (e.g. bson.M itself, which is `type M = map[string]any`)
   785→   785→// are unwrapped first.
   786→   786→func isBsonM(t types.Type) bool {
   787→   787→        // Unwrap type aliases (Go 1.22+ represents `type M = ...` as
   788→   788→        // *types.Alias when GOEXPERIMENT=aliastypes or go >= 1.25).
   789→   789→        if alias, ok := t.(*types.Alias); ok {
   790→   790→                t = alias.Underlying()
   791→   791→        }
   792→   792→        m, ok := t.(*types.Map)
   793→   793→        if !ok {
   794→   794→                return false
   795→   795→        }
   796→   796→        // Key must be string.
   797→   797→        basic, ok := m.Key().(*types.Basic)
   798→   798→        if !ok || basic.Kind() != types.String {
   799→   799→                return false
   800→   800→        }
   801→   801→        // Element must be interface{} (empty interface).
   802→   802→        iface, ok := m.Elem().(*types.Interface)
   803→   803→        if !ok {
   804→   804→                return false
   805→   805→        }
   806→   806→        return iface.NumMethods() == 0
   807→   807→}

   849→   849→// structBsonFields returns the bson tag names of all fields of s
   850→   850→// (recursively flattened through embedded structs).
   851→   851→func structBsonFields(s *types.Struct) []string {
   852→   852→        out := []string{}
   853→   853→        for i := 0; i < s.NumFields(); i++ {
   854→   854→                f := s.Field(i)
   855→   855→                tag := s.Tag(i)
   856→   856→                if f.Embedded() {
   857→   857→                        t := f.Type()
   858→   858→                        if ptr, ok := t.(*types.Pointer); ok {
   859→   859→                                t = ptr.Elem()
   860→   860→                        }
   861→   861→                        if named, ok := t.(*types.Named); ok {
   862→   862→                                t = named.Underlying()
   863→   863→                        }
   864→   864→                        if embedded, ok := t.(*types.Struct); ok {
   865→   865→                                out = append(out, structBsonFields(embedded)...)
   866→   866→                                continue
   867→   867→                        }
   868→   868→                }
   869→   869→                name := parseTag(tag, "bson")
   870→   870→                if name == "" {
   871→   871→                        name = f.Name()
   872→   872→                }
   873→   873→                out = append(out, name)
   874→   874→        }
   875→   875→        return out
   876→   876→}

   878→   878→// parseTag is the same helper used by the struct_flattener tool. It
   879→   879→// uses reflect.StructTag for canonical parsing.
   880→   880→func parseTag(tag, key string) string {
   881→   881→        if tag == "" {
   882→   882→                return ""
   883→   883→        }
   884→   884→        v := reflect.StructTag(tag).Get(key)
   885→   885→        if v == "" {
   886→   886→                return ""
   887→   887→        }
   888→   888→        if i := strings.IndexByte(v, ','); i >= 0 {
   889→   889→                v = v[:i]
   890→   890→        }
   891→   891→        return v
   892→   892→}

   894→   894→// typeString renders a types.Type as a package-qualified name where
   895→   895→// possible. Used for the structType field of struct_type findings.
   896→   896→func typeString(t types.Type) string {
   897→   897→        switch v := t.(type) {
   898→   898→        case *types.Named:
   899→   899→                obj := v.Obj()
   900→   900→                pkg := obj.Pkg()
   901→   901→                if pkg == nil {
   902→   902→                        return obj.Name()
   903→   903→                }
   904→   904→                return pkg.Path() + "." + obj.Name()
   905→   905→        case *types.Pointer:
   906→   906→                return "*" + typeString(v.Elem())
   907→   907→        case *types.Slice:
   908→   908→                return "[]" + typeString(v.Elem())
   909→   909→        default:
   910→   910→                return t.String()
   911→   911→        }
   912→   912→}

   914→   914→// relPath converts an absolute file path to one relative to the current
   915→   915→// working directory (which, by contract, is the Go module root).
   916→   916→func relPath(p string) string {
   917→   917→        if p == "" {
   918→   918→                return ""
   919→   919→        }
   920→   920→        abs, err := filepath.Abs(p)
   921→   921→        if err != nil {
   922→   922→                return p
   923→   923→        }
   924→   924→        wd, err := os.Getwd()
   925→   925→        if err != nil {
   926→   926→                return p
   927→   927→        }
   928→   928→        rel, err := filepath.Rel(wd, abs)
   929→   929→        if err != nil || strings.HasPrefix(rel, "..") {
   930→   930→                return p
   931→   931→        }
   932→   932→        return rel
   933→   933→}
