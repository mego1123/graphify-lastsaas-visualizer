   149→// connectDB loads config and connects to MongoDB, printing helpful guidance on failure.
   150→func connectDB() (*db.MongoDB, *config.Config, func()) {
   151→        env := config.GetEnv()
   152→        cfg, err := config.Load(env)
   153→        if err != nil {
   154→                fmt.Fprintf(os.Stderr, "Error loading config: %v\n\n", err)
   155→                printConfigHelp(env)
   156→                os.Exit(1)
   157→        }
   158→
   159→        database, err := db.NewMongoDB(cfg.Database.URI, cfg.Database.Name)
   160→        if err != nil {
   161→                fmt.Fprintf(os.Stderr, "Error connecting to MongoDB: %v\n\n", err)
   162→                printMongoHelp(env)
   163→                os.Exit(1)
   164→        }
   165→
   166→        cleanup := func() {
   167→                ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
   168→                defer cancel()
   169→                database.Close(ctx)
   170→        }
   171→
   172→        return database, cfg, cleanup
   173→}

   216→func cmdSetup() {
   217→        database, cfg, cleanup := connectDB()
   218→        defer cleanup()
   219→
   220→        ctx, cancel := context.WithTimeout(context.Background(), 30*time.Second)
   221→        defer cancel()
   222→
   223→        // Check if already initialized
   224→        var sys models.SystemConfig
   225→        err := database.SystemConfig().FindOne(ctx, bson.M{}).Decode(&sys)
   226→        if err == nil && sys.Initialized {
   227→                fmt.Println("System is already initialized.")
   228→                fmt.Println()
   229→                fmt.Printf("  Database: %s\n", cfg.Database.Name)
   230→
   231→                // Look up root tenant owner
   232→                var rootTenant models.Tenant
   233→                if err := database.Tenants().FindOne(ctx, bson.M{"isRoot": true}).Decode(&rootTenant); err == nil {
   234→                        var membership models.TenantMembership
   235→                        if err := database.TenantMemberships().FindOne(ctx, bson.M{"tenantId": rootTenant.ID, "role": "owner"}).Decode(&membership); err == nil {
   236→                                var owner models.User
   237→                                if err := database.Users().FindOne(ctx, bson.M{"_id": membership.UserID}).Decode(&owner); err == nil {
   238→                                        fmt.Printf("  Owner:    %s (%s)\n", owner.DisplayName, owner.Email)
   239→                                }
   240→                        }
   241→                }
   242→
   243→                fmt.Println()
   244→                fmt.Println("If you need to change a user's password, use: lastsaas change-password --email <email>")
   245→                os.Exit(0)
   246→        }
   247→
   248→        fmt.Println("=== LastSaaS Initial Setup ===")
   249→        fmt.Println()
   250→
   251→        reader := bufio.NewReader(os.Stdin)
   252→
   253→        orgName := prompt(reader, "Organization name")
   254→        displayName := prompt(reader, "Your name")
   255→        email := prompt(reader, "Email address")
   256→        email = strings.TrimSpace(strings.ToLower(email))
   257→
   258→        if orgName == "" || displayName == "" || email == "" {
   259→                fmt.Fprintln(os.Stderr, "All fields are required.")
   260→                os.Exit(1)
   261→        }
   262→
   263→        passwordService := auth.NewPasswordService()
   264→
   265→        password := promptPassword("Password")
   266→        confirm := promptPassword("Confirm password")
   267→
   268→        if password != confirm {
   269→                fmt.Fprintln(os.Stderr, "Passwords do not match.")
   270→                os.Exit(1)
   271→        }
   272→
   273→        if err := passwordService.ValidatePasswordStrength(password); err != nil {
   274→                fmt.Fprintf(os.Stderr, "Password too weak: %v\n", err)
   275→                fmt.Fprintln(os.Stderr, "Requirements: 10+ characters, uppercase, lowercase, number, special character")
   276→                os.Exit(1)
   277→        }
   278→
   279→        passwordHash, err := passwordService.HashPassword(password)
   280→        if err != nil {
   281→                fmt.Fprintf(os.Stderr, "Failed to hash password: %v\n", err)
   282→                os.Exit(1)
   283→        }
   284→
   285→        now := time.Now()
   286→
   287→        // Create root tenant
   288→        tenant := models.Tenant{
   289→                ID:        primitive.NewObjectID(),
   290→                Name:      strings.TrimSpace(orgName),
   291→                Slug:      "root",
   292→                IsRoot:    true,
   293→                IsActive:  true,
   294→                CreatedAt: now,
   295→                UpdatedAt: now,
   296→        }
   297→        if err := validation.Validate(&tenant); err != nil {
   298→                fmt.Fprintf(os.Stderr, "Tenant validation failed: %v\n", err)
   299→                os.Exit(1)
   300→        }
   301→        if _, err := database.Tenants().InsertOne(ctx, tenant); err != nil {
   302→                fmt.Fprintf(os.Stderr, "Failed to create root tenant: %v\n", err)
   303→                os.Exit(1)
   304→        }
   305→
   306→        // Create owner user
   307→        user := models.User{
   308→                ID:            primitive.NewObjectID(),
   309→                Email:         email,
   310→                DisplayName:   strings.TrimSpace(displayName),
   311→                PasswordHash:  passwordHash,
   312→                AuthMethods:   []models.AuthMethod{models.AuthMethodPassword},
   313→                EmailVerified: true,
   314→                IsActive:      true,
   315→                CreatedAt:     now,
   316→                UpdatedAt:     now,
   317→        }
   318→        if err := validation.Validate(&user); err != nil {
   319→                database.Tenants().DeleteOne(ctx, bson.M{"_id": tenant.ID})
   320→                fmt.Fprintf(os.Stderr, "User validation failed: %v\n", err)
   321→                os.Exit(1)
   322→        }
   323→        if _, err := database.Users().InsertOne(ctx, user); err != nil {
   324→                database.Tenants().DeleteOne(ctx, bson.M{"_id": tenant.ID})
   325→                fmt.Fprintf(os.Stderr, "Failed to create user: %v\n", err)
   326→                os.Exit(1)
   327→        }
   328→
   329→        // Create owner membership
   330→        membership := models.TenantMembership{
   331→                ID:        primitive.NewObjectID(),
   332→                UserID:    user.ID,
   333→                TenantID:  tenant.ID,
   334→                Role:      models.RoleOwner,
   335→                JoinedAt:  now,
   336→                UpdatedAt: now,
   337→        }
   338→        if err := validation.Validate(&membership); err != nil {
   339→                database.Users().DeleteOne(ctx, bson.M{"_id": user.ID})
   340→                database.Tenants().DeleteOne(ctx, bson.M{"_id": tenant.ID})
   341→                fmt.Fprintf(os.Stderr, "Membership validation failed: %v\n", err)
   342→                os.Exit(1)
   343→        }
   344→        if _, err := database.TenantMemberships().InsertOne(ctx, membership); err != nil {
   345→                database.Users().DeleteOne(ctx, bson.M{"_id": user.ID})
   346→                database.Tenants().DeleteOne(ctx, bson.M{"_id": tenant.ID})
   347→                fmt.Fprintf(os.Stderr, "Failed to create membership: %v\n", err)
   348→                os.Exit(1)
   349→        }
   350→
   351→        // Mark system as initialized
   352→        sysConfig := models.SystemConfig{
   353→                ID:            primitive.NewObjectID(),
   354→                Initialized:   true,
   355→                InitializedAt: &now,
   356→                InitializedBy: &user.ID,
   357→                Version:       version.Current,
   358→        }
   359→        if _, err := database.SystemConfig().InsertOne(ctx, sysConfig); err != nil {
   360→                database.TenantMemberships().DeleteOne(ctx, bson.M{"_id": membership.ID})
   361→                database.Users().DeleteOne(ctx, bson.M{"_id": user.ID})
   362→                database.Tenants().DeleteOne(ctx, bson.M{"_id": tenant.ID})
   363→                fmt.Fprintf(os.Stderr, "Failed to mark system as initialized: %v\n", err)
   364→                os.Exit(1)
   365→        }
   366→
   367→        // Send welcome message to the new owner
   368→        welcomeMsg := models.Message{
   369→                ID:        primitive.NewObjectID(),
   370→                UserID:    user.ID,
   371→                Subject:   "Welcome to LastSaaS v" + version.Current,
   372→                Body:      "Your system has been initialized. Welcome to LastSaaS!",
   373→                IsSystem:  true,
   374→                Read:      false,
   375→                CreatedAt: now,
   376→        }
   377→        database.Messages().InsertOne(ctx, welcomeMsg)
   378→
   379→        fmt.Println()
   380→        fmt.Println("System initialized successfully!")
   381→        fmt.Println()
   382→        fmt.Printf("  Organization: %s\n", tenant.Name)
   383→        fmt.Printf("  Owner:        %s (%s)\n", user.DisplayName, user.Email)
   384→        fmt.Printf("  Database:     %s\n", cfg.Database.Name)
   385→        fmt.Println()
   386→        fmt.Println("You can now start the server and log in.")
   387→}

   391→func cmdChangePassword() {
   392→        fs := flag.NewFlagSet("change-password", flag.ExitOnError)
   393→        emailFlag := fs.String("email", "", "User email address (required)")
   394→        fs.Parse(os.Args[2:])
   395→
   396→        if *emailFlag == "" {
   397→                fmt.Fprintln(os.Stderr, "Usage: lastsaas change-password --email <email>")
   398→                os.Exit(1)
   399→        }
   400→
   401→        database, _, cleanup := connectDB()
   402→        defer cleanup()
   403→
   404→        ctx, cancel := context.WithTimeout(context.Background(), 15*time.Second)
   405→        defer cancel()
   406→
   407→        email := strings.TrimSpace(strings.ToLower(*emailFlag))
   408→
   409→        var user models.User
   410→        err := database.Users().FindOne(ctx, bson.M{"email": email}).Decode(&user)
   411→        if err != nil {
   412→                fmt.Fprintf(os.Stderr, "User not found: %s\n", email)
   413→                os.Exit(1)
   414→        }
   415→
   416→        fmt.Printf("Changing password for: %s (%s)\n", user.DisplayName, user.Email)
   417→
   418→        passwordService := auth.NewPasswordService()
   419→
   420→        password := promptPassword("New password")
   421→        confirm := promptPassword("Confirm new password")
   422→
   423→        if password != confirm {
   424→                fmt.Fprintln(os.Stderr, "Passwords do not match.")
   425→                os.Exit(1)
   426→        }
   427→
   428→        if err := passwordService.ValidatePasswordStrength(password); err != nil {
   429→                fmt.Fprintf(os.Stderr, "Password too weak: %v\n", err)
   430→                fmt.Fprintln(os.Stderr, "Requirements: 10+ characters, uppercase, lowercase, number, special character")
   431→                os.Exit(1)
   432→        }
   433→
   434→        passwordHash, err := passwordService.HashPassword(password)
   435→        if err != nil {
   436→                fmt.Fprintf(os.Stderr, "Failed to hash password: %v\n", err)
   437→                os.Exit(1)
   438→        }
   439→
   440→        _, err = database.Users().UpdateOne(ctx,
   441→                bson.M{"_id": user.ID},
   442→                bson.M{"$set": bson.M{
   443→                        "passwordHash": passwordHash,
   444→                        "updatedAt":    time.Now(),
   445→                }},
   446→        )
   447→        if err != nil {
   448→                fmt.Fprintf(os.Stderr, "Failed to update password: %v\n", err)
   449→                os.Exit(1)
   450→        }
   451→
   452→        // Revoke all refresh tokens for this user
   453→        database.RefreshTokens().DeleteMany(ctx, bson.M{"userId": user.ID})
   454→
   455→        fmt.Println("Password updated successfully.")
   456→}

   460→func cmdSendMessage() {
   461→        fs := flag.NewFlagSet("send-message", flag.ExitOnError)
   462→        emailFlag := fs.String("email", "", "Recipient email address (required)")
   463→        messageFlag := fs.String("message", "", "Message body (required)")
   464→        subjectFlag := fs.String("subject", "System Message", "Message subject")
   465→        fs.Parse(os.Args[2:])
   466→
   467→        if *emailFlag == "" || *messageFlag == "" {
   468→                fmt.Fprintln(os.Stderr, "Usage: lastsaas send-message --email <email> --message \"Your message here\"")
   469→                fmt.Fprintln(os.Stderr, "  --subject \"Optional subject\" (default: \"System Message\")")
   470→                os.Exit(1)
   471→        }
   472→
   473→        // Validate inputs
   474→        subject := strings.TrimSpace(*subjectFlag)
   475→        body := strings.TrimSpace(*messageFlag)
   476→
   477→        if !utf8.ValidString(subject) || !utf8.ValidString(body) {
   478→                fmt.Fprintln(os.Stderr, "Invalid characters in subject or message.")
   479→                os.Exit(1)
   480→        }
   481→        if len(subject) > 200 {
   482→                fmt.Fprintln(os.Stderr, "Subject must be 200 characters or less.")
   483→                os.Exit(1)
   484→        }
   485→        if len(body) > 5000 {
   486→                fmt.Fprintln(os.Stderr, "Message must be 5000 characters or less.")
   487→                os.Exit(1)
   488→        }
   489→        if body == "" {
   490→                fmt.Fprintln(os.Stderr, "Message body cannot be empty.")
   491→                os.Exit(1)
   492→        }
   493→
   494→        database, _, cleanup := connectDB()
   495→        defer cleanup()
   496→
   497→        ctx, cancel := context.WithTimeout(context.Background(), 15*time.Second)
   498→        defer cancel()
   499→
   500→        email := strings.TrimSpace(strings.ToLower(*emailFlag))
   501→
   502→        var user models.User
   503→        err := database.Users().FindOne(ctx, bson.M{"email": email}).Decode(&user)
   504→        if err != nil {
   505→                fmt.Fprintf(os.Stderr, "User not found: %s\n", email)
   506→                os.Exit(1)
   507→        }
   508→
   509→        msg := models.Message{
   510→                ID:        primitive.NewObjectID(),
   511→                UserID:    user.ID,
   512→                Subject:   subject,
   513→                Body:      body,
   514→                IsSystem:  true,
   515→                Read:      false,
   516→                CreatedAt: time.Now(),
   517→        }
   518→
   519→        if _, err := database.Messages().InsertOne(ctx, msg); err != nil {
   520→                fmt.Fprintf(os.Stderr, "Failed to send message: %v\n", err)
   521→                os.Exit(1)
   522→        }
   523→
   524→        fmt.Printf("Message sent to %s (%s).\n", user.DisplayName, user.Email)
   525→}

   556→        case "reset":
   557→                if len(os.Args) < 4 {

   562→        default:
   563→                fmt.Fprintf(os.Stderr, "Unknown config subcommand: %s\n", os.Args[2])
   564→                os.Exit(1)
   565→        }

   568→func cmdConfigList() {
   569→        database, _, cleanup := connectDB()
   570→        defer cleanup()
   571→
   572→        ctx, cancel := context.WithTimeout(context.Background(), 10*time.Second)
   573→        defer cancel()
   574→
   575→        cursor, err := database.ConfigVars().Find(ctx, bson.M{})
   576→        if err != nil {
   577→                fmt.Fprintf(os.Stderr, "Failed to query config vars: %v\n", err)
   578→                os.Exit(1)
   579→        }
   580→        defer cursor.Close(ctx)
   581→
   582→        var vars []models.ConfigVar
   583→        if err := cursor.All(ctx, &vars); err != nil {
   584→                fmt.Fprintf(os.Stderr, "Failed to read config vars: %v\n", err)
   585→                os.Exit(1)
   586→        }
   587→
   588→        if jsonOutput {
   589→                printJSON(vars)
   590→                return
   591→        }
   592→
   593→        if len(vars) == 0 {
   594→                fmt.Println("No configuration variables found.")
   595→                return
   596→        }
   597→
   598→        fmt.Printf("%-40s %-10s %-8s %s\n", "NAME", "TYPE", "SYSTEM", "VALUE")

   607→                sys := ""
   608→                if v.IsSystem {
   609→                        sys = "yes"
   610→                }
   611→                fmt.Printf("%-40s %-10s %-8s %s\n", v.Name, v.Type, sys, value)
   612→        }

   677→func cmdConfigReset(name string) {
   678→        // Find the default value from SystemDefaults
   679→        var defaultVal string
   680→        found := false
   681→        for _, def := range configstore.SystemDefaults {
   682→                if def.Name == name {
   683→                        defaultVal = def.Value
   684→                        found = true
   685→                        break
   686→                }
   687→        }
   688→        if !found {
   689→                fmt.Fprintf(os.Stderr, "No system default found for: %s\n", name)
   690→                fmt.Fprintln(os.Stderr, "Only system-defined variables can be reset to defaults.")
   691→                os.Exit(1)
   692→        }
   693→
   694→        database, _, cleanup := connectDB()
   695→        defer cleanup()
   696→
   697→        ctx, cancel := context.WithTimeout(context.Background(), 10*time.Second)
   698→        defer cancel()
   699→
   700→        var v models.ConfigVar
   701→        err := database.ConfigVars().FindOne(ctx, bson.M{"name": name}).Decode(&v)
   702→        if err != nil {
   703→                fmt.Fprintf(os.Stderr, "Config variable not found in database: %s\n", name)
   704→                os.Exit(1)
   705→        }
   706→
   707→        if v.Value == defaultVal {
   708→                fmt.Printf("'%s' is already at its default value.\n", name)
   709→                return
   710→        }
   711→
   712→        _, err = database.ConfigVars().UpdateOne(ctx,
   713→                bson.M{"name": name},
   714→                bson.M{"$set": bson.M{"value": defaultVal, "updatedAt": time.Now()}},
   715→        )
   716→        if err != nil {
   717→                fmt.Fprintf(os.Stderr, "Failed to reset config variable: %v\n", err)
   718→                os.Exit(1)
   719→        }
   720→
   721→        fmt.Printf("Reset '%s' to default value.\n", name)
   722→        fmt.Println("Note: The running server will pick up this change on next cache reload or restart.")
   723→}

   727→func cmdTransferRootOwner() {
   728→        fs := flag.NewFlagSet("transfer-root-owner", flag.ExitOnError)
   729→        emailFlag := fs.String("email", "", "Email of the new root tenant owner (required)")
   730→        fs.Parse(os.Args[2:])
   731→
   732→        if *emailFlag == "" {
   733→                fmt.Fprintln(os.Stderr, "Usage: lastsaas transfer-root-owner --email <new-owner-email>")
   734→                os.Exit(1)
   735→        }
   736→
   737→        database, _, cleanup := connectDB()
   738→        defer cleanup()
   739→
   740→        ctx, cancel := context.WithTimeout(context.Background(), 30*time.Second)
   741→        defer cancel()
   742→
   743→        // Find root tenant
   744→        var rootTenant models.Tenant
   745→        if err := database.Tenants().FindOne(ctx, bson.M{"isRoot": true}).Decode(&rootTenant); err != nil {
   746→                fmt.Fprintln(os.Stderr, "Root tenant not found. Is the system initialized?")
   747→                os.Exit(1)
   748→        }
   749→
   750→        // Find new owner by email
   751→        newEmail := strings.TrimSpace(strings.ToLower(*emailFlag))
   752→        var newOwner models.User
   753→        if err := database.Users().FindOne(ctx, bson.M{"email": newEmail}).Decode(&newOwner); err != nil {
   754→                fmt.Fprintf(os.Stderr, "User not found: %s\n", newEmail)
   755→                os.Exit(1)
   756→        }
   757→
   758→        // Verify new owner is a member of root tenant
   759→        var newMembership models.TenantMembership
   760→        if err := database.TenantMemberships().FindOne(ctx, bson.M{
   761→                "userId":   newOwner.ID,
   762→                "tenantId": rootTenant.ID,
   763→        }).Decode(&newMembership); err != nil {
   764→                fmt.Fprintf(os.Stderr, "User %s is not a member of the root tenant.\n", newEmail)
   765→                fmt.Fprintln(os.Stderr, "They must be added to the root tenant before ownership can be transferred.")
   766→                os.Exit(1)
   767→        }
   768→
   769→        if newMembership.Role == models.RoleOwner {
   770→                fmt.Printf("User %s is already the owner of the root tenant.\n", newEmail)
   771→                os.Exit(0)
   772→        }
   773→
   774→        // Find current owner
   775→        var currentOwnerMembership models.TenantMembership
   776→        if err := database.TenantMemberships().FindOne(ctx, bson.M{
   777→                "tenantId": rootTenant.ID,
   778→                "role":     "owner",
   779→        }).Decode(&currentOwnerMembership); err != nil {
   780→                fmt.Fprintln(os.Stderr, "Could not find current root tenant owner.")
   781→                os.Exit(1)
   782→        }
   783→
   784→        var currentOwner models.User
   785→        if err := database.Users().FindOne(ctx, bson.M{"_id": currentOwnerMembership.UserID}).Decode(&currentOwner); err != nil {
   786→                fmt.Fprintln(os.Stderr, "Could not find current owner user record.")
   787→                os.Exit(1)
   788→        }
   789→
   790→        // Interactive confirmation
   791→        fmt.Println("=== Transfer Root Tenant Ownership ===")
   792→        fmt.Println()
   793→        fmt.Printf("  Current owner: %s (%s)\n", currentOwner.DisplayName, currentOwner.Email)
   794→        fmt.Printf("  New owner:     %s (%s)\n", newOwner.DisplayName, newOwner.Email)
   795→        fmt.Println()
   796→        fmt.Println("The current owner will be demoted to admin.")
   797→        fmt.Println("This is a critical security operation.")
   798→        fmt.Println()
   799→        fmt.Print("Type 'yes' to confirm: ")
   800→
   801→        reader := bufio.NewReader(os.Stdin)
   802→        answer, _ := reader.ReadString('\n')
   803→        answer = strings.TrimSpace(answer)
   804→
   805→        if answer != "yes" {
   806→                fmt.Println("Aborted.")
   807→                os.Exit(0)
   808→        }
   809→
   810→        now := time.Now()
   811→
   812→        // Demote current owner to admin
   813→        _, err := database.TenantMemberships().UpdateOne(ctx,
   814→                bson.M{"_id": currentOwnerMembership.ID},
   815→                bson.M{"$set": bson.M{"role": "admin", "updatedAt": now}},
   816→        )
   817→        if err != nil {
   818→                fmt.Fprintf(os.Stderr, "Failed to demote current owner: %v\n", err)
   819→                os.Exit(1)
   820→        }
   821→
   822→        // Promote new owner
   823→        _, err = database.TenantMemberships().UpdateOne(ctx,
   824→                bson.M{"_id": newMembership.ID},
   825→                bson.M{"$set": bson.M{"role": "owner", "updatedAt": now}},
   826→        )
   827→        if err != nil {
   828→                // Try to rollback
   829→                database.TenantMemberships().UpdateOne(ctx,
   830→                        bson.M{"_id": currentOwnerMembership.ID},
   831→                        bson.M{"$set": bson.M{"role": "owner", "updatedAt": now}},
   832→                )
   833→                fmt.Fprintf(os.Stderr, "Failed to promote new owner: %v\n", err)
   834→                os.Exit(1)
   835→        }
   836→
   837→        // Write system log entry directly
   838→        logEntry := models.SystemLog{
   839→                ID:        primitive.NewObjectID(),
   840→                Severity:  models.LogCritical,
   841→                Message:   fmt.Sprintf("Root tenant ownership transferred from %s (%s) to %s (%s) via CLI", currentOwner.DisplayName, currentOwner.Email, newOwner.DisplayName, newOwner.Email),
   842→                CreatedAt: now,
   843→        }
   844→        database.SystemLogs().InsertOne(ctx, logEntry)
   845→
   846→        fmt.Println()
   847→        fmt.Printf("Root tenant ownership transferred to %s (%s).\n", newOwner.DisplayName, newOwner.Email)
   848→        fmt.Printf("Previous owner %s has been demoted to admin.\n", currentOwner.Email)
   849→}
