   206→   206→def mask_source(src: str) -> str:
   207→   207→    """Replace string literals and comments with spaces.
   208→   208→
   209→   209→    Length and newlines are preserved so character offsets still line up
   210→   210→    with the original source — only the *content* of strings/comments is
   211→   211→    blanked out. This makes brace/paren matching safe.
   212→   212→    """
   213→   213→    out = list(src)
   214→   214→    i = 0
   215→   215→    n = len(src)
   216→   216→    while i < n:
   217→   217→        c = src[i]
   218→   218→        # Line comment
   219→   219→        if c == '/' and i + 1 < n and src[i + 1] == '/':
   220→   220→            while i < n and src[i] != '\n':
   221→   221→                out[i] = ' '
   222→   222→                i += 1
   223→   223→            continue
   224→   224→        # Block comment
   225→   225→        if c == '/' and i + 1 < n and src[i + 1] == '*':
   226→   226→            out[i] = ' '
   227→   227→            out[i + 1] = ' '
   228→   228→            i += 2
   229→   229→            while i < n:
   230→   230→                if src[i] == '*' and i + 1 < n and src[i + 1] == '/':
   231→   231→                    out[i] = ' '
   232→   232→                    out[i + 1] = ' '
   233→   233→                    i += 2
   234→   234→                    break
   235→   235→                if src[i] != '\n':
   236→   236→                    out[i] = ' '
   237→   237→                i += 1
   238→   238→            continue
   239→   239→        # Rune literal '...'
   240→   240→        if c == '\'':
   241→   241→            out[i] = ' '
   242→   242→            i += 1
   243→   243→            while i < n and src[i] not in ('\'', '\n'):
   244→   244→                if src[i] == '\\' and i + 1 < n:
   245→   245→                    out[i] = ' '
   246→   246→                    out[i + 1] = ' '
   247→   247→                    i += 2
   248→   248→                    continue
   249→   249→                out[i] = ' '
   250→   250→                i += 1
   251→   251→            if i < n and src[i] == '\'':
   252→   252→                out[i] = ' '
   253→   253→                i += 1
   254→   254→            continue
   255→   255→        # Double-quoted string "..."
   256→   256→        if c == '"':
   257→   257→            out[i] = ' '
   258→   258→            i += 1
   259→   259→            while i < n and src[i] not in ('"', '\n'):
   260→   260→                if src[i] == '\\' and i + 1 < n:
   261→   261→                    out[i] = ' '
   262→   262→                    out[i + 1] = ' '
   263→   263→                    i += 2
   264→   264→                    continue
   265→   265→                out[i] = ' '
   266→   266→                i += 1
   267→   267→            if i < n and src[i] == '"':
   268→   268→                out[i] = ' '
   269→   269→                i += 1
   270→   270→            continue
   271→   271→        # Back-quoted raw string `...`
   272→   272→        if c == '`':
   273→   273→            out[i] = ' '
   274→   274→            i += 1
   275→   275→            while i < n and src[i] != '`':
   276→   276→                if src[i] != '\n':
   277→   277→                    out[i] = ' '
   278→   278→                i += 1
   279→   279→            if i < n and src[i] == '`':
   280→   280→                out[i] = ' '
   281→   281→                i += 1
   282→   282→            continue
   283→   283→        i += 1
   284→   284→    return ''.join(out)

   287→   287→def find_matching_brace(masked: str, open_pos: int) -> int:
   288→   288→    """Given position of `{` in masked source, return matching `}` position."""
   289→   289→    depth = 0
   290→   290→    i = open_pos
   291→   291→    n = len(masked)
   292→   292→    while i < n:
   293→   293→        c = masked[i]
   294→   294→        if c == '{':
   295→   295→            depth += 1
   296→   296→        elif c == '}':
   297→   297→            depth -= 1
   298→   298→            if depth == 0:
   299→   299→                return i
   300→   300→        i += 1
   301→   301→    return -1

   304→   304→def parse_functions(masked: str) -> list[FuncRange]:
   305→   305→    """Find named top-level function/method declarations and their body ranges."""
   306→   306→    funcs: list[FuncRange] = []
   307→   307→    for m in FUNC_DECL_RE.finditer(masked):
   308→   308→        name = m.group('name')
   309→   309→        # Walk forward from end of match (right after `(`) to find the body `{`.
   310→   310→        i = m.end()
   311→   311→        n = len(masked)
   312→   312→        depth_paren = 1  # we're inside `(` already (match ends after `(`)
   313→   313→        brace_pos = -1
   314→   314→        while i < n:
   315→   315→            c = masked[i]
   316→   316→            if c == '(':
   317→   317→                depth_paren += 1
   318→   318→            elif c == ')':
   319→   319→                depth_paren -= 1
   320→   320→            elif c == '{' and depth_paren == 0:
   321→   321→                brace_pos = i
   322→   322→                break
   323→   323→            i += 1
   324→   324→        if brace_pos < 0:
   325→   325→            continue
   326→   326→        close_pos = find_matching_brace(masked, brace_pos)
   327→   327→        if close_pos < 0:
   328→   328→            continue
   329→   329→        start_line = masked[:m.start()].count('\n') + 1
   330→   330→        end_line = masked[:close_pos].count('\n') + 1
   331→   331→        funcs.append(FuncRange(
   332→   332→            name=name,
   333→   333→            start_line=start_line,
   334→   334→            end_line=end_line,
   335→   335→            is_init=(name == "init"),
   336→   336→        ))
   337→   337→    return funcs

   340→   340→def containing_function(funcs: list[FuncRange], line: int) -> str:
   341→   341→    """Innermost named function whose body contains `line`."""
   342→   342→    best: Optional[FuncRange] = None
   343→   343→    best_span = None
   344→   344→    for f in funcs:
   345→   345→        if f.start_line <= line <= f.end_line:
   346→   346→            span = f.end_line - f.start_line
   347→   347→            if best is None or span < best_span:
   348→   348→                best = f
   349→   349→                best_span = span
   350→   350→    return best.name if best else "<top-level>"

   357→   357→def is_err_ident(name: str) -> bool:
   358→   358→    """Heuristic: does this identifier look like an error variable?"""
   359→   359→    if not name:
   360→   360→        return False
   361→   361→    n = name.lower()
   362→   362→    return (
   363→   363→        n == "err"
   364→   364→        or n == "e"
   365→   365→        or n.startswith("err")
   366→   366→        or n.endswith("err")
   367→   367→        or n.endswith("error")
   368→   368→    )

   371→   371→def classify_err_body(body_masked: str) -> str:
   372→   372→    """Classify the body of `if X != nil { body }`."""
   373→   373→    stripped = body_masked.strip()
   374→   374→    if not stripped:
   375→   375→        return "swallowed"
   376→   376→    if PANIC_RE.search(stripped):
   377→   377→        return "panic_on_error"
   378→   378→    # Terminal CLI error handling: program exits with non-zero status after
   379→   379→    # reporting the error. The error info is preserved and the program fails.
   380→   380→    if TERMINAL_EXIT_RE.search(stripped) or LOG_FATAL_RE.search(stripped):
   381→   381→        return "proper_handling"
   382→   382→    # Test termination — `t.Fatal*` reports the failure and stops the test.
   383→   383→    if TEST_FATAL_RE.search(stripped):
   384→   384→        return "proper_handling"
   385→   385→    # Returns the captured err directly (or wraps it via fmt.Errorf/errors.Wrap)
   386→   386→    if RETURN_ERR_RE.search(stripped):
   387→   387→        return "proper_handling"
   388→   388→    # Returns a *new* error (no `err` in the return expression).
   389→   389→    # The original error is dropped — call it swallowed but note it.
   390→   390→    if RETURN_NEW_ERR_RE.search(stripped):
   391→   391→        return "swallowed"
   392→   392→    has_return = RETURN_ANY_RE.search(stripped) is not None
   393→   393→    # Error reported via HTTP/response helper + return — error info is sent
   394→   394→    # to the client/consumer, but the original `err` is not propagated.
   395→   395→    if has_return and ERROR_REPORT_RE.search(stripped):
   396→   396→        return "logged_only"
   397→   397→    # Bare `return` or `return <non-err>` — original error dropped
   398→   398→    if has_return:
   399→   399→        return "swallowed"
   400→   400→    # No return at all. Check for non-terminal test/HTTP/log reporting:
   401→   401→    # `t.Error*`, `respondWithError`, `http.Error`, etc.
   402→   402→    if TEST_ERROR_RE.search(stripped):
   403→   403→        return "logged_only"
   404→   404→    if ERROR_REPORT_RE.search(stripped):
   405→   405→        return "logged_only"
   406→   406→    # Otherwise: if only logging calls remain, it's logged_only;
   407→   407→    # else the error is genuinely swallowed.
   408→   408→    no_logs = LOG_CALL_RE.sub('', stripped)
   409→   409→    no_logs = re.sub(r'[\s;{}]+', '', no_logs)
   410→   410→    if not no_logs:
   411→   411→        return "logged_only"
   412→   412→    return "swallowed"

   419→   419→def make_snippet(lines: list[str], start: int, end: int, max_lines: int = 6) -> str:
   420→   420→    """Extract snippet from lines (1-indexed) for display."""
   421→   421→    if start < 1:
   422→   422→        start = 1
   423→   423→    if end > len(lines):
   424→   424→        end = len(lines)
   425→   425→    if end < start:
   426→   426→        end = start
   427→   427→    chunk = lines[start - 1:end]
   428→   428→    if len(chunk) > max_lines:
   429→   429→        chunk = chunk[:max_lines] + [f"... ({end - start - max_lines + 1} more lines)"]
   430→   430→    return "\n".join(line.rstrip() for line in chunk)

   433→   433→def extract_outermost_method(stripped: str) -> Optional[tuple[str, str]]:
   434→   434→    """For a statement-form call line, find the outermost call.
   435→   435→
   436→   436→    Returns ``(call_target, method_name)`` where ``call_target`` is the
   437→   437→    full dotted prefix up to and including the method, and ``method_name``
   438→   438→    is the last identifier before the outermost ``(``.
   439→   439→
   440→   440→    Returns ``None`` if the line is not a clean statement-form call
   441→   441→    (i.e. if there's anything significant after the outermost ``)``).
   442→   442→    """
   443→   443→    depth_paren = 0
   444→   444→    depth_bracket = 0
   445→   445→    depth_brace = 0
   446→   446→    last_call_paren = -1
   447→   447→    for i, c in enumerate(stripped):
   448→   448→        if c == '(':
   449→   449→            if (
   450→   450→                depth_paren == 0
   451→   451→                and depth_bracket == 0
   452→   452→                and depth_brace == 0
   453→   453→                and i > 0
   454→   454→            ):
   455→   455→                j = i - 1
   456→   456→                while j >= 0 and stripped[j].isspace():
   457→   457→                    j -= 1
   458→   458→                if j >= 0 and (
   459→   459→                    stripped[j].isalnum()
   460→   460→                    or stripped[j] == '_'
   461→   461→                    or stripped[j] == ')'
   462→   462→                ):
   463→   463→                    last_call_paren = i
   464→   464→            depth_paren += 1
   465→   465→        elif c == ')':
   466→   466→            depth_paren -= 1
   467→   467→            if depth_paren < 0:
   468→   468→                return None
   469→   469→        elif c == '[':
   470→   470→            depth_bracket += 1
   471→   471→        elif c == ']':
   472→   472→            depth_bracket -= 1
   473→   473→            if depth_bracket < 0:
   474→   474→                return None
   475→   475→        elif c == '{':
   476→   476→            depth_brace += 1
   477→   477→        elif c == '}':
   478→   478→            depth_brace -= 1
   479→   479→            if depth_brace < 0:
   480→   480→                return None
   481→   481→    if depth_paren != 0 or depth_bracket != 0 or depth_brace != 0:
   482→   482→        return None  # unbalanced
   483→   483→    if last_call_paren < 0:
   484→   484→        return None
   485→   485→    # Find matching `)` for the outermost call.
   486→   486→    depth = 1
   487→   487→    end_pos = -1
   488→   488→    for i in range(last_call_paren + 1, len(stripped)):
   489→   489→        c = stripped[i]
   490→   490→        if c == '(':
   491→   491→            depth += 1
   492→   492→        elif c == ')':
   493→   493→            depth -= 1
   494→   494→            if depth == 0:
   495→   495→                end_pos = i
   496→   496→                break
   497→   497→    if end_pos < 0:
   498→   498→        return None
   499→   499→    # Anything after the outermost `)` must be just `;` or whitespace.
   500→   500→    rest = stripped[end_pos + 1:].strip()
   501→   501→    rest = rest.lstrip(';').strip()
   502→   502→    if rest:
   503→   503→        return None
   504→   504→    # Walk back over the identifier (method name) before this `(`.
   505→   505→    j = last_call_paren - 1
   506→   506→    while j >= 0 and stripped[j].isspace():
   507→   507→        j -= 1
   508→   508→    end_ident = j + 1
   509→   509→    while j >= 0 and (stripped[j].isalnum() or stripped[j] == '_'):
   510→   510→        j -= 1
   511→   511→    method = stripped[j + 1:end_ident]
   512→   512→    if not method or not method[0].isalpha():
   513→   513→        return None
   514→   514→    # The full call target is everything up to and including `method`.
   515→   515→    target = stripped[:last_call_paren].strip()
   516→   516→    return (target, method)
   517→   517→
   518→   518→
   519→   519→# --------------------------------------------------------------------------- #
   520→   520→# Per-file audit
   521→   521→# --------------------------------------------------------------------------- #
   522→   522→
   523→   523→def audit_file(path: Path, project_root: Path) -> tuple[list[Finding], FileStats]:
   524→   524→    """Audit a single .go file. Returns (findings, stats)."""
   525→   525→    try:
   526→   526→        src = path.read_text(encoding='utf-8', errors='replace')
   527→   527→    except Exception:
   528→   528→        return [], FileStats(path=str(path), is_test=path.name.endswith("_test.go"))
   529→   529→
   530→   530→    try:
   531→   531→        rel_path = str(path.relative_to(project_root))
   532→   532→        if rel_path == ".":
   533→   533→            rel_path = path.name
   534→   534→    except ValueError:
   535→   535→        rel_path = str(path)
   536→   536→
   537→   537→    is_test = path.name.endswith("_test.go")
   538→   538→    masked = mask_source(src)
   539→   539→    lines = src.splitlines()
   540→   540→    funcs = parse_functions(masked)
   541→   541→    stats = FileStats(
   542→   542→        path=rel_path,
   543→   543→        is_test=is_test,
   544→   544→        lines=len(lines),
   545→   545→    )
   546→   546→    findings: list[Finding] = []
   547→   547→
   548→   548→    # Track line numbers inside any `if X != nil { ... }` block so the
   549→   549→    # missing_check scan can skip them (avoid double-counting).
   550→   550→    err_block_lines: set[int] = set()
   551→   551→
   552→   552→    # ----- 1. `if X != nil { ... }` blocks --------------------------------- #
   553→   553→    for m in IF_NIL_RE.finditer(masked):
   554→   554→        # The full match text includes init + condition + `{`.
   555→   555→        match_text = m.group(0)
   556→   556→        # Condition is what's after the last `;` (if any).
   557→   557→        cond = match_text.rsplit(';', 1)[-1] if ';' in match_text else match_text
   558→   558→        if not any(is_err_ident(tok) for tok in re.findall(r'\b\w+\b', cond)):
   559→   559→            continue
   560→   560→
   561→   561→        # `{` is the last character of the match.
   562→   562→        brace_pos = m.end() - 1
   563→   563→        close_pos = find_matching_brace(masked, brace_pos)
   564→   564→        if close_pos < 0:
   565→   565→            continue
   566→   566→
   567→   567→        body_masked = masked[brace_pos + 1:close_pos]
   568→   568→        start_line = masked[:m.start()].count('\n') + 1
   569→   569→        end_line = masked[:close_pos].count('\n') + 1
   570→   570→
   571→   571→        for ln in range(start_line, end_line + 1):
   572→   572→            err_block_lines.add(ln)
   573→   573→
   574→   574→        pattern = classify_err_body(body_masked)
   575→   575→        severity = PATTERN_SEVERITY[pattern]
   576→   576→        func_name = containing_function(funcs, start_line)
   577→   577→        snippet = make_snippet(lines, start_line, end_line)
   578→   578→
   579→   579→        findings.append(Finding(
   580→   580→            file=rel_path,
   581→   581→            line=start_line,
   582→   582→            end_line=end_line,
   583→   583→            function=func_name,
   584→   584→            pattern=pattern,
   585→   585→            severity=severity,
   586→   586→            snippet=snippet,
   587→   587→        ))
   588→   588→
   589→   589→        stats.total_error_sites += 1
   590→   590→        if pattern == "proper_handling":
   591→   591→            stats.proper += 1
   592→   592→        elif pattern == "logged_only":
   593→   593→            stats.logged += 1
   594→   594→        elif pattern == "swallowed":
   595→   595→            stats.swallowed += 1
   596→   596→        elif pattern == "panic_on_error":
   597→   597→            stats.panic += 1
   598→   598→
   599→   599→    # ----- 2. `result, _ := F(...)` ignored errors ------------------------- #
   600→   600→    for m in IGNORED_RE.finditer(masked):
   601→   601→        line_num = masked[:m.start()].count('\n') + 1
   602→   602→        line_text = lines[line_num - 1] if line_num <= len(lines) else ""
   603→   603→        # Skip `for k, _ := range m` and similar non-error discards.
   604→   604→        if re.search(r'\brange\b', line_text):
   605→   605→            continue
   606→   606→        # Skip if line is inside an err block (would be double-counted)
   607→   607→        if line_num in err_block_lines:
   608→   608→            continue
   609→   609→        func_name = containing_function(funcs, line_num)
   610→   610→        snippet = make_snippet(lines, line_num, line_num)
   611→   611→        findings.append(Finding(
   612→   612→            file=rel_path,
   613→   613→            line=line_num,
   614→   614→            end_line=line_num,
   615→   615→            function=func_name,
   616→   616→            pattern="ignored",
   617→   617→            severity=PATTERN_SEVERITY["ignored"],
   618→   618→            snippet=snippet,
   619→   619→            note="error explicitly discarded with `_`",
   620→   620→        ))
   621→   621→        stats.total_error_sites += 1
   622→   622→        stats.ignored += 1
   623→   623→
   624→   624→    # ----- 3. missing_check: statement-form calls -------------------------- #
   625→   625→    for line_idx, raw_line in enumerate(lines, start=1):
   626→   626→        if line_idx in err_block_lines:
   627→   627→            continue
   628→   628→        # Strip trailing `// ...` comment (but keep `/* */` — rare on stmts).
   629→   629→        code_part = re.sub(r'//.*$', '', raw_line).rstrip()
   630→   630→        stripped = code_part.strip()
   631→   631→        if not stripped:
   632→   632→            continue
   633→   633→        if stripped.startswith(('defer ', 'defer(', 'go ', 'go(')):
   634→   634→            continue
   635→   635→        if CONTROL_KW_RE.match(stripped):
   636→   636→            continue
   637→   637→        # Skip assignments — `result := F()` (single return) is hard to judge
   638→   638→        # without type info. Focus on bare statement-form calls.
   639→   639→        if ':=' in stripped or re.match(r'^[\w.,\s\[\]{}*<>-]+=\s*\S', stripped):
   640→   640→            continue
   641→   641→        # Skip lines with nolint-style comments
   642→   642→        if '//nolint' in raw_line or '//nosec' in raw_line:
   643→   643→            continue
   644→   644→        low = raw_line.lower()
   645→   645→        if '//ok' in low or '// ok' in low:
   646→   646→            continue
   647→   647→
   648→   648→        call = extract_outermost_method(stripped)
   649→   649→        if call is None:
   650→   650→            continue
   651→   651→        call_target, method = call
   652→   652→        # Only flag method calls (with a `.` receiver). Free-function calls
   653→   653→        # like `Write(...)` or `Close(...)` are almost always custom helpers
   654→   654→        # whose signature we can't infer, so skip them to avoid noise.
   655→   655→        if '.' not in call_target:
   656→   656→            continue
   657→   657→        if method not in ERROR_RETURNING_NAMES:
   658→   658→            continue
   659→   659→
   660→   660→        func_name = containing_function(funcs, line_idx)
   661→   661→        snippet = make_snippet(lines, line_idx, line_idx)
   662→   662→        findings.append(Finding(
   663→   663→            file=rel_path,
   664→   664→            line=line_idx,
   665→   665→            end_line=line_idx,
   666→   666→            function=func_name,
   667→   667→            pattern="missing_check",
   668→   668→            severity=PATTERN_SEVERITY["missing_check"],
   669→   669→            snippet=snippet,
   670→   670→            note=f"statement-form call to known error-returning '{call_target}()'",
   671→   671→        ))
   672→   672→        stats.total_error_sites += 1
   673→   673→        stats.missing += 1
   674→   674→
   675→   675→    return findings, stats
   676→   676→

   679→   679→# Aggregation / reporting

   681→   681→
   682→   682→def collect_files(root: Path) -> list[Path]:
   683→   683→    """All .go files under root, excluding vendor / generated dirs.
   684→   684→
   685→   685→    `root` may be a single .go file or a directory.
   686→   686→    """
   687→   687→    skip_dirs = {"vendor", "node_modules", ".git", "graphify-out", "testdata"}
   688→   688→    if root.is_file():
   689→   689→        return [root] if root.suffix == ".go" else []
   690→   690→    out: list[Path] = []
   691→   691→    for p in root.rglob("*.go"):
   692→   692→        if any(part in skip_dirs for part in p.parts):
   693→   693→            continue
   694→   694→        out.append(p)
   695→   695→    return sorted(out)
   696→   696→
   697→   697→
   698→   698→def audit_project(root: Path) -> dict:
   699→   699→    """Audit all .go files under root and return the combined report dict."""
   700→   700→    files = collect_files(root)
   701→   701→    all_findings: list[Finding] = []
   702→   702→    all_stats: list[FileStats] = []
   703→   703→
   704→   704→    for path in files:
   705→   705→        findings, stats = audit_file(path, root)
   706→   706→        all_stats.append(stats)
   707→   707→        # Only include detailed findings from NON-test files in the main list.
   708→   708→        if not stats.is_test:
   709→   709→            all_findings.extend(findings)
   710→   710→
   711→   711→    # ----- summary --------------------------------------------------------- #
   712→   712→    non_test_stats = [s for s in all_stats if not s.is_test]
   713→   713→    test_stats = [s for s in all_stats if s.is_test]
   714→   714→
   715→   715→    def totals(stats: list[FileStats]) -> dict:
   716→   716→        return {
   717→   717→            "files": len(stats),
   718→   718→            "lines": sum(s.lines for s in stats),
   719→   719→            "total_error_sites": sum(s.total_error_sites for s in stats),
   720→   720→            "proper": sum(s.proper for s in stats),
   721→   721→            "logged": sum(s.logged for s in stats),
   722→   722→            "swallowed": sum(s.swallowed for s in stats),
   723→   723→            "ignored": sum(s.ignored for s in stats),
   724→   724→            "missing": sum(s.missing for s in stats),
   725→   725→            "panic": sum(s.panic for s in stats),
   726→   726→        }
   727→   727→
   728→   728→    nt = totals(non_test_stats)
   729→   729→    tt = totals(test_stats)
   730→   730→    pct_proper = (
   731→   731→        100.0 * nt["proper"] / nt["total_error_sites"]
   732→   732→        if nt["total_error_sites"] else 0.0
   733→   733→    )
   734→   734→
   735→   735→    # Most problematic files (non-test) by total problematic count.
   736→   736→    problematic_files = sorted(
   737→   737→        [s for s in non_test_stats if s.problematic > 0],
   738→   738→        key=lambda s: (-s.problematic, -s.total_error_sites, s.path),
   739→   739→    )[:15]
   740→   740→
   741→   741→    pattern_breakdown = Counter(f.pattern for f in all_findings)
   742→   742→    severity_breakdown = Counter(f.severity for f in all_findings)
   743→   743→
   744→   744→    return {
   745→   745→        "root": str(root),
   746→   746→        "summary": {
   747→   747→            "total_files": len(files),
   748→   748→            "non_test_files": len(non_test_stats),
   749→   749→            "test_files": len(test_stats),
   750→   750→            "non_test": nt,
   751→   751→            "test": tt,
   752→   752→            "pct_proper_handled": round(pct_proper, 2),
   753→   753→        },
   754→   754→        "pattern_breakdown": dict(pattern_breakdown),
   755→   755→        "severity_breakdown": dict(severity_breakdown),
   756→   756→        "most_problematic_files": [
   757→   757→            {
   758→   758→                "file": s.path,
   759→   759→                "problematic_count": s.problematic,
   760→   760→                "total_error_sites": s.total_error_sites,
   761→   761→                "breakdown": {
   762→   762→                    "swallowed": s.swallowed,
   763→   763→                    "ignored": s.ignored,
   764→   764→                    "missing_check": s.missing,
   765→   765→                    "logged_only": s.logged,
   766→   766→                    "panic_on_error": s.panic,
   767→   767→                    "proper_handling": s.proper,
   768→   768→                },
   769→   769→            }
   770→   770→            for s in problematic_files
   771→   771→        ],
   772→   772→        "findings": [asdict(f) for f in all_findings],
   773→   773→        "file_stats": [asdict(s) for s in all_stats],
   774→   774→    }
