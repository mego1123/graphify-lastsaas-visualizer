   147→def __getattr__(name: str) -> str:
   148→    # PEP 562: lazily resolve the legacy always-on section constants for external
   149→    # importers (e.g. the install-string tests). In-module code calls _always_on()
   150→    # directly; nothing is read at import time, so a missing block can no longer
   151→    # brick the CLI on `import graphify.__main__` (#1121 follow-up).
   152→    base = _ALWAYS_ON_ALIASES.get(name)
   153→    if base is not None:
   154→        return _always_on(base)
   155→    raise AttributeError(f"module {__name__!r} has no attribute {name!r}")

   164→def _check_skill_version(skill_dst: Path) -> None:
   165→    """Warn if the installed skill is from an older graphify version."""
   166→    version_file = skill_dst.parent / ".graphify_version"
   167→    try:
   168→        if not version_file.exists():
   169→            return
   170→    except OSError:
   171→        return
   172→    try:
   173→        skill_exists = skill_dst.exists()
   174→    except OSError:
   175→        return
   176→    if not skill_exists:
   177→        print("  warning: skill dir exists but SKILL.md is missing. Run 'graphify install' to repair.", file=sys.stderr)
   178→        return
   179→    # A progressive SKILL.md links to its references/ sidecar. If the body points
   180→    # at references/ but the dir is gone (manual delete, partial upgrade), the
   181→    # on-demand fragments won't load — flag it for repair.
   182→    try:
   183→        body = skill_dst.read_text(encoding="utf-8")
   184→    except OSError:
   185→        body = ""
   186→    if "references/" in body and not (skill_dst.parent / "references").exists():
   187→        print("  warning: skill references/ sidecar is missing. Run 'graphify install' to repair.", file=sys.stderr)
   188→    try:
   189→        installed = version_file.read_text(encoding="utf-8").strip()
   190→    except OSError:
   191→        return
   192→    if installed != __version__:
   193→        if _version_tuple(installed) > _version_tuple(__version__):
   194→            # The skill on disk is NEWER than the running package. `graphify install`
   195→            # writes the package's OWN (older) bundled skill and re-stamps the version,
   196→            # so following the old "run install" advice would silently DOWNGRADE the
   197→            # skill. The real fix is to upgrade the package (#1568). Common for a stale
   198→            # `uv tool` CLI, or a contributor whose dev checkout stamped a newer skill.
   199→            print(
   200→                f"  warning: skill is from graphify {installed}, but the package is "
   201→                f"{__version__} (older). Upgrade the package "
   202→                f"(e.g. 'uv tool upgrade graphifyy' or 'pip install -U graphifyy'); "
   203→                f"running 'graphify install' would downgrade the skill.",
   204→                file=sys.stderr,
   205→            )
   206→        else:
   207→            print(f"  warning: skill is from graphify {installed}, package is {__version__}. Run 'graphify install' to update.", file=sys.stderr)

   210→def _version_tuple(version: str) -> tuple[int, ...]:
   211→    """Parse a version string into a comparable integer tuple (``0.9.2`` -> ``(0, 9, 2)``).
   212→
   213→    Reads the leading digits of each dot-segment, so pre/post-release suffixes
   214→    (``1.0.0rc1``) compare by their numeric core. A non-numeric or empty segment
   215→    becomes 0, so a malformed stamp degrades to a conservative comparison rather
   216→    than raising.
   217→    """
   218→    parts: list[int] = []
   219→    for segment in str(version).split("."):
   220→        digits = ""
   221→        for ch in segment:
   222→            if ch.isdigit():
   223→                digits += ch
   224→            else:
   225→                break
   226→        parts.append(int(digits) if digits else 0)
   227→    return tuple(parts)

   448→def _silence_broken_pipe() -> None:
   449→    """Handle a downstream reader that closed the pipe early. Redirect stdout to
   450→    devnull so the interpreter's shutdown flush does not raise a second time, then
   451→    exit 0 — the reader (head, `Select-Object -First N`, `sed q`) has what it needs."""
   452→    try:
   453→        devnull = os.open(os.devnull, os.O_WRONLY)
   454→        os.dup2(devnull, sys.stdout.fileno())
   455→    except Exception:
   456→        pass
   457→    sys.exit(0)

   460→def main() -> None:
   461→    """Console entry point. Wraps the CLI so that when a downstream consumer closes
   462→    stdout early, graphify treats it as success instead of crashing with an
   463→    unhandled write-to-closed-pipe error and exit 255 — which made CI wrappers and
   464→    agent harnesses read a successful query as a command failure (#1807)."""
   465→    try:
   466→        _run_cli()
   467→        # Flush explicitly, inside the guard. Piped stdout is block-buffered, so a
   468→        # small fully-buffered output would otherwise only flush at interpreter
   469→        # shutdown — outside this try — where a reader that closed the pipe surfaces
   470→        # as a noisy "Exception ignored on flushing sys.stdout" and a nonzero exit.
   471→        sys.stdout.flush()
   472→    except BrokenPipeError:
   473→        _silence_broken_pipe()
   474→    except OSError as exc:
   475→        # Windows surfaces a write to a closed pipe as OSError(EINVAL) rather than
   476→        # BrokenPipeError; EPIPE is the POSIX form when it slips past the above.
   477→        if getattr(exc, "errno", None) in (errno.EPIPE, errno.EINVAL):
   478→            _silence_broken_pipe()
   479→        else:
   480→            raise

   483→def _run_cli() -> None:
   484→    for _stream in (sys.stdout, sys.stderr):
   485→        if _stream is not None and hasattr(_stream, "reconfigure"):
   486→            try:
   487→                _stream.reconfigure(encoding="utf-8", errors="replace")
   488→            except Exception:
   489→                pass
   490→    # Check all known skill install locations for a stale version stamp.
   491→    # Skip during install/uninstall (hook writes trigger a fresh check anyway).
   492→    # Skip during hook-check — it runs on every editor tool use and must be silent.
   493→    # Deduplicate paths so platforms sharing the same install dir don't warn twice.
   494→    _silent_cmds = {"install", "uninstall", "hook-check", "hook-guard"}
   495→    if not any(arg in _silent_cmds for arg in sys.argv):
   496→        # Resolve each platform's real user-scope destination so per-platform
   497→        # overrides (gemini, opencode, devin, antigravity, amp) check the dir
